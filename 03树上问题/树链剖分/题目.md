#define endl '\n'
const int mod = 1e9 + 7;

int p;

int n;

void dfs1(int x) {
    siz[x] = 1;
    dep[x] = dep[parent[x]] + 1;
    for (auto y : ver[x]) {
        if (y == parent[x]) continue;
        parent[y] = x;
        dfs1(y);
        siz[x] += siz[y];
        if (siz[y] > siz[son[x]]) {

    dfn[x] = ++cntd;

    if (son[x]) dfs2(son[x], up);

            x = parent[top[x]];
        } else {
            y = parent[top[y]];
        }
    }

}

    return dep[x] + dep[y] - 2 * dep[lca(x, y)];
}
void work(int root = 1) { // 在此初始化
    dfs1(root);
    dfs2(root, root);

/*
用途：线段树（区间加法、区间和查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间加 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 内部空间分配为 4 倍节点大小，单点修改可使用 add(p, p, v)。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> add_tag;

    SegTree(int size) : n(size), sum((size + 1) << 2, 0), add_tag((size + 1) << 2, 0) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 懒标记处理
    void lazy(int i, T v, int len) {
        sum[i] += v * len;
        add_tag[i] += v;
    }

    // 向下传递懒标记
    void down(int i, int ln, int rn) {
        if (add_tag[i] != 0) {
            lazy(i << 1, add_tag[i], ln);
            lazy(i << 1 | 1, add_tag[i], rn);
            add_tag[i] = 0;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
        add_tag[i] = 0;
    }

    // 区间加：[jobl, jobr] 增加 jobv
    void add(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            add(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void add(int l, int r, T v) { add(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

void ap() {
	int m, r; cin >> n >> m >> r >> p;

    ver.resize(n + 1);
    siz.resize(n + 1);
    dep.resize(n + 1);

    top.resize(n + 1);

    dfn.resize(n + 1), seg.resize(n + 1);

    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    for (int i = 0; i < n - 1; i++) {
        int x, y; cin >> x >> y;

        add(x, y);
    }

    work(r);

    SegTree<int> tree(n);
    vector<int> t(n);
    for (int i = 1; i <= n; i++) {
        t[i] = a[seg[i]];
    }
    tree.build(t);

    for (int i = 0; i < m; i++) {
        int op; cin >> op;

        if (op == 1) {
            int x, y, z; cin >> x >> y >> z;
            
            while (top[x] != top[y]) {
                if (dep[top[x]] <= dep[top[y]]) {
                    tree.add(dfn[top[y]], dfn[y], z);
                    y = parent[top[y]];
                } else {
                    tree.add(dfn[top[x]], dfn[x], z);
                    x = parent[top[x]];
                }
            }
            int u = dfn[x], v = dfn[y];
            tree.add(min(u, v), max(u, v), z);
            
        } else if (op == 2) {
            int x, y; cin >> x >> y;

            int ans = 0;
            while (top[x] != top[y]) {
                if (dep[top[x]] <= dep[top[y]]) {
                    ans =  (ans + tree.query(dfn[top[y]], dfn[y])) % p;
                    y = parent[top[y]];
                } else {

/*
用途：线段树（区间重置 / 区间赋值、区间和查询、区间最大值查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间重置 O(log n)，区间和查询 O(log n)，区间最大值查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(l, r, v)     : 将区间 [l, r] 内所有元素重置/赋值为 v。
  - query_sum(l, r)     : 查询区间 [l, r] 的累加和。
  - query_max(l, r)     : 查询区间 [l, r] 的最大值。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 区间重置操作会直接覆盖历史所有数值与未下传标记。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> mx;
    vector<T> change;
    vector<bool> update_tag;

    SegTree(int size) : n(size),
                        sum((size + 1) << 2, 0),
                        mx((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总：维护区间和与区间最大值
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
        mx[i] = max(mx[i << 1], mx[i << 1 | 1]);
    }

    // 懒标记处理（重置覆盖）：区间赋值后，最大值为 v，区间和为 v * len
    void lazy(int i, T v, int len) {
        sum[i] = v * len;
        mx[i] = v;
        change[i] = v;
        update_tag[i] = true;
    }

    // 向下传递懒标记
    void down(int i, int ln, int rn) {
        if (update_tag[i]) {
            lazy(i << 1, change[i], ln);
            lazy(i << 1 | 1, change[i], rn);
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            mx[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query_sum(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query_sum(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query_sum(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 区间查询：[jobl, jobr] 的区间最大值
    T query_max(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return mx[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = numeric_limits<T>::lowest();
        if (jobl <= mid) {
            ans = max(ans, query_max(jobl, jobr, l, mid, i << 1));
        }
        if (jobr > mid) {
            ans = max(ans, query_max(jobl, jobr, mid + 1, r, i << 1 | 1));
        }
        return ans;
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    T query_sum(int l, int r) { return query_sum(l, r, 1, n, 1); }
    T query_max(int l, int r) { return query_max(l, r, 1, n, 1); }
};

void ap() {
	int n; cin >> n;

    HLD hld(n);
    for (int i = 0; i < n - 1; i++) {
        int a, b; cin >> a >> b;

        hld.add(a, b);
    }

    hld.work();

    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];

    vector<int> t(n + 1);
    for (int i = 1; i <= n; i++) {
        t[i] = w[hld.seg[i]];
    }

    SegTree<int> tree(n);
    tree.build(t);
    
    int q; cin >> q;
    while (q--) {
        string s; cin >> s;
        if (s == "CHANGE") {
            int u, t; cin >> u >> t;

            int x = hld.dfn[u];
            tree.update(x, x, t);
            
        } else if (s == "QMAX") {
            int u, v; cin >> u >> v;

            int ans = -1e9;
            
            while (hld.top[u] != hld.top[v]) {
                if (hld.dep[hld.top[u]] > hld.dep[hld.top[v]]) {
                    ans = max(ans, tree.query_max(hld.dfn[hld.top[u]], hld.dfn[u]));
                    u = hld.parent[hld.top[u]];
                } else {
                    ans = max(ans, tree.query_max(hld.dfn[hld.top[v]], hld.dfn[v]));
                    v = hld.parent[hld.top[v]];
                }
            }

            int x = min(hld.dfn[u], hld.dfn[v]);
            int y = max(hld.dfn[v], hld.dfn[u]);
            ans = max(ans, tree.query_max(x, y));

            cout << ans << endl;
            
        } else {
            int u, v; cin >> u >> v;

            int ans = 0;
            while (hld.top[u] != hld.top[v]) {
                if (hld.dep[hld.top[u]] > hld.dep[hld.top[v]]) {
                    ans += tree.query_sum(hld.dfn[hld.top[u]], hld.dfn[u]);
                    u = hld.parent[hld.top[u]];
                } else {
                    ans += tree.query_sum(hld.dfn[hld.top[v]], hld.dfn[v]);
                    v = hld.parent[hld.top[v]];
                }
            }

            int x = min(hld.dfn[u], hld.dfn[v]);
            int y = max(hld.dfn[v], hld.dfn[u]);
            ans += tree.query_sum(x, y);

            cout << ans << endl;
        }
    }
    
    
	return ;
}

	int T = 1;

    void down(int i, int ln, int rn) {
        if (update_tag[i]) {
            lazy(i << 1, change[i], ln);
            lazy(i << 1 | 1, change[i], rn);
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    T query(int l, int r) { return query(l, r, 1, n, 1); }

template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> change;
    vector<bool> update_tag;
    vector<int> lcolor, rcolor;

    SegTree(int size) : n(size),
                        lcolor((size + 1) << 2, 0),
                        rcolor((size + 1) << 2, 0),
                        sum((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
        if (rcolor[i << 1] == lcolor[i << 1 | 1]) sum[i]--;

        lcolor[i] = lcolor[i << 1];
        rcolor[i] = rcolor[i << 1 | 1];
    }

    // 懒标记处理（重置覆盖）
    void lazy(int i, T v) {
        sum[i] = 1;
        lcolor[i] = v;
        rcolor[i] = v;
        change[i] = v;
        update_tag[i] = true;
    }

    // 向下传递懒标记
    void down(int i) {
        if (update_tag[i]) {
            lazy(i << 1, change[i]);
            lazy(i << 1 | 1, change[i]);
            change[i] = 0;
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = 1;
            lcolor[i] = w[seg[l]];
            rcolor[i] = w[seg[l]];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv);
            return;
        }
        int mid = (l + r) >> 1;
        down(i);
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：修复了分支互斥与交界判断
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i);
        if (jobr <= mid) {
            return query(jobl, jobr, l, mid, i << 1);
        } else if (jobl > mid) {
            return query(jobl, jobr, mid + 1, r, i << 1 | 1);
        } else {
            int ans = query(jobl, jobr, l, mid, i << 1) + query(jobl, jobr, mid + 1, r, i << 1 | 1);
            if (rcolor[i << 1] == lcolor[i << 1 | 1]) ans--;
            return ans;
        }
    }

    int q_point(int jobi, int l, int r, int i) {
        if (l == r) {
            return lcolor[i];
        }
        down(i);
        int mid = (l + r) >> 1;
        if (jobi <= mid) {
            return q_point(jobi, l, mid, i << 1);
        } else {
            return q_point(jobi, mid + 1, r, i << 1 | 1);
        }
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
    int q_point(int jobi) { return q_point(jobi, 1, n, 1); }
};

    void build(const A &a) {
        fill(bit.begin(), bit.end(), T{});
        for (int i = 1; i <= n; ++i) {
            bit[i] += a[i];
            int j = i + (i & -i);
            if (j <= n) bit[j] += bit[i];
        }
    }

    //将位置 x 增加 v
    void modify(int x, const T &v) {
        assert(1 <= x && x <= n);
        for (; x <= n; x += x & -x) bit[x] += v;
    }

    void update(int l, int r, const T &v) {
        if (l > r) return;
        assert(1 <= l && r <= n);
        modify(l, v);
        if (r < n) modify(r + 1, -v);
    }

    //查询前缀和 [1, x]
    T ask(int x) const {
        x = min(x, n);
        T res{};
        for (; x > 0; x -= x & -x) res += bit[x];
        return res;
    }

    T ask(int l, int r) const {
        if (l > r) return T{};
        assert(1 <= l && r <= n);
        return ask(r) - ask(l - 1);
    }

    int askKth(const T &k) const {
        if (!(k > T{})) return 1;
        int pos = 0;
        T pre{};
        int stp = 1;
        while ((stp << 1) <= n) stp <<= 1;
        for (; stp; stp >>= 1) {
            int nxt = pos + stp;
            if (nxt <= n && pre + bit[nxt] < k) {
                pre += bit[nxt];
                pos = nxt;
            }
        }
        return pos + 1;
    }
};
