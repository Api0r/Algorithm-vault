
- 求lca的最佳写法，预处理时间复杂度 O(N) ；单次查询 O(log⁡N) ，常数较小。lca不需要用到dfn，seg，cntd
- 利用dfn序建立线段树，可范围修改、查询 - **子树，两点的路径**的信息

- 边权变点权，边权继承到dfn大的节点上。求两点路径的边权和，即**路径上的点权和减去lca的点权**。也可以用线段树维护

```c++
//top[x] = up : 节点x所在重链的头节点是up
//seg[cntd] = x : dfn编号为cntd的节点为x

struct HLD {
    int n, cntd;
    vector<vector<int>> ver;
    vector<int> siz, dep;
    vector<int> top, son, parent;
    vector<int> dfn, seg;

    HLD(int n) : n(n) {
        ver.resize(n + 1);
        siz.resize(n + 1);
        dep.resize(n + 1);

        top.resize(n + 1);
        son.resize(n + 1);
        parent.resize(n + 1);

        dfn.resize(n + 1);
        seg.resize(n + 1);
        cntd = 0;
    }
    void add(int x, int y) { // 建立双向边
        ver[x].push_back(y);
        ver[y].push_back(x);
    }
    void dfs1(int x) {
        siz[x] = 1;
        dep[x] = dep[parent[x]] + 1;
        for (auto y : ver[x]) {
            if (y == parent[x]) continue;
            parent[y] = x;
            dfs1(y);
            siz[x] += siz[y];
            if (siz[y] > siz[son[x]]) {
                son[x] = y;
            }
        }
    }
    // 来到节点x，节点x所在重链的头节点是up
    void dfs2(int x, int up) {
        top[x] = up;
        dfn[x] = ++cntd;
        seg[cntd] = x;
        if (son[x]) dfs2(son[x], up);
        for (auto y : ver[x]) {
            if (y == parent[x] || y == son[x]) continue;
            dfs2(y, y);
        }
    }
    int lca(int x, int y) {
        while (top[x] != top[y]) {
            if (dep[top[x]] > dep[top[y]]) {
                x = parent[top[x]];
            } else {
                y = parent[top[y]];
            }
        }
        return dep[x] < dep[y] ? x : y;
    }
    int clac(int x, int y) { // 查询两点间距离
        return dep[x] + dep[y] - 2 * dep[lca(x, y)];
    }
    void work(int root = 1) { // 在此初始化
        dfs1(root);
        dfs2(root, root);
    }
};

```
