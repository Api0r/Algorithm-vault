```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int n;
vector<vector<array<int, 2>>> g;
vector<int> pot, sz;

int sum = 0;
int mn = 1e9;
int root = 0;

vector<int> path;

map<array<int, 2>, int> wight;

void dfs(int x, int fa) {
    sz[x] = pot[x];
    int mx = 0;

    for (auto & [y, w] : g[x]) if (y != fa) {
        dfs(y, x);

        sz[x] += sz[y];

        mx = max(mx, sz[y]);
    }

    mx = max(mx, sum - sz[x]);

    if (mx < mn) {
        mn = mx;
        root = x;
    }
}

void dfs1(int x, int fa) {
    // cout << "x : " << x << "  path[x] : " << path[x] << endl;
    // cout << path[fa] << ' ' << (wight.count({x, fa}) ? wight[{x, fa}] : 0) << endl;
    
    path[x] = path[fa] + (wight.count({x, fa}) ? wight[{x, fa}] : 0);
    
    // cout << path[x] << endl;
    // cout << endl;

    for (auto & [y, w] : g[x]) if (y != fa) {
        dfs1(y, x);
    }
}

void ap() {
	cin >> n;
    pot.assign(n + 1, 0), sz.assign(n + 1, 0), g.assign(n + 1, {});

    for (int i = 1; i <= n; i++) cin >> pot[i], sum += pot[i];
    
    for (int i = 0; i < n - 1; i++) {
        int x, y, w; cin >> x >> y >> w;

        g[x].push_back({y, w});
        g[y].push_back({x, w});

        wight[{x, y}] = w;
        wight[{y, x}] = w;
    }

    dfs(1, 0);

    path.assign(n + 1, 0);

    dfs1(root, 0);

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        ans += path[i] * pot[i];
    }

    // cout << "root : " << root << endl;
    // for (int i = 1; i <= n; i++) cout << path[i] << ' ';
    // cout << endl;

    
    cout << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
