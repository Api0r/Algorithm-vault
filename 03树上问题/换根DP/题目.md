```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int n;
string s;

vector<vector<int>> g;
int dp[200005];//以x为根的子树，在删除x的同时，所需删除的节点数
int diff[200005];//以x为根的子树，子树内含异色边的分支数

int ans;

void dfs1(int x, int fa) {
    dp[x] = 1;
    diff[x] = 0;

    for (int & y : g[x]) {
        if (y != fa) {
            dfs1(y, x);

            //如果向y的这一个分支中有异色边，或者(x, y) 为异色边
            //那么x向y这一条分支，就有异色边
            if (diff[y] > 0 || (s[x] != s[y])) {
                diff[x]++;
            }

            if (diff[y] > 0) {
                dp[x] += dp[y];
            }
        }
    }
}

void dfs2(int x, int fa) {
    ans = min(ans, dp[x]);

    //因为x本身就是算的是整棵树的，但是其他节点都是算的子树，所以将剩下的节点的信息补全
    for (int & y : g[x]) {
        if (y != fa) {

            if (diff[x] - (diff[y] || s[x] != s[y])) {
                dp[y] += dp[x] - dp[y];
            }

            //这里顺序不能换，因为会修改diff[y]
            if ((s[x] != s[y]) || diff[x] - (diff[y] || s[x] != s[y])) {
                diff[y]++;
            }

            dfs2(y, x);
        }
    }
}

        g[u].push_back(v);
        g[v].push_back(u);
    }

    dfs1(1, 0);
```

```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

template <class T> struct Rerooting {
    struct Edge {
        int to, id;
    };
    using Merge = function<T(const T &, const T &)>;
    using Transfer = function<T(const T &, int, int, int)>;

    int n, m = 0;
    T id;
    Merge mg;
    Transfer trn;
    vector<vector<Edge>> e;
    vector<int> fa, pe, ord;
    vector<T> bas, dn, up, ans;

    Rerooting(int n, T id, Merge mg, Transfer trn) : n(n), id(id), mg(move(mg)), trn(move(trn)), e(n + 1), fa(n + 1), pe(n + 1), bas(n + 1, id), dn(n + 1, id), up(n + 1, id), ans(n + 1, id) {}

    int add(int u, int v) {
        int id = m++;
        e[u].push_back({v, id});
        e[v].push_back({u, id});
        return id;
    }

    void work(int rt = 1) {
        fill(fa.begin(), fa.end(), 0);
        fill(pe.begin(), pe.end(), -1);
        ord.clear();
        ord.push_back(rt);
        for (int i = 0; i < ord.size(); ++i) {
            int u = ord[i];
            for (auto [v, id] : e[u]) {
                if (v == fa[u]) continue;
                fa[v] = u;
                pe[v] = id;
                ord.push_back(v);
            }
        }

        fill(dn.begin(), dn.end(), id);
        for (int i = ord.size(); i--;) {
            int u = ord[i];
            dn[u] = bas[u];
            for (auto [v, id] : e[u]) {
                if (fa[v] != u) continue;
                dn[u] = mg(dn[u], trn(dn[v], v, u, id));
            }
        }

        fill(up.begin(), up.end(), id);
        for (int u : ord) {
            int deg = e[u].size();
            vector<T> get(deg), pre(deg + 1, id), suf(deg + 1, id);
            for (int i = 0; i < deg; ++i) {
                auto [v, id] = e[u][i];
                get[i] = (v == fa[u]) ? up[u] : trn(dn[v], v, u, id);
                pre[i + 1] = mg(pre[i], get[i]);
            }
            for (int i = deg - 1; i >= 0; --i) suf[i] = mg(get[i], suf[i + 1]);
            ans[u] = mg(bas[u], pre[deg]);

            for (int i = 0; i < deg; ++i) {
                auto [v, id] = e[u][i];
                if (fa[v] != u) continue;
                T oth = mg(bas[u], mg(pre[i], suf[i + 1]));
                up[v] = trn(oth, u, v, id);
            }
        }
    }
};

void ap() {
    int n; cin >> n;
    string s; cin >> s;
    s = "0" + s;

    using p = array<int, 2>; // p[0]: 该方向分支内的异色边总数, p[1]: 进入该分支所需的节点数
    p id = {0, 0};

    auto mg = [](const p &a, const p &b) -> p {
        return {a[0] + b[0], a[1] + b[1]};
    };

    auto trn = [&](const p &d, int v, int u, int) -> p {
        // d[0] 是以 v 为根的子树内的异色边数
        // 加上边 (u, v) 是否异色，得到向 v 看去的整个分支的异色边总数
        int total_edge_diff = d[0] + (s[u] != s[v] ? 1 : 0);

        // 关键判断：v 内部（不含边 (u, v)）是否存在异色边？
        // 存在则说明 v 分支不纯，必须进入 v，花费为 d[1]（d[1] 已包含 bas[v] 的 1 点）；
        // 否则留在外部作为纯色连通块，花费为 0。
        int cost = (d[0] > 0 ? d[1] : 0);

        return {total_edge_diff, cost};
    };

    Rerooting<p> tr(n, id, mg, trn);

    for (int i = 1; i <= n; i++) {
        tr.bas[i] = {0, 1};
    }

    int total_diff = 0;
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        tr.add(u, v);
        if (s[u] != s[v]) total_diff++;
    }

    if (total_diff == 0) {
        cout << 1 << endl;
        return;
    }

    tr.work(1);

    int ans = 1e9;
    for (int i = 1; i <= n; i++) {
        ans = min(ans, tr.ans[i][1]);
    }

    cout << ans << endl;
    return;
}
```

这段代码非常精妙，它用**换根 DP** 实现了这一目标。

## 一、核心数学与图论逻辑

题目的要求是：选择一个**非空连通子图 $S$** 删除，使得剩余的每个连通块都是单色的。

换一个角度看：

- 假设我们**强制包含某个节点 $x \in S$**。

- 节点 $x$ 的周围连接着若干个分支（子树）。

- 对于 $x$ 的任意一个邻居分支 $y$：

    - 如果分支 $y$ 的**内部存在异色边**，那么这个分支绝对不可能作为一个“纯色连通块”完整保留在外面。因为 $x \in S$，$S$ 必须是连通的，所以 $S$ **必须顺着边 $(x, y)$ 延伸进节点 $y$**。

    - 如果分支 $y$ 的**内部完全同色**（没有任何异色边），那么就算连接边 $(x, y)$ 本身是异色边，由于端点 $x$ 已被删掉，边 $(x, y)$ 也随之切断，分支 $y$ 留在外面就是一个合法的单色连通块。因此 **$S$ 根本不需要延伸进 $y$**。

**关键结论：**

当指定节点 $x \in S$ 时，$S$ 只需要延伸进那些**其内部含有异色边**的相邻分支。

因此：

- 只要以 $x$ 为根，递归地把所有“内部有异色边”的子树代价累加起来，再加上 $x$ 自身（计 1），就能得到包含 $x$ 的最小连通块大小。

- 最终答案就是对整棵树中所有节点 $x$ 作为根时所得大小取最小值：$\min_{x=1}^n dp[x]$。

## 二、数组与变量含义

在代码中，对于以 $x$ 为根的有根树（或在 DFS 过程中的局部有向子树）：

1. **`diff[x]`**：

    - 并不是简单记录异色边的具体条数，而是记录：**以 $x$ 为根的子树中，有多少个子分支“内部含有异色边”**。

    - **`diff[x] > 0`（即 `diff[x] != 0`）等价于：以 $x$ 为根的子树内部至少存在一条异色边。**

    - **`diff[x] == 0` 等价于：以 $x$ 为根的整个子树内所有节点颜色完全一致（纯色子树）。**

2. **`dp[x]`**：

    - 设 $x \in S$，为了切断子树 $x$ 内部的所有异色边，$S \cap T_x$ 最少需要包含的节点数量。

    - 基础值是自身：$dp[x] = 1$。

## 三、`Dfs1`（自底向上预处理树形 DP）

```c++
void Dfs1(int x, int fa) {
    diff[x] = 0;
    dp[x] = 1;
    for (const auto& y : graph[x]) {
        if (y != fa) {
            Dfs1(y, x);
            // 判定分支 y 是否能让以 x 为根的子树产生内部异色边
            if (diff[y] || s[x] != s[y]) {
                diff[x]++;
            }
            // 如果 y 内部自身有异色边，连通块 S 必须延伸到 y
            if (diff[y]) {
                dp[x] += dp[y];
            }
        }
    }
}
```

- **`if (diff[y] || s[x] != s[y]) diff[x]++;`**

    - 分支 $y$ 内部如果有异色边（`diff[y] > 0`），或者边 $(x, y)$ 本身异色（`s[x] != s[y]`），那么在以 $x$ 为根的整棵子树中，朝向 $y$ 的这一方向就**含有异色边**。

    - 因此 $x$ 的“含异色边分支数”加 1。只要 `diff[x] > 0`，就代表 $T_x$ 不纯。

- **`if (diff[y]) dp[x] += dp[y];`**

    - 注意这里判断的是 **`diff[y]`** 而不是 `s[x] != s[y]`！

    - 如果 `diff[y] == 0`，即便 $s[x] \ne s[y]$，因为 $x$ 已经属于 $S$ 被删除了，边 $(x, y)$ 也会被切断，此时以 $y$ 为根的子树内全部同色，可以安全留在外部，**不需要选入 $y$**。

    - 只有当 `diff[y] > 0`（$y$ 内部自己有冲突）时，$S$ 才不得不选入 $y$，累加 $dp[y]$。

## 四、`Dfs2`（自顶向下换根 DP）

`Dfs2` 的作用是将根节点从 $x$ 转移到它的邻居 $y$。

```c++
void Dfs2(int x, int fa) {
    res = min(res, dp[x]); // 当前以 x 为全树根的答案
    for (const auto& y : graph[x]) {
        if (y != fa) {
            // 1. 从 x 排除掉 y 之后，剩下的外部部分内部是否有异色边？
            //    diff[x] 减去 y 这一支对它的贡献 (s[x] != s[y] || diff[y])
            if (diff[x] - (s[x] != s[y] || diff[y])) {
                // 如果除了 y 之外，x 这一侧仍有异色边，那么当根换成 y 时，连通块必须延伸到 x
                dp[y] += dp[x] - (diff[y] ? dp[y] : 0);
            }

            // 2. 更新 y 视角的含异色边分支数
            //    如果边 (x, y) 异色，或者 x 这一侧内部含有异色边
            if (s[x] != s[y] || diff[x] - (s[x] != s[y] || diff[y])) {
                diff[y]++;
            }

            // 递归进入 y 为根的状态
            Dfs2(y, x);
        }
    }
}
```

换根计算的拆解：

1. **`diff[x] - (s[x] != s[y] || diff[y])`**：

    - 这是在计算：在原树中把子树 $y$ 及其边 $(x, y)$ 删去后，**剩下以 $x$ 为根的部分，其内部是否含有异色边**。

    - 原本 $x$ 统计的“有异色边分支数”是 `diff[x]`，其中分支 $y$ 贡献了 `(s[x] != s[y] || diff[y])`（即 0 或 1）。两者相减如果 $> 0$，说明 $x$ 侧内部依然有异色边。

2. **更新 `dp[y]`**：

    - 只有当 $x$ 侧内部含有异色边时，以 $y$ 为中心选连通块时才**必须延伸到 $x$**。

    - 延伸到 $x$ 需加上 $x$ 侧的花费：`dp[x] - (diff[y] ? dp[y] : 0)`（即从总花费 `dp[x]` 中减去原本 $y$ 分支贡献给它的花费）。

3. **更新 `diff[y]`**：

    - 对于 $y$ 来说，$x$ 方向也是它的一个邻居分支。

    - 如果边 $(x, y)$ 异色，或者 $x$ 侧内部有异色边，那么 $x$ 方向就构成了 $y$ 的一个“含异色边分支”，执行 `diff[y]++`。

## 五、算法总结

| **维度** | **分析** |
| --- | --- |
| **状态含义** | `diff[u]` 维护子树内含异色边的分支数（用于判断是否为单色树）；`dp[u]` 维护连通块覆盖代价。 |
| **转移策略** | 仅当邻居分支内部**自身存在异色边**时才将连通块延伸进去，否则留在外部作为单色块。 |
| **时空复杂度** | 树只被常数次前序/后序遍历，时间复杂度 **$O(n)$**，空间复杂度 **$O(n)$**，运行效率达到最优。 |

## 六、换根 DP 的深入拆解

`Dfs2` 的本质是**把全树的“根”从父节点 $x$ 顺着边推向子节点 $y$**。

在 `Dfs1` 结束时，只有节点 $1$ 的 `dp[1]` 和 `diff[1]` 包含了全树的所有方向。对于任意子节点 $y$：

- `dp[y]` 和 `diff[y]` 里**只包含了以 $y$ 为根的子树的信息**；

- 缺少了从 $y$ 往上看、$x$ 这一侧（即除 $y$ 子树以外整棵树剩余部分）的信息。

换根的核心就是：**算出“从原整树中扣除子树 $y$ 后，$x$ 这一侧的独立信息”，然后把它当成 $y$ 的一个新子节点合入 $y$**。

### 1. 核心拆解：$x$ 这一侧扣掉 $y$ 后还剩什么？

当全树根从 $x$ 转移到 $y$ 时，$x$ 变成了 $y$ 的一个子节点。

首先需要回答：**在不考虑 $y$ 及其连接边 $(x, y)$ 的前提下，$x$ 所在的这一侧内部有没有异色边？**

原代码用下式进行判定：

```c++
diff[x] - (s[x] != s[y] || diff[y])
```

#### 1. 为什么是减去 `(s[x] != s[y] || diff[y])`？

在 `Dfs1` 中，计算 `diff[x]` 时是这样累加的：

```c++
if (diff[y] || s[x] != s[y]) {
    diff[x]++;
}
```

这表示：如果 $y$ 内部自身有异色边（`diff[y] != 0`），或者边 $(x, y)$ 是异色的（`s[x] != s[y]`），$y$ 就会作为 $x$ 的一个“含异色边分支”，让 `diff[x]` 的计数增加了 $1$。

因此：

- `(s[x] != s[y] || diff[y])` 的值要么是 $1$（说明当初 $y$ 给 `diff[x]` 贡献了 1），要么是 $0$。

- `diff[x] - (s[x] != s[y] || diff[y])` 就是**排除掉 $y$ 这一侧的贡献后，$x$ 周围其它分支中“含异色边分支”的数量**。

#### 2. 减出来的结果意味着什么？

- **若该差值 $> 0$**：

    说明 $x$ 往其他方向看去，至少还有一个分支内部含有异色边。

    换句话说，**$x$ 这一侧内部是不纯色的（含有异色边）**。

- **若该差值 $== 0$**：

    说明 $x$ 的其他方向全是纯同色分支。

    换句话说，**$x$ 这一侧内部完全单色（没有异色边）**。

### 2. 第一步：更新 `dp[y]`

```c++
if (diff[x] - (s[x] != s[y] || diff[y])) {
    dp[y] += dp[x] - (diff[y] ? dp[y] : 0);
}
```

现在要把根换到 $y$。$y$ 正在考虑：**我要不要把连通块 $S$ 延伸进节点 $x$？**

1. **判断条件**：

    - 只有当 **$x$ 这一侧内部存在异色边** 时（即 `diff[x] - (s[x] != s[y] || diff[y]) > 0`），$x$ 侧才不能作为一个完整的纯色块留在外面，连通块 $S$ 必须延伸到 $x$。

    - 如果 $x$ 这一侧内部完全没有异色边，即使边 $(x, y)$ 本身异色，由于 $y \in S$，边 $(x, y)$ 已经被切断，整个 $x$ 侧可以留存为单色块，连通块**不需要延伸进 $x$**，所以什么都不做。

2. **累加代价**：

    - 如果必须延伸进 $x$，$x$ 侧需要的点数是多少？

    - 节点 $x$ 原来的总点数是 `dp[x]`。如果当初在以 $x$ 为根时，连通块也延伸进了 $y$（即 `diff[y] != 0`），那么 `dp[x]` 里是包含了 `dp[y]` 的。

    - 现在 $y$ 成了新的根节点，不能重复计算自己子树的代价，所以要减去当初的贡献：

        `dp[x] - (diff[y] ? dp[y] : 0)`。

    - 这部分纯粹属于 $x$ 侧的代价，直接累加给 `dp[y]`。

### 3. 第二步：更新 `diff[y]`

```c++
if (s[x] != s[y] || diff[x] - (s[x] != s[y] || diff[y])) {
    diff[y]++;
}
```

现在要更新 $y$ 的“含异色边分支数”，即把 $x$ 视作 $y$ 的一个子分支：

- 回忆 `Dfs1` 中分支被计入的规则：

    **当且仅当该分支内部有异色边，或者连接该分支的边是异色的。**

- 对应到 $x$ 这一侧分支：

    - 连接边 $(x, y)$ 是否异色：`s[x] != s[y]`

    - $x$ 侧内部是否有异色边：`diff[x] - (s[x] != s[y] || diff[y]) > 0`

- 只要这两个条件满足其一，在以 $y$ 为全树根的视角下，$x$ 就是一个“含异色边分支”，因此执行 `diff[y]++`。

### 4. 为什么代码里不需要回溯（还原现场）？

通常的换根 DP 需要在访问完子节点后把数据还原（比如 `dp[y] -= ...`），但在这份代码中：

```c++
Dfs2(y, x);
```

之后并没有任何回溯操作，为什么依然是正确的？

因为树是一层一层往下走的（从根走向叶子）：

1. 循环里更新的是子节点 `y` 的数据（`dp[y]` 和 `diff[y]`），**父节点 `x` 的值在循环中完全没有被修改**。

2. 每次计算不同的孩子 $y_1, y_2, \dots$ 时，用的都是 $x$ 原始的 `dp[x]` 和 `diff[x]`。

3. 到了孩子 $y$ 这一层，$y$ 的值已经被彻底更新成了“以 $y$ 为全树根”的全局数据，接下来继续带着这套数据去更新 $y$ 的下层孩子即可。

因此不仅代码极短，而且省去了保存现场和回溯的开销。
