
# 1. 选或不选

[P1352 没有上司的舞会 - 洛谷](https://www.luogu.com.cn/problem/P1352#ide)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<int> d(n + 1, 1);
    
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int x, y; cin >> x >> y;

        g[x].push_back(y);
        g[y].push_back(x);

        d[x] = 0;
    }

    int root = 1;
    for (int i = 1; i <= n; i++) {
        if (d[i]) {
            root = i;
            break;
        }
    }

    vector<int> no(n + 1), yes(n + 1);

    auto dfs = [&](auto&& dfs, int x, int fa) -> void {
        yes[x] = a[x];
        no[x] = 0;

        for (int & y : g[x]) if (y != fa) {
            dfs(dfs, y, x);

            yes[x] += no[y];
            no[x] += max(yes[y], no[y]);
        }
    }; dfs(dfs, root, 0);

    cout << max(yes[root], no[root]) << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
