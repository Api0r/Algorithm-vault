缩点之后，枚举每一个中心点

```c++
struct DSU {
    vector<int> fa, p, e, f;

    DSU(int n) {
        fa.resize(n + 1);
        iota(fa.begin(), fa.end(), 0);
        p.resize(n + 1, 1);
        e.resize(n + 1);
        f.resize(n + 1);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 设x是y的祖先
        if (x == y) f[get(x)] = 1;
        x = get(x), y = get(y);
        e[x]++;
        if (x == y) return false;
        if (x < y) swap(x, y); // 将编号小的合并到大的上
        fa[y] = x;
        f[x] |= f[y], p[x] += p[y], e[x] += e[y];
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    bool F(int x) { // 判断连通块内是否存在自环
        return f[get(x)];
    }
    int size(int x) { // 输出连通块中点的数量
        return p[get(x)];
    }
    int E(int x) { // 输出连通块中边的数量
        return e[get(x)];
    }
};

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    vector<vector<int>> g(n + 1);
    
    DSU dsu(n);
    
    vector<array<int, 2>> b(n - 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        
        b[i] = {u, v};
        if (a[u] == a[v]) dsu.merge(u, v);
    }
    
    //缩点之后建图
    //颜色相同的边合成一个连通块，不需要连边
    //颜色不同的连通块需要连边
    //可证明只连find的代表节点不会出现重边， 仍是一颗树
    for (int i = 0; i < n - 1; i++) if (a[b[i][0]] != a[b[i][1]]) {
        int x = dsu.get(b[i][0]), y = dsu.get(b[i][1]);
        
        g[x].push_back(y);
        g[y].push_back(x);
    }
    
    vector<int> cnt(n + 1);// col[i]_cnt
    int ans = 0;
    //枚举中心点，可证明时间复杂度
    for (int x = 1; x <= n; x++) if (x == dsu.get(x)) {
        for (int y : g[x]) {
            ans += cnt[a[y]] * dsu.size(y);
            cnt[a[y]] += dsu.size(y);
        }
        
        for (int y : g[x]) cnt[a[y]] = 0;
    }
    
    cout << ans << endl;
    
	return ;
}
```

这两个性质是该算法兼具**正确性**与**极致高效性**的核心基石，以下给出严谨的数学证明。

### 一、建图不会出现重边的证明（反证法）

**待证命题**：

  

设原图 $T = (V, E)$ 是一棵树（$\vert{}V\vert{} = n, \vert{}E\vert{} = n - 1$）。将同色相邻点合并后得到若干互不相交的连通子图 $C_1, C_2, \dots, C_k$。

  

对于任意两个不同的连通块 $C_A$ 与 $C_B$（$A \neq B$），原图 $T$ 中连接 $C_A$ 的某个点与 $C_B$ 的某个点的跨块边**至多只有一条**。

  

**证明**：

  

假设命题不成立，即存在两个不同的连通块 $C_A$ 和 $C_B$，它们之间在原图 $T$ 中存在至少两条不同的边：

  

- 边 $e_1 = (u_1, v_1)$，其中 $u_1 \in C_A, v_1 \in C_B$
    
      
    
- 边 $e_2 = (u_2, v_2)$，其中 $u_2 \in C_A, v_2 \in C_B$（且 $e_1 \neq e_2$）
    
      
    

由于连通块的定义：

  

1. **$C_A$ 内部是连通的**：因此在 $C_A$ 内部必然存在一条从 $u_1$ 到 $u_2$ 的简单路径 $P_A$（若 $u_1 = u_2$，则 $P_A$ 为单个点）。且 $P_A$ 上的所有节点均属于 $C_A$。
    
      
    
2. **$C_B$ 内部是连通的**：因此在 $C_B$ 内部必然存在一条从 $v_2$ 到 $v_1$ 的简单路径 $P_B$（若 $v_2 = v_1$，则 $P_B$ 为单个点）。且 $P_B$ 上的所有节点均属于 $C_B$。
    
      
    

由于 $C_A \cap C_B = \emptyset$（两个连通块互不相交），我们可以将上述路径与两条跨块边拼接起来：

  

$$u_1 \xrightarrow{P_A} u_2 \xrightarrow{e_2} v_2 \xrightarrow{P_B} v_1 \xrightarrow{e_1} u_1$$

- 因为 $e_1 \neq e_2$，且 $C_A$ 与 $C_B$ 无公共点，该闭合环路中除端点相接外不经过重复边。
    
      
    
- 这说明在原图 $T$ 中构造出了一个**简单环**。
    
      
    

这与题目条件 **“保证给出的是一棵树（无向无环连通图）”** 产生直接矛盾。

  

**结论**：

  

反设不成立，任意两个不同连通块之间在原树中**至多只存在一条边**。因此缩点后直接加边不会产生任何重边。

  

### 二、时间复杂度严格为 $O(n \alpha(n))$ 的证明（代数与平摊分析）

程序的运行瓶颈主要在两处：**建图与并查集操作**、**嵌套循环统计与清空**。

  

#### 1. 边的数量与度数守恒

设缩点后的图为 $G' = (V', E')$：

  

- 缩点后的每一个超级节点对应原树中的一个同色连通块，节点集合为 $V'$（显然 $\vert{}V'\vert{} \le n$）。
    
      
    
- 缩点后的边集 $E'$ 是由原树中所有满足 $a[u] \neq a[v]$ 的树边收缩而成。因为原树的总边数仅为 $n - 1$，所以：
    
      
    
    $$\vert{}E'\vert{} \le n - 1$$
    
- 根据图论基本握手定理（所有顶点的度数和等于边数的两倍）：
    
      
    
    $$\sum_{x \in V'} \deg_{G'}(x) = 2 \vert{}E'\vert{} \le 2(n - 1) < 2n$$
    
    其中 $\deg_{G'}(x) = \vert{}g[x]\vert{}$ 表示缩点后节点 $x$ 的邻居数量。
    
      
    

#### 2. 外层循环与内层统计的平摊分析

代码中统计部分的核心逻辑为：

```c++
for (int x = 1; x <= n; x++) if (x == dsu.get(x)) {
    for (int y : g[x]) {
        // O(1) 桶更新与乘法累加
        vec.push_back(col);
    }
    for (int c : vec) cnt[c] = 0; // O(1) 单点清空
    vec.clear();
}
```

- **外层判断**：`x` 从 $1$ 到 $n$ 遍历，每次执行一次 `dsu.get(x)`，总共执行 $n$ 次并查集查找，耗时为：
    
      
    
    $$T_{\text{outer}} = O(n \alpha(n))$$
    
- **内层统计遍历**：仅当 $x$ 是超级点（根节点 $x \in V'$）时才进入。内层循环 `for (int y : g[x])` 的执行次数恰好是当前块的度数 $\vert{}g[x]\vert{}$。
    
      
    
    所有合法连通块的内层总迭代次数为：
    
      
    
    $$T_{\text{inner}} = \sum_{x \in V'} \vert{}g[x]\vert{} \le 2(n - 1)$$
    
- **桶的清空开销**：在处理节点 $x$ 时，压入 `vec` 的元素个数同样等于 $\vert{}g[x]\vert{}$。随后的清空循环 `for (int c : vec)` 执行次数为 $\vert{}vec\vert{} = \vert{}g[x]\vert{}$。
    
      
    
    在全部超级点处理完毕的过程中，所有清空操作的总迭代次数为：
    
      
    
    $$T_{\text{clear}} = \sum_{x \in V'} \vert{}vec\vert{} = \sum_{x \in V'} \vert{}g[x]\vert{} \le 2(n - 1)$$
    

#### 3. 各阶段耗时汇总

1. **同色并查集合并**：遍历 $n - 1$ 条边，执行 $O(n)$ 次并查集操作，耗时 $O(n \alpha(n))$；
    
      
    
2. **跨块建边**：遍历 $n - 1$ 条边，执行常数次查找与 `push_back`，耗时 $O(n \alpha(n))$；
    
      
    
3. **统计答案与局部清零**：
    
      
    
    $$T_{\text{total\_calc}} = T_{\text{outer}} + T_{\text{inner}} + T_{\text{clear}} = O(n \alpha(n)) + O(n) + O(n) = O(n \alpha(n))$$
    

**结论**：

整个算法的总体时间复杂度严格为 **$O(n \alpha(n))$**。

在 $n = 2 \times 10^5$ 时，$\alpha(n) \le 4$，基本操作总数不超过 $2 \times 10^6$ 次，计算机可在约 **0.05 秒** 内执行完毕，绝对不会超时。



**BFS的写法**

```c++
struct DSU {
    vector<int> fa, p, e, f;

    DSU(int n) {
        fa.resize(n + 1);
        iota(fa.begin(), fa.end(), 0);
        p.resize(n + 1, 1);
        e.resize(n + 1);
        f.resize(n + 1);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 设x是y的祖先
        if (x == y) f[get(x)] = 1;
        x = get(x), y = get(y);
        e[x]++;
        if (x == y) return false;
        if (x < y) swap(x, y); // 将编号小的合并到大的上
        fa[y] = x;
        f[x] |= f[y], p[x] += p[y], e[x] += e[y];
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    bool F(int x) { // 判断连通块内是否存在自环
        return f[get(x)];
    }
    int size(int x) { // 输出连通块中点的数量
        return p[get(x)];
    }
    int E(int x) { // 输出连通块中边的数量
        return e[get(x)];
    }
};

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    vector<vector<int>> g(n + 1);
    
    DSU dsu(n);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        
        g[u].push_back(v);
        g[v].push_back(u);
        if (a[u] == a[v]) dsu.merge(u, v);
    }
    
    int ans = 0;
    
    vector<int> vis(n + 1);
    vector<int> cnt(n + 1);
    auto bfs = [&](int u) -> void {
        queue<int> q;
        q.push(u);
        vis[u] = 1;
        
        vector<int> vec;
        while (!q.empty()) {
            int x = q.front();
            q.pop();
            
            for (int y : g[x]) {
                if (a[x] != a[y]) {
                    y = dsu.get(y);
                    ans += dsu.size(y) * cnt[a[y]];
                    cnt[a[y]] += dsu.size(y);
                    
                    vec.push_back(a[y]);
                } else if (!vis[y]) {
                    q.push(y);
                    vis[y] = 1;
                }
                
            }
        }
        for (int x : vec) cnt[x] = 0;
    };
    
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) bfs(i);
    }
    cout << ans << endl;
    
	return ;
}
```
