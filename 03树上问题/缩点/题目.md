struct DSU {
    vector<int> fa, p, e, f;

    DSU(int n) {
        fa.resize(n + 1);
        iota(fa.begin(), fa.end(), 0);
        p.resize(n + 1, 1);
        e.resize(n + 1);
        f.resize(n + 1);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 设x是y的祖先
        if (x == y) f[get(x)] = 1;
        x = get(x), y = get(y);
        e[x]++;
        if (x == y) return false;
        if (x < y) swap(x, y); // 将编号小的合并到大的上
        fa[y] = x;
        f[x] |= f[y], p[x] += p[y], e[x] += e[y];
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    bool F(int x) { // 判断连通块内是否存在自环
        return f[get(x)];
    }
    int size(int x) { // 输出连通块中点的数量
        return p[get(x)];
    }
    int E(int x) { // 输出连通块中边的数量
        return e[get(x)];
    }
};

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    vector<vector<int>> g(n + 1);
    
    DSU dsu(n);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        
        g[u].push_back(v);
        g[v].push_back(u);
        if (a[u] == a[v]) dsu.merge(u, v);
    }
    
    int ans = 0;
    
    vector<int> vis(n + 1);
    vector<int> cnt(n + 1);
    auto bfs = [&](int u) -> void {
        queue<int> q;
        q.push(u);
        vis[u] = 1;
        
        vector<int> vec;
        while (!q.empty()) {
            int x = q.front();
            q.pop();
            
            for (int y : g[x]) {
                if (a[x] != a[y]) {
                    y = dsu.get(y);
                    ans += dsu.size(y) * cnt[a[y]];
                    cnt[a[y]] += dsu.size(y);
                    
                    vec.push_back(a[y]);
                } else if (!vis[y]) {
                    q.push(y);
                    vis[y] = 1;
                }
                
            }
        }
        for (int x : vec) cnt[x] = 0;
    };
    
    for (int i = 1; i <= n; i++) {
        if (!vis[i]) bfs(i);
    }
    cout << ans << endl;
    
	return ;
}

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    vector<vector<int>> g(n + 1);
    
    DSU dsu(n);
    
    vector<array<int, 2>> b(n - 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        
        b[i] = {u, v};
        if (a[u] == a[v]) dsu.merge(u, v);
    }
    
    //缩点之后建图
    //颜色相同的边合成一个连通块，不需要连边
    //颜色不同的连通块需要连边
    //可证明只连find的代表节点不会出现重边， 仍是一颗树
    for (int i = 0; i < n - 1; i++) if (a[b[i][0]] != a[b[i][1]]) {
        int x = dsu.get(b[i][0]), y = dsu.get(b[i][1]);
        
        g[x].push_back(y);
        g[y].push_back(x);
    }
    
    vector<int> cnt(n + 1);// col[i]_cnt
    int ans = 0;
    //枚举中心点，可证明时间fa
    for (int x = 1; x <= n; x++) if (x == dsu.get(x)) {
        for (int y : g[x]) {
            ans += cnt[a[y]] * dsu.size(y);
            cnt[a[y]] += dsu.size(y);
        }
        
        for (int y : g[x]) cnt[a[y]] = 0;
    }
    
    cout << ans << endl;
    
	return ;
}
