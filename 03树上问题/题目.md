```c++
int n;
int ans1 = 0, ans2 = 0;

vector<vector<int>> g;
vector<int> sub, dep;

void dfs(int i, int fa, int d) {
    sub[i] = 1;
    dep[i] = d;
    
    for (int x : g[i]) {
        if (x == fa) continue;
        
        dfs(x, i, d + 1);
        sub[i] += sub[x];
        
        int cnt = sub[x];
        int other = n - cnt - dep[x];
        ans1 += cnt * other;
    }
    
    ans2 += sub[i] - 1;
    return;
}

void ap() {
	cin >> n;
    
    g.assign(n, vector<int>());
    sub.assign(n, 0), dep.assign(n, 0);
    ans1 = 0, ans2 = 0;
    
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        u--, v--;
        
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    dfs(0, -1, 0);
    
    cout << (ans1 + ans2) % MOD << endl;
    
    
	return ;
}
```

# 4.贪心构造 + 思维

[F-小芳的最小生成树_牛客小白月赛139](https://ac.nowcoder.com/acm/contest/141882/F)

- **条件一：$T$ 是图 $G$ 唯一的最小生成树**。 根据最小生成树的**环切性质**，如果我们在树 $T$ 的两个节点 $a, b$ 之间加一条权值为 $w_{new}$ 的新边，那么 $a, b$ 和它们在树上的唯一路径就会构成一个环。要想让 $T$ 是**唯一**的最小生成树，新加的这条边的权值 $w_{new}$ **必须严格大于**树上 $a$ 到 $b$ 路径上的每一条边的权值（即 $w_{new} > \max_{e \in path(a,b)} w_e$）。否则，用 $w_{new}$ 替换路径上的最大边，就能得到另一棵权值和更小或相等的生成树。
    
- **条件二：删去 $T$ 中任意一条边 $e=(x,y)$（设其权值为 $v$），新图 $G'$ 的最小生成树权值和恰好增加 $1$**。 删去树边 $e=(x,y)$ 后，原来的树 $T$ 会分裂成两个连通块。为了重新把它们连起来构成生成树，我们需要在横跨这两个连通块的所有新加边中，挑一条权值最小的边补上去。 原本总权值是 $x$，删掉 $e$ 少了 $v$，要求新生成树权值和为 $x+1$，说明补上去的这条最小替代边的权值必须**恰好等于 $v + 1$**。

既然新边 $(a, b)$ 在树上跨过的路径越长，路径上包含的树边就越多，要求所有这些树边的权值都 $\le v$ 就越容易踩雷（一旦跨过了一条权值 $> v$ 的树边就非法了）。

因此，**最安全、最经济的做法是让新边只跨过 2 条树边**（跨过 1 条树边相当于直接给 $(x,y)$ 加重边，但题目要求图 $G$ 是**简单无向图**，不能有重边和自环）。

连接一条新边，在满足条件一的前提下，只能处理最大的那一条边(在条件二下的需求)



**问题转化为：对每条边，找到一条不大于其边权的邻边。连接两个端点，则处理完这条边**

**具体来说，对于树边 $(x, y)$（权值为 $v$）： 我们尝试在 $y$ 的其他邻居里找一个点 $z$（或者在 $x$ 的其他邻居里找一个点 $z$），连接新边 $(x, z)$（或 $(y, z)$），权值设为 $v + 1$：**

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

//对每条边，找到一条不大于其边权的邻边。连接两个端点，则处理完这条边

vector<vector<array<int, 2>>> g;
int first[202020], second[202020];//第一小，第二小

vector<array<int, 3>> ans;
set<array<int, 2>> st;

//预处理节点x的邻接边中的最小，次小边权，存储的是其邻接表的下标i
void dfs1(int x, int fa) {
    first[x] = -1;
    second[x] = -1;
    
    for (int i = 0; i < g[x].size(); i++) {
        auto [y, w] = g[x][i];
        
        if (first[x] == -1 || w < g[x][first[x]][1]) {
            second[x] = first[x];
            first[x] = i;
        } else if (second[x] == -1 || w < g[x][second[x]][1]) {
            second[x] = i;
        }
        
        //注意：递归不可以递父边，但是求最小次小的时候需要父节点
        if (y != fa) dfs1(y, x);
    }
}

//找到边(x, y)的最小邻边
int find(int x, int y) {
    if (first[x] != -1 && g[x][first[x]][0] == y) {
        return second[x];
    }
    return first[x];
}

void add(int x, int y, int w) {
    if (x < y) swap(x, y);
    
    if (!st.count({x, y})) {
        st.insert({x, y});
        ans.push_back({x, y, w});
    }
}

//边(x, y)
//找到其可行的邻边，并加到答案中
//otherwise, return false
bool Do(int x, int y, int w) {
    int t = find(x, y);
    if (t != -1) {
        auto [z, v] = g[x][t];
        if (v <= w) {
            add(y, z, w + 1);
            return true;
        }
    }
    
    t = find(y, x);
    if (t != -1) {
        auto [z, v] = g[y][t];
        if (v <= w) {
            add(x, z, w + 1);
            return true;
        }
    }
    
    return false;
}

//对每一条树边，都进行操作
bool dfs2(int x, int fa) {
    for (int i = 0; i < g[x].size(); i++) if (g[x][i][0] != fa) {
        auto [y, w] = g[x][i];
        
        if (!Do(x, y, w)) {
            return false;
        }
        
        if (!dfs2(y, x)) {
            return false;
        }
    }
    return true;
}


void ap() {
	int n; cin >> n;
    g.assign(n + 1, {});
    ans.clear();
    st.clear();
    
    for (int i = 0; i < n - 1; i++) {
        int u, v, w; cin >> u >> v >> w;
        
        g[u].push_back({v, w});
        g[v].push_back({u, w});
        
        if (u < v) swap(u, v);
        st.insert({u, v});//不允许重边
    }
    
    dfs1(1, 0);
    
    if (!dfs2(1, 0)) {
        cout << "No" << endl;
        return;
    }
    
    cout << "Yes" << endl;
    
    //处理剩下的操作次数
    for (int i = 1; i <= n && ans.size() < n - 1; i++) {
        for (int j = i + 1; j <= n && ans.size() < n - 1; j++) {
            add(i, j, 2000000000);
        }
    }
    
    for (auto [x, y, w] : ans) {
        cout << x << ' ' << y << ' ' << w << endl;
    }
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
