```c++
int n;
int ans1 = 0, ans2 = 0;

vector<vector<int>> g;
vector<int> sub, dep;

void dfs(int i, int fa, int d) {
    sub[i] = 1;
    dep[i] = d;
    
    for (int x : g[i]) {
        if (x == fa) continue;
        
        dfs(x, i, d + 1);
        sub[i] += sub[x];
        
        int cnt = sub[x];
        int other = n - cnt - dep[x];
        ans1 += cnt * other;
    }
    
    ans2 += sub[i] - 1;
    return;
}

void ap() {

    sub.assign(n, 0), dep.assign(n, 0);
    ans1 = 0, ans2 = 0;
    
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        u--, v--;
        
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    dfs(0, -1, 0);
    
    cout << (ans1 + ans2) % MOD << endl;
    
    
	return ;
}
```

# 4.贪心构造 + 思维

[F-小芳的最小生成树_牛客小白月赛139](https://ac.nowcoder.com/acm/contest/141882/F)

- **条件一：$T$ 是图 $G$ 唯一的最小生成树**。 根据最小生成树的**环切性质**，如果我们在树 $T$ 的两个节点 $a, b$ 之间加一条权值为 $w_{new}$ 的新边，那么 $a, b$ 和它们在树上的唯一路径就会构成一个环。要想让 $T$ 是**唯一**的最小生成树，新加的这条边的权值 $w_{new}$ **必须严格大于**树上 $a$ 到 $b$ 路径上的每一条边的权值（即 $w_{new} > \max_{e \in path(a,b)} w_e$）。否则，用 $w_{new}$ 替换路径上的最大边，就能得到另一棵权值和更小或相等的生成树。
    
- **条件二：删去 $T$ 中任意一条边 $e=(x,y)$（设其权值为 $v$），新图 $G'$ 的最小生成树权值和恰好增加 $1$**。 删去树边 $e=(x,y)$ 后，原来的树 $T$ 会分裂成两个连通块。为了重新把它们连起来构成生成树，我们需要在横跨这两个连通块的所有新加边中，挑一条权值最小的边补上去。 原本总权值是 $x$，删掉 $e$ 少了 $v$，要求新生成树权值和为 $x+1$，说明补上去的这条最小替代边的权值必须**恰好等于 $v + 1$**。

既然新边 $(a, b)$ 在树上跨过的路径越长，路径上包含的树边就越多，要求所有这些树边的权值都 $\le v$ 就越容易踩雷（一旦跨过了一条权值 $> v$ 的树边就非法了）。

因此，**最安全、最经济的做法是让新边只跨过 2 条树边**（跨过 1 条树边相当于直接给 $(x,y)$ 加重边，但题目要求图 $G$ 是**简单无向图**，不能有重边和自环）。

连接一条新边，在满足条件一的前提下，只能处理最大的那一条边(在条件二下的需求)



**问题转化为：对每条边，找到一条不大于其边权的邻边。连接两个端点，则处理完这条边**

**具体来说，对于树边 $(x, y)$（权值为 $v$）： 我们尝试在 $y$ 的其他邻居里找一个点 $z$（或者在 $x$ 的其他邻居里找一个点 $z$），连接新边 $(x, z)$（或 $(y, z)$），权值设为 $v + 1$：**

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

//对每条边，找到一条不大于其边权的邻边。连接两个端点，则处理完这条边

vector<vector<array<int, 2>>> g;
int first[202020], second[202020];//第一小，第二小

vector<array<int, 3>> ans;
set<array<int, 2>> st;

//预处理节点x的邻接边中的最小，次小边权，存储的是其邻接表的下标i
void dfs1(int x, int fa) {
    first[x] = -1;
    second[x] = -1;
    
    for (int i = 0; i < g[x].size(); i++) {
        auto [y, w] = g[x][i];
        
        if (first[x] == -1 || w < g[x][first[x]][1]) {
            second[x] = first[x];
            first[x] = i;
        } else if (second[x] == -1 || w < g[x][second[x]][1]) {
            second[x] = i;
        }
        
        //注意：递归不可以递父边，但是求最小次小的时候需要父节点
        if (y != fa) dfs1(y, x);
    }
}

//找到边(x, y)的最小邻边
int find(int x, int y) {
    if (first[x] != -1 && g[x][first[x]][0] == y) {
        return second[x];
    }
    return first[x];
}

void add(int x, int y, int w) {
    if (x < y) swap(x, y);
    
    if (!st.count({x, y})) {
        st.insert({x, y});
        ans.push_back({x, y, w});
    }
}

//边(x, y)
//找到其可行的邻边，并加到答案中
//otherwise, return false
bool Do(int x, int y, int w) {
    int t = find(x, y);
    if (t != -1) {
        auto [z, v] = g[x][t];
        if (v <= w) {
            add(y, z, w + 1);
            return true;
        }
    }
    
    t = find(y, x);
    if (t != -1) {
        auto [z, v] = g[y][t];
        if (v <= w) {
            add(x, z, w + 1);
            return true;
        }
    }
    
    return false;
}

//对每一条树边，都进行操作
bool dfs2(int x, int fa) {
    for (int i = 0; i < g[x].size(); i++) if (g[x][i][0] != fa) {
        auto [y, w] = g[x][i];
        
        if (!Do(x, y, w)) {
            return false;
        }
        
        if (!dfs2(y, x)) {
            return false;
        }
    }
    return true;
}


void ap() {
	int n; cin >> n;
    g.assign(n + 1, {});
    ans.clear();
    st.clear();
    
    for (int i = 0; i < n - 1; i++) {
        int u, v, w; cin >> u >> v >> w;
        
        g[u].push_back({v, w});
        g[v].push_back({u, w});
        
        if (u < v) swap(u, v);
        st.insert({u, v});//不允许重边
    }
    
    dfs1(1, 0);
    
    if (!dfs2(1, 0)) {
        cout << "No" << endl;
        return;
    }
    
    cout << "Yes" << endl;
    
    //处理剩下的操作次数
    for (int i = 1; i <= n && ans.size() < n - 1; i++) {
        for (int j = i + 1; j <= n && ans.size() < n - 1; j++) {
            add(i, j, 2000000000);
        }
    }
    
    for (auto [x, y, w] : ans) {
        cout << x << ' ' << y << ' ' << w << endl;
    }
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
```

# 5. 树链剖分 + 树上差分

[F-小红的根（hard）_牛客周赛 Round 167](https://ac.nowcoder.com/acm/contest/142115/F)

记 $S(u)$ 为以 $u$ 为根的子树包含的节点的集合，记深度更小的那个节点的儿子中，以它为根的子树中包含深度更大的节点的那个儿子为 $c$。

| **节点关系（在以 1 为根的树中）** | **种类 1：a 是 b 的祖先 ** | **种类 2：a,b 互不为祖先 **  |
| -------------------- | ------------------- | -------------------- |
| **$a$ 是 $b$ 的祖先**    | $S(1) - S(c)$       | $S(c) - S(b)$        |
| **$b$ 是 $a$ 的祖先**    | $S(a)$              | $S(c) - S(a)$        |
| **上述情况都不是**          | $S(a)$              | $S(1) - S(a) - S(b)$ |
可以发现上面的计算都是基于子树，我们不妨使用树上差分，每次标记合法点集的所有点，最终标记次数恰好为 k 的即为全局合法点

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

struct HLD {
    int n, idx;
    vector<vector<int>> ver;
    vector<int> siz, dep;
    vector<int> top, son, parent;

    HLD(int n) {
        this->n = n;
        ver.resize(n + 1);
        siz.resize(n + 1);
        dep.resize(n + 1);

        top.resize(n + 1);
        son.resize(n + 1);
        parent.resize(n + 1);
    }
    void add(int x, int y) { // 建立双向边
        ver[x].push_back(y);
        ver[y].push_back(x);
    }
    void dfs1(int x) {
        siz[x] = 1;
        dep[x] = dep[parent[x]] + 1;
        for (auto y : ver[x]) {
            if (y == parent[x]) continue;
            parent[y] = x;
            dfs1(y);
            siz[x] += siz[y];
            if (siz[y] > siz[son[x]]) {
                son[x] = y;
            }
        }
    }
    void dfs2(int x, int up) {
        top[x] = up;
        if (son[x]) dfs2(son[x], up);
        for (auto y : ver[x]) {
            if (y == parent[x] || y == son[x]) continue;
            dfs2(y, y);
        }
    }
    int lca(int x, int y) {
        while (top[x] != top[y]) {
            if (dep[top[x]] > dep[top[y]]) {
                x = parent[top[x]];
            } else {
                y = parent[top[y]];
            }
        }
        return dep[x] < dep[y] ? x : y;
    }
    int clac(int x, int y) { // 查询两点间距离
        return dep[x] + dep[y] - 2 * dep[lca(x, y)];
    }
    
    // 前提：x 必须是 y 的真祖先 (dep[x] < dep[y] 且 lca(x, y) == x)
    int get_son_towards(int x, int y) {
        while (top[x] != top[y]) {
            // 如果 y 所在重链的链顶就是 x 的直系儿子
            if (parent[top[y]] == x) {
                return top[y];
            }
            // 否则将 y 沿着链向上跳
            y = parent[top[y]];
        }
        // 此时 x 和 y 位于同一条重链上，x 往 y 方向的儿子必然是 x 的重儿子
        return son[x];
    }
    
    void work(int root = 1) { // 在此初始化
        dfs1(root);
        dfs2(root, root);
    }
};

void ap() {
	int n, k; cin >> n >> k;
    
    HLD hld(n);
    
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        
        hld.add(u, v);
        
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    hld.work(1);
    
    vector<int> d(n + 1);
    for (int i = 0; i < k; i++) {
        int opt, a, b; cin >> opt >> a >> b;
        
        int t = hld.lca(a, b);
        if (t == a) {
            int c = hld.get_son_towards(a, b);
            if (opt == 1) {
                d[1]++;
                d[c]--;
            } else if (opt == 2) {
                d[c]++;
                d[b]--;
            }
        } else if (t == b) {
            if (opt == 1) {
                d[a]++;
            } else if (opt == 2) {
                int c = hld.get_son_towards(b, a);
                d[c]++;
                d[a]--;
            }
        } else {
            if (opt == 1) {
                d[a]++;
            } else {
                d[1]++;
                d[a]--, d[b]--;
            }
        }
    }
    
    auto dfs = [&](auto&& dfs, int x, int fa) -> void {
        d[x] += d[fa];
        for (int & y : g[x]) if (y != fa) {
            dfs(dfs, y, x);
        }
    };
    dfs(dfs, 1, 0);
    
    int ans = 0;
    for (int i = 1; i <= n; i++) if (d[i] == k) ans++;
    
    cout << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 6. 贪心

https://codeforces.com/contest/2277/problem/C

注意到如果 v 和 n 不在 1 的同一个子树中，(1,v) 一定是没用的，于是先忽略掉这种边。又发现 1 到 n 的路径上至多经过一条新加的边，于是我们可以独立考虑留下每一条边的最大答案。考虑加入 (1,v) 后，删除哪条边是最优的，发现是原树上 1 到 n 方向上的第一条边，此时的答案是 dv​+1，其中 dv​ 表示 v 与 n 在原树上的距离。最终答案就是所有边的答案的最大值，与 d1​ 的最大值。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;

    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int x, y; cin >> x >> y;

        g[x].push_back(y);
        g[y].push_back(x);
    }

    vector<int> a(m);
    for (int & i : a) cin >> i;

    vector<int> d(n + 1, -1);
    queue<int> q;

    d[n] = 0;
    q.push(n);

    while (q.size()) {
        auto x = q.front();
        q.pop();

                q.push(y);
            }
        }

        ans = max(ans, d[x] + 1);
    }
```

# 7. 贪心构造

https://codeforces.com/contest/2283/problem/D

发现
- n为偶数时，`k = n + 1`能将全部节点变为黑色，达到最优上限，直接就是答案
- n为奇数时，`k = n, k = n + 1, k = n + 2` 时只有一个不为黑色，k为其他值时黑色数量小于`n - 1`个，而黑色数量优先，即只能在上面三种k中取一种最优的
- 黑色最优，其次灰色，即剩下的节点如果可以染成灰色，则是奇数中最优的
- 如果可以染成灰色，任选一个满足的即可，否则都无法染成灰色，三个中任选一个即可

- 对于已知的k，不是黑色的节点x已知
- 如何判断节点x能否染成灰色？

总结：
- 考虑以节点x为根
- 枚举每一对`(u, v)` 如果其简单路径包含节点x，则节点x能染成灰色
- 考虑x的邻接连通块，如果`u,v` 处于同一个连通块中，则不能染色，否则可以染色
- 由于可以以x为根，x的邻接连通块即为子树
- 对x的子树进行染色做标记，最终枚举`(u, v)` 判断颜色是否不同即可

# 8. 构造

# 9. 换根 + 期望

[Shrinking Tree - 题目 - QOJ.ac](https://qoj.ac/contest/4686/problem/24861)

- 枚举每条边，每条边将树分为两个部分A,B
- 考虑将A缩成一个点，则答案加上 (A中的最大值 * p)
- 同理，考虑将B缩成一个点，则答案加上 (B中的最大值 * p)

A，B中的最大值：
- 以任一节点(1)为根，AB中有一个是子树中的最大值，另一个是除去子树中的最大值
- 跑一遍dfs，可以求出以1为根节点，子树的sz和子树中的最大值
- 再跑一遍dfs，考虑处理除去子树中的最大值
- 考虑后序遍历，先处理节点x邻接表中的前后缀最大，只有fa节点分支的最大值不知道，设为`ex_sub[x]` ，处理完后，根据前后缀最大，可以处理出x每个邻接节点y的`ex_sub[y]` 
- 然后递归`(y, x)` ，发现此时处理y的邻接节点时，需要的fa节点的分支即为`ex_sub[y]
- 即后序遍历的情况下，`ex_sub[fa]` 都是预先处理好了的 

```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
const int mod = 998244353;

const int MAXN = 4e5 + 50;

int ksm(int n, int m) {
    int res = 1;
    n %= mod;
    while (m) {
        if (m & 1) res = res * n % mod;
        n = n * n % mod;
        m >>= 1;
    }
    return res;
}

int fac[MAXN], inv[MAXN];

int C(int n, int m) {
    if (m < 0 || m > n || n < 0) return 0;
    return fac[n] * inv[m] % mod * inv[n - m] % mod;
}

void precompute() {
    fac[0] = inv[0] = 1;
    for (int i = 1; i < MAXN; i++) {
        fac[i] = fac[i - 1] * i % mod;
    }
    inv[MAXN - 1] = ksm(fac[MAXN - 1], mod - 2);
    for (int i = MAXN - 2; i >= 1; i--) {
        inv[i] = inv[i + 1] * (i + 1) % mod;
    }
}

void ap() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    // 树形 DP 求子树大小、子树内最大权值 max_sub 以及子树外最大权值 max_out
    vector<int> sz(n + 1, 0);
    vector<int> max_sub(n + 1, 0);
    vector<int> max_out(n + 1, 0);

    auto dfs1 = [&](auto& self, int u, int p) -> void {
        sz[u] = 1;
        max_sub[u] = a[u];
        for (int v : g[u]) {
            if (v == p) continue;
            self(self, v, u);
            sz[u] += sz[v];
            max_sub[u] = max(max_sub[u], max_sub[v]);
        }
    };

    auto dfs2 = [&](auto& self, int u, int p) -> void {
        int deg = g[u].size();
        vector<int> pref(deg + 2, 0), suff(deg + 2, 0);

        for (int i = 0; i < deg; i++) {
            int v = g[u][i];
            int val = (v == p) ? max_out[u] : max_sub[v];
            pref[i + 1] = max(pref[i], val);
        }
        for (int i = deg - 1; i >= 0; i--) {
            int v = g[u][i];
            int val = (v == p) ? max_out[u] : max_sub[v];
            suff[i + 1] = max(suff[i + 2], val);
        }

        for (int i = 0; i < deg; i++) {
            int v = g[u][i];
            if (v == p) continue;
            max_out[v] = max({a[u], pref[i], suff[i + 2]});
            self(self, v, u);
        }
    };

    dfs1(dfs1, 1, 0);
    dfs2(dfs2, 1, 0);

    int total_ways = C(n - 1, k);
    int ans = 0;

    // 枚举每一条非根边 (u, parent)
    for (int u = 2; u <= n; u++) {
        // 1. 子树 u 缩为一个叶子结点
        int s = sz[u];
        int need_u = s - 1;
        if (k >= need_u && (n - 1 - s) >= (k - need_u)) {
            int ways = C(n - 1 - s, k - need_u);
            ans = (ans + ways % mod * (max_sub[u] % mod)) % mod;
        }

        // 2. 子树 u 外部整体缩为一个叶子结点
        int s_out = n - s;
        int need_out = s_out - 1;
        if (k >= need_out && (s - 1) >= (k - need_out)) {
            int ways = C(s - 1, k - need_out);
            ans = (ans + ways % mod * (max_out[u] % mod)) % mod;
        }
    }

    ans = ans * ksm(total_ways, mod - 2) % mod;
    cout << ans << endl;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    precompute();
    int T = 1;
    cin >> T;
    while (T--) {
        ap();
    }
    return 0;
}
```
