[E-Flower_Rainbow_and_Game_牛客小白月赛127](https://ac.nowcoder.com/acm/contest/126634/E)

```c++
int n;
int ans1 = 0, ans2 = 0;

vector<vector<int>> g;
vector<int> sub, dep;

void dfs(int i, int fa, int d) {
    sub[i] = 1;
    dep[i] = d;
    
    for (int x : g[i]) {
        if (x == fa) continue;
        
        dfs(x, i, d + 1);
        sub[i] += sub[x];
        
        int cnt = sub[x];
        int other = n - cnt - dep[x];
        ans1 += cnt * other;
    }
    
    ans2 += sub[i] - 1;
    return;
}

void ap() {
	cin >> n;
    
    g.assign(n, vector<int>());
    sub.assign(n, 0), dep.assign(n, 0);
    ans1 = 0, ans2 = 0;
    
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        u--, v--;
        
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    dfs(0, -1, 0);
    
    cout << (ans1 + ans2) % MOD << endl;
    
    
	return ;
}
```

# 3.树上路径 + 贪心

https://codeforces.com/problemset/problem/2065/F

```c++
void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i ++) {
        int u, v; cin >> u >> v;

        g[u].push_back(v);
        g[v].push_back(u);
    }

    vector<int> ans(n + 1, 0);
    for (int x = 1; x <= n; x++) {
        map<int, int> cnt;
        cnt[a[x]]++;
        for (int & y : g[x]) {
            cnt[a[y]]++;
        }

        for (auto [t, c] : cnt) {
            if (c >= 2) {
                ans[t] = 1;
            }
        }

    }

    for (int i = 1; i <= n; i++) cout << ans[i];
    cout << endl;

	return ;
}
```
