
[P3379 【模板】最近公共祖先（LCA） - 洛谷](https://www.luogu.com.cn/problem/P3379)

```c++
#include<bits/stdc++.h>
using namespace std;
#define endl '\n'

signed main () {
    ios::sync_with_stdio(0), cin.tie(0), cout.tie(0);
    
    int n, m, s; cin >> n >> m >> s;
    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int x, y; cin >> x >> y;
        
        g[x].push_back(y);
        g[y].push_back(x);
    }

    int power = __lg(n) + 2;//不+2会错，一般是直接声明一个大常数
    vector<vector<int>> st(n + 1, vector<int>(power, -1));
    vector<int> dep(n + 1, 1);
    
    auto dfs = [&](this auto&& dfs, int x, int fa) -> void {
        st[x][0] = fa;
        for (int y : g[x]) {
            if (y != fa) {
                dep[y] = dep[x] + 1;
                dfs(y, x);
            }
        }
    };
    dfs(s, -1);

    for (int p = 1; p < power; p++) {
        for (int i = 1; i <= n; i++) {
            if (st[i][p - 1] != -1) st[i][p] = st[st[i][p - 1]][p - 1];
        }
    }

    auto get_kth_ancestor = [&](int i, int k) -> int {
        for (; k; k &= k - 1) {
            i = st[i][countr_zero((unsigned) k)];
        }
        return i;
    };

    auto lca = [&](int x, int y) -> int {
        if (dep[x] > dep[y]) {
            swap(x, y);
        }  //dep[x] <= dep[y];

        y = get_kth_ancestor(y, dep[y] - dep[x]);
        if (y == x) {
            return x;
        }

        for (int i = st[x].size() - 1; i >= 0; i--) {
            int px = st[x][i], py = st[y][i];
            if (px != py) {
                x = px;
                y = py;
            }
        }
        return st[x][0];
    };
    
    for (int i = 0; i < m; i++) {
        int a, b; cin >> a >> b;

        cout << lca(a, b) << endl;
    }

    return 0;
}
```
