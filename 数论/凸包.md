
凸包算法 —— Andrew 算法（Monotone Chain，单调链法

该算法的核心是分别求出“下凸包”和“上凸包”，然后进行合并。它的效率和 Graham 扫描法相同，但由于只需要对坐标进行双关键字排序（不需要算极角），写起来非常顺手，且不容易产生精度误差。

### Andrew 算法 C++ 模板

C++

```c++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

// 定义二维点结构体
struct Point {
    long long x, y;

    // 排序规则：先按 x 升序，x 相同则按 y 升序
    bool operator<(const Point& other) const {
        if (x != other.x) return x < other.x;
        return y < other.y;
    }
};

// 向量叉积 (Cross Product)
// 返回值 > 0: p0->p1 在 p0->p2 的顺时针方向（即 p0-p1-p2 构成向左转）
// 返回值 < 0: 向右转
// 返回值 = 0: 三点共线
// 注意：如果坐标较大，叉积可能会爆 int，所以用 long long
long long cross_product(const Point& p0, const Point& p1, const Point& p2) {
    return (p1.x - p0.x) * (p2.y - p0.y) - (p1.y - p0.y) * (p2.x - p0.x);
}

// 求凸包函数
vector<Point> get_convex_hull(vector<Point>& points) {
    int n = points.size();
    if (n <= 3) return points; // 点数小于等于3时，它们本身就是凸包

    // 1. 将所有点进行排序
    sort(points.begin(), points.end());

    vector<Point> hull;

    // 2. 求下凸包 (Lower Hull)
    for (int i = 0; i < n; ++i) {
        // 当新加入的点使得当前的最后两个点构成的向量向右转（或者共线，取决于是否保留边上的点）时，弹出栈顶
        // 注：如果想保留凸包边上的共线点，把 <= 0 改为 < 0
        while (hull.size() >= 2 && cross_product(hull[hull.size() - 2], hull.back(), points[i]) <= 0) {
            hull.pop_back();
        }
        hull.push_back(points[i]);
    }

    // 3. 求上凸包 (Upper Hull)
    // 此时 hull.size() 就是下凸包的点数 + 1（包含终点），记为 limit
    // 从倒数第二个点开始反向遍历
    size_t limit = hull.size();
    for (int i = n - 2; i >= 0; --i) {
        while (hull.size() > limit && cross_product(hull[hull.size() - 2], hull.back(), points[i]) <= 0) {
            hull.pop_back();
        }
        hull.push_back(points[i]);
    }

    // 因为起点和终点在求上下凸包时都被加入了两次，所以最后把重复的终点弹出
    hull.pop_back();

    return hull;
}

int main() {
    // 快速输入输出
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    if (cin >> n) {
        vector<Point> points(n);
        for (int i = 0; i < n; ++i) {
            cin >> points[i].x >> points[i].y;
        }

        vector<Point> hull = get_convex_hull(points);

        // 输出凸包顶点的数量和坐标（按逆时针顺序排列）
        cout << hull.size() << "\n";
        for (const auto& p : hull) {
            cout << p.x << " " << p.y << "\n";
        }
    }
    return 0;
}
```

### 模板关键点说明：

1. **叉积的判断条件 (`<= 0`)：**
    
    - 使用 `<= 0` 会严格过滤掉**共线**的点，这意味着最后生成的凸包数组中，只有凸多边形的**顶点**。
        
    - 如果题目要求你**把凸包边上的点也全部输出来**（例如部分网格点题目），需要把循环里的 `<= 0` 改成 `< 0`。
        
2. **坐标类型：**
    
    - 模板中使用了 `long long` 来存储坐标和计算叉积。如果题目给出的坐标范围在 $10^9$ 级别，两数相乘会达到 $10^{18}$，用 `int` 会直接溢出导致死循环或错误答案。
        
3. **点数特判：**
    
    - 如果输入的点集点数少于 3 个，直接返回原数组即可。
