
# 一、区间调度，不相交最大数量
[D-小红的最佳区间_牛客周赛 Round 143](https://ac.nowcoder.com/acm/contest/134529/D)
```c++
void ap() {
    int n, k; cin >> n >> k;
    map<int ,int> d;
    for (int i = 0; i < n; i++) {
        int l, r; cin >> l >> r;
        d[l - k]++;
        d[r + 1]--;
    }
     
    int s = 0;
    int ans = 1;
    for (auto & [x, v] : d) {
        s += v;
        ans = max(ans, s);
    }
     
    cout << ans;
     
    return ;
}
```


# 二、区间调度
[Problem - A - Codeforces](https://codeforces.com/gym/695065/problem/A)
```c++
int cmp(const pair<int, int> & a, const pair<int, int> & b) {
    if (a.second != b.second) {
        return a.second < b.second;
    }

    return a.first < b.first;
}

void ap() {
	int n; cin >> n;
    vector<int> r(n);
    for (int & i : r) cin >> i;

    vector<pair<int, int>> d(n - 1);
    for (int i = 0; i < n - 1; i++) {
        cin >> d[i].first >> d[i].second;
    }

    vector<pair<int, int>> a(n);
    a[0].first = 0, a[0].second = 0;

    for (int i = 1; i < n; i++) {
        a[i].first = a[i - 1].first + d[i - 1].first;
        a[i].second = a[i - 1].second + d[i - 1].second;
    }

    vector<pair<int, int>> p(n);
    for (int i = 0; i < n; i++) {
        int x = a[i].first + a[i].second;
        p[i] = {x - r[i], x + r[i]};
    }

    sort(p.begin(), p.end(), cmp);

    int cnt = 0;
    int end = -1e18;
    for (auto [start, e] : p) {
        if (start > end) {
            cnt++;
            end = e;
        }
    }

    cout << cnt << endl;

	return ;
}
```

# 三、区间调度 + 二分
[D - Maximize the Gap](https://atcoder.jp/contests/abc463/tasks/abc463_d)
```c++
vector<array<int, 2>> a;
int k;

int cmp(const array<int, 2> & a, const array<int, 2> & b) {
    if (a[1] == b[1]) {
        return a[0] > b[0];
    }
    return a[1] < b[1];
}

bool check(int x) {
    int end = -1e9, cnt = 0;
    for (auto & [s, e] : a) {
        if (s >= end) {
            cnt++;
            end = e + x;
        }
    }

    return cnt >= k;
}

void ap() {
	int n; cin >> n >> k;
    a.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i][0] >> a[i][1];
    }

    sort(a.begin(), a.end(), cmp);

    int l = 0, r = 1e18;
    while (l + 1 < r) {
        int mid = l + (r - l) / 2;
        if (check(mid)) {
            l = mid;
        } else {
            r = mid;
        }
    }

    if (l) {
        cout << l << endl;
    } else {
        cout << -1 << endl;
    }

	return ;
}
```
