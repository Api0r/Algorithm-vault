
[D-小月的相同数对_牛客周赛 Round 158](https://ac.nowcoder.com/acm/contest/139659/D)
```c++
void ap() {
	int n, k; cin >> n >> k;
    vector<int> a(n);
    for (int & i : a) cin >> i;
    
    int left = 0, ans = 1e9;
    map<int, int> cnt;
    int sum = 0;
    for (int i = 0; i < n; i++) {
        int x = a[i];
        
        //入
        cnt[x]++;
        sum += cnt[x] - 1;
        
        //卡
        while (left <= i && sum >= k) {
	        //更新答案
            ans = min(ans, i - left + 1);
            
            cnt[a[left]]--;
            sum -= cnt[a[left]];
            left++;
        }
    
    }
    
    if (ans == 1e9) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    
	return ;
}
```
