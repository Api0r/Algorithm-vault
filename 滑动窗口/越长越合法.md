
# 1.求最小

[D-小月的相同数对_牛客周赛 Round 158](https://ac.nowcoder.com/acm/contest/139659/D)


```c++
void ap() {
	int n, k; cin >> n >> k;
    vector<int> a(n);
    for (int & i : a) cin >> i;
    
    int left = 0, ans = 1e9;
    map<int, int> cnt;
    int sum = 0;
    for (int i = 0; i < n; i++) {
        int x = a[i];
        
        //入
        cnt[x]++;
        sum += cnt[x] - 1;
        
        //卡
        while (left <= i && sum >= k) {
	        //更新答案
            ans = min(ans, i - left + 1);
            
            cnt[a[left]]--;
            sum -= cnt[a[left]];
            left++;
        }
    
    }
    
    if (ans == 1e9) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    
	return ;
}
```

# 2. 求子数组个数

[E-小月的区间_牛客周赛 Round 162](https://ac.nowcoder.com/acm/contest/140489/E)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;
    
    vector<int> pre(n);
    pre[0] = a[0];
    for (int i = 1; i < n; i++) {
        pre[i] = max(pre[i - 1], a[i]);
    }
    
    vector<int> suf(n);
    suf[n - 1] = a[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        suf[i] = min(suf[i + 1], a[i]);
    }
    
    //if (l > 0) pre[l - 1] < pre[r] &&
    //if (r < n - 1) suf[l] < suf[r + 1]
    
    int ans = 0;
    int left = 0;
    for (int i = 0; i < n; i++) {
        while (left <= i && (left == 0 || pre[left - 1] < pre[i]) && (i == n - 1 || suf[left] < suf[i + 1])) {
            left++;
        }
        
        ans += left;
        
    }
    
    cout << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
