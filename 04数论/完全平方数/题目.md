
# 1. 质因数分解角度

https://codeforces.com/contest/2275/problem/F

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    vector<vector<int>> p(n);
    for (int i = 0; i < n; i++) {
        int x = a[i];

        for (int j = 2; j * j <= x; j++) {
            if (x % j == 0) {
                int cnt = 0;
                while (x % j == 0) {
                    x /= j;
                    cnt++;
                }
                if (cnt & 1) p[i].push_back(j);
            }
        }
        if (x > 1) p[i].push_back(x);

    }

    set<int> odd;//prefix_odd
    map<vector<int>, int> cnt;

    for (int i = 0; i < n; i++) {
        for (int x : p[i]) {
            if (odd.count(x)) {
                odd.erase(x);
            } else {
                odd.insert(x);
            }
        }

        if (odd.size() <= 7) {
            cnt[vector<int>(odd.begin(), odd.end())]++;
        }

    }

    int ans = 0;
    for (int i = 0; i < n; i++) {
        ans += cnt[p[i]];
    }

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
