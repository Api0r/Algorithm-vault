
# 1. 质因数分解角度

https://codeforces.com/contest/2275/problem/F

质因数分解后，所有质因子的质数都必须为偶数

对每个数进行质因数分解，找到其指数为奇数的质数集合vector
- 如果x的集合为空，则需要和同理prefix_product的集合相等的匹配
- x的集合不为空，则需要同x的集合完全一样的prefix_product集合匹配

所以我们只需要再处理出prefix_product的集合，用`map<vector<int>, int>` 存储，最后计算ans

对于prefix_product的集合，可以利用之前处理出来的x的集合，(只有奇数会改变奇偶性)


tips: $$2 \times 3 \times 5 \times 7 \times 11 \times 13 \times 17 = 510,510 \le 10^6$$

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    vector<vector<int>> p(n);
    for (int i = 0; i < n; i++) {
        int x = a[i];

        for (int j = 2; j * j <= x; j++) {
            if (x % j == 0) {
                int cnt = 0;
                while (x % j == 0) {
                    x /= j;
                    cnt++;
                }
                if (cnt & 1) p[i].push_back(j);
            }
        }
        if (x > 1) p[i].push_back(x);

    }

    set<int> odd;//prefix_odd
    map<vector<int>, int> cnt;

    for (int i = 0; i < n; i++) {
        for (int x : p[i]) {
            if (odd.count(x)) {
                odd.erase(x);
            } else {
                odd.insert(x);
            }
        }

        if (odd.size() <= 7) {
            cnt[vector<int>(odd.begin(), odd.end())]++;
        }

    }

    int ans = 0;
    for (int i = 0; i < n; i++) {
        ans += cnt[p[i]];
    }

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
