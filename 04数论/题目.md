```c++
#include <bits/stdc++.h>

#define endl '\n'

    int x = 0, y = 0;

        }
    }
```

# 4. 容斥原理

[E - Typo Pairs](https://atcoder.jp/contests/abc483/tasks/abc483_e)

- **集合 $A$**：$S_i$ 和 $S_j$ **完全相同**（汉明距离为 0）的数对集合（共 1 个集合）。
    
- **集合 $B_k$** ($1 \le k \le M$)：**除了第 $k$ 个字符外**，其余 $M-1$ 个字符全都相同的数对集合。注意：第 $k$ 个字符本身可以相同，也可以不同（即汉明距离 $\le 1$）。共有 $M$ 个这样的集合。
    
- **集合 $C_{k, k'}$** ($1 \le k < k' \le M$)：**除了第 $k$ 和第 $k'$ 个字符外**，其余 $M-2$ 个字符全都相同的数对集合。同样，第 $k, k'$ 个字符本身可以相同也可以不同（即汉明距离 $\le 2$）。共有 $\frac{M(M-1)}{2}$ 个这样的集合。



- **当 $S_i$ 与 $S_j$ 完全相同时（$d = 0$）**：
    
    - 哪怕遮住任意 1 个或 2 个字符，剩下的部分依然完全相同。因此它属于**所有集合**：
        
    - 在 $A$ 中出现 **$1$ 次**；
        
    - 在 $B_1, \dots, B_M$ 中出现 **$M$ 次**；
        
    - 在 $C_{1,2}, \dots, C_{M-1,M}$ 中出现 **$\frac{M(M-1)}{2}$ 次**。
        
- **当 $S_i$ 与 $S_j$ 只有第 $k$ 个字符不同时（$d = 1$）**：
    
    - 只有把第 $k$ 个字符遮住，剩下的才会完全相等：
        
    - 在 $A$ 中出现 **$0$ 次**；
        
    - 在 $B$ 系列集合中，只有 $B_k$ 遮住了第 $k$ 位，因此只出现 **$1$ 次**；
        
    - 在 $C$ 系列集合中，只要挑选的两个位置里包含 $k$（另一个位置任意选，有 $M-1$ 种选法），剩下的就相等，因此出现在 **$M-1$ 个** $C$ 集合中。
        
- **当 $S_i$ 与 $S_j$ 有第 $k, k'$ 两个字符不同时（$d = 2$）**：
    
    - 必须同时把 $k$ 和 $k'$ 遮住，剩下的才会相等：
        
    - 在 $A$ 中出现 **$0$ 次**；
        
    - 在 $B$ 系列集合中出现 **$0$ 次**；
        
    - 在 $C$ 系列集合中，只出现在 $C_{k, k'}$ 这 **$1$ 个**集合中。
        
- **当 $S_i$ 与 $S_j$ 有 3 个及以上字符不同时（$d \ge 3$）**：
    
    - 最多遮住 2 个字符不可能让剩下的完全相等，因此在所有集合中均出现 **$0$ 次**。




我们的目标是让 $d = 0, 1, 2$ 的数对最终都恰好被算 **$1$ 次**，而 $d \ge 3$ 的数对算 **$0$ 次**。

设我们将所有 $C_{k,k'}$ 的大小加起来（系数为 $1$），减去 $(M-2)$ 倍的所有 $B_k$ 大小，再加上 $\frac{(M-1)(M-2)}{2}$ 倍的 $A$ 的大小，就得到了题解中的**公式 (1)**：

$$X = \frac{(M-1)(M-2)}{2} \vert{}A\vert{} - (M-2) \sum_{k=1}^{M} \vert{}B_k\vert{} + \sum_{k=1}^{M-1} \sum_{k'=k+1}^{M} \vert{}C_{k,k'}\vert{}$$


```c++
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;
#define int long long

// 辅助函数：给定 N 个字符串，统计其中有多少对完全相同的字符串
int count_pairs(vector<string>& v) {
    sort(v.begin(), v.end());
    int pairs = 0;
    int cnt = 1;
    for (int i = 1; i < (int)v.size(); ++i) {
        if (v[i] == v[i - 1]) {
            cnt++;
        } else {
            pairs += cnt * (cnt - 1) / 2;
            cnt = 1;
        }
    }
    pairs += cnt * (cnt - 1) / 2;
    return pairs;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;
    vector<string> s(n);
    for (int i = 0; i < n; ++i) {
        cin >> s[i];
    }

    // 1. 求 |A|：完全不删字符
    vector<string> temp = s;
    int A = count_pairs(temp);

    // 2. 求 sum(|B_k|)：枚举删掉第 k 个字符
    int sum_B = 0;
    for (int k = 0; k < m; ++k) {
        for (int i = 0; i < n; ++i) {
            // 把除了第 k 位以外的字符拼起来
            string t = "";
            for (int j = 0; j < m; ++j) {
                if (j != k) t += s[i][j];
            }
            temp[i] = t;
        }
        sum_B += count_pairs(temp);
    }

    // 3. 求 sum(|C_{k, k'}|)：枚举删掉第 k 和 k' 两个字符
    int sum_C = 0;
    for (int k1 = 0; k1 < m; ++k1) {
        for (int k2 = k1 + 1; k2 < m; ++k2) {
            for (int i = 0; i < n; ++i) {
                // 把除了 k1 和 k2 以外的字符拼起来
                string t = "";
                for (int j = 0; j < m; ++j) {
                    if (j != k1 && j != k2) t += s[i][j];
                }
                temp[i] = t;
            }
            sum_C += count_pairs(temp);
        }
    }

    // 4. 代入题解公式 (1)
    int ans = (m - 1) * (m - 2) / 2 * A - (m - 2) * sum_B + sum_C;
    cout << ans << "\n";

    return 0;
}
```


# 5. 组合数学 + 图论

https://codeforces.com/gym/106565/problem/G

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
int mod = 1e9 + 7;
int MOD = 998244353;

int ksm(int n, int m){
    int res = 1;
    n %= mod;
    while(m){
        if(m & 1){
            res = res * n % mod;
        }
        n = n * n % mod;
        m = m >> 1;
    }
    return res;
}

void ap() {
	int n, m; cin >> n >> m;
    mod = m;

    int k = (n - 1) * (n - 2) / 2 + 1 - (n % 2);
    cout << ksm(2, k) << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
