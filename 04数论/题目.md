
# 1. 二的幂次贪心 + 倒排

https://codeforces.com/contest/2284/problem/A2

倒序排序，能取大的则取，贪心地让小的去补
关键点在于小的能补的更多

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    sort(a.rbegin(), a.rend());

    int x = 0, y = 0;
    for (int i = 0; i < n; i++) {
        if (x < y) {
            x += (1LL << a[i]);
        } else {
            y += (1LL << a[i]);
        }
    }

    cout << x * y << endl;


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
    int n, k; cin >> n >> k;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    sort(a.rbegin(), a.rend());
    
    reverse(a.begin(), a.begin() + k);

    int cnt = 1;
    int s = a[0];

    for (int i = 1; i < n; i++) {
        if (a[i] * i >= s) cnt++;
        s += a[i];
    }

    if (cnt == k) {
        cout << "YES" << endl;
        for (int x : a) cout << x << ' ';
        cout << endl;
    } else {
        cout << "NO" << endl;
    }

    return ;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T = 1;
    cin >> T;
    while (T--) {
        ap();
    } 
    return 0;
}
```
