问题可以转化为：
- 每次操作任选k个数，将这k个数整体+1 或 -1。操作cost为k
- 可以操作任意次，最终要使所有数都相等
- 求最小cost，以及在这个最小cost中的最小操作次数

发现：
- 最小的cost就是将每个数+1 或 -1 使得所有数相等的次数
- 使所有数相等的最小次数，就是全都变为中位数所需的次数

总结：
- cost可以通过中位数的思路用上述方法求出
- 一次操作可以任选k个数+1 或 -1 
- 则操作次数为+1 中的最大次数，加上 -1 中的最小次数



```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    sort(a.begin(), a.end());

    int mid = n / 2;

    int cnt1 = 0, cnt2 = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < a[mid]) {
            cnt1 += a[mid] - a[i];
        }

        if (a[i] > a[mid]) {
            cnt2 += a[i] - a[mid];
        }
    }

    cout << cnt1 + cnt2 << ' ' << a[mid] - a[0] + a.back() - a[mid] << endl;


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
