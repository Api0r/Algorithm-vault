#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    sort(a.begin(), a.end());

    int mid = n / 2;

    int cnt1 = 0, cnt2 = 0;
    for (int i = 0; i < n; i++) {
        if (a[i] < a[mid]) {
            cnt1 += a[mid] - a[i];
        }

        if (a[i] > a[mid]) {
            cnt2 += a[i] - a[mid];
        }
    }

    cout << cnt1 + cnt2 << ' ' << a[mid] - a[0] + a.back() - a[mid] << endl;


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
