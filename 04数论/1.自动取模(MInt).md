```c++
/*
用途：自动取模整数。直接写 MInt<P>，P=0 时先 MInt<0>::setMod(p)。
接口：pow(n)、inv()、四则运算、输入输出；默认 Z 为模 998244353。
约定：inv()/除法要求模数为质数；动态模数需满足 1 < p < 2^31。
性能：内部值固定为 i32，故在 #define int long long 下也不会把大数组翻倍。
*/

using i32 = int32_t;

constexpr i32 mod = 998244353;

//快速幂模板
template <class T> T mypow(T a, int n) {
    assert(n >= 0);
    T r = T(1);
    while (n) {
        if (n & 1) r *= a;
        a *= a;
        n >>= 1;
    }
    return r;
}

template <i32 P> struct MInt {
    static_assert(P >= 0);

    i32 x = 0;
    inline static i32 md = P ? P : mod;

    MInt() = default;
    MInt(int v) : x(norm(v)) {}

    static i32 askMod() {
        return P ? P : md;
    }

    static void setMod(i32 v) {
        static_assert(P == 0, "Only MInt<0> supports a dynamic modulus.");
        assert(v > 1);
        md = v;
    }

    static i32 norm(int v) {
        v %= askMod();
        if (v < 0) v += askMod();
        return static_cast<i32>(v);
    }

    i32 val() const {
        return x;
    }

    explicit operator i32() const {
        return x;
    }

    bool ask0() const {
        return x == 0;
    }

    MInt pow(int n) const {
        return mypow(*this, n);
    }
	
	//费马小定理求逆元 (inv)
    MInt inv() const {
        assert(x);
        return pow(askMod() - 2);
    }

    MInt operator-() const {
        return MInt(x ? askMod() - x : 0);
    }

    MInt& operator+=(MInt a) {
        i32 p = askMod();
        if (x >= p - a.x) x -= p - a.x;
        else x += a.x;
        return *this;
    }

    MInt& operator-=(MInt a) {
        x -= a.x;
        if (x < 0) x += askMod();
        return *this;
    }

    MInt& operator*=(MInt a) {
        x = static_cast<i32>(static_cast<int>(x) * a.x % askMod());
        return *this;
    }

    MInt& operator/=(MInt a) {
        return *this *= a.inv();
    }

    MInt& operator++() {
        return *this += 1;
    }

    MInt operator++(i32) {
        MInt a = *this;
        ++*this;
        return a;
    }

    MInt& operator--() {
        return *this -= 1;
    }

    MInt operator--(i32) {
        MInt a = *this;
        --*this;
        return a;
    }

    friend MInt operator+(MInt a, MInt b) {
        return a += b;
    }

    friend MInt operator-(MInt a, MInt b) {
        return a -= b;
    }

    friend MInt operator*(MInt a, MInt b) {
        return a *= b;
    }

    friend MInt operator/(MInt a, MInt b) {
        return a /= b;
    }

    friend bool operator==(MInt a, MInt b) {
        return a.x == b.x;
    }

    friend bool operator!=(MInt a, MInt b) {
        return a.x != b.x;
    }

    friend istream& operator>>(istream& is, MInt& a) {
        int x;
        is >> x;
        a = MInt(x);
        return is;
    }

    friend ostream& operator<<(ostream& os, const MInt& a) {
        return os << a.x;
    }
};

template <i32 V, i32 P> const MInt<P> IV = MInt<P>(V).inv();

using Z = MInt<mod>;
```

```c++
template <const int mod>
struct modint{
    int normal(int _x){
        return _x + (_x < 0) * mod + (_x >= mod) * -mod;
    }
    friend modint qp(modint base, int p){
        modint res = 1;
        while(p){
            if(p & 1) res *= base;
            base *= base, p >>= 1;
        }
        return res;
    }
    int x;
    modint(int _x = 0, bool good = 0){
        if(good){
            x = _x;
        } else {
            x = normal((_x % mod));
        }
    }
    modint operator - (){
        return modint(normal(-x), 1);
    }
    modint inv (){
        return qp(*this, mod - 2);
    }
    modint& operator += (modint v){
        x = normal(x + v.x);
        return *this;
    }
    modint& operator -= (modint v){
        return *this += -v;
    }
    modint& operator *= (modint v){
        x = (x * v.x % mod);
        return *this;
    }
    modint operator /= (modint v){
        return *this *= v.inv();
    }
    friend modint operator + (modint u, modint v){
        return u += v;
    }
    friend modint operator - (modint u, modint v){
        return u -= v;
    }
    friend modint operator * (modint u, modint v){
        return u *= v;
    }
    friend modint operator / (modint u, modint v){
        return u /= v;
    }
    friend std::ostream& operator << (std::ostream& os, modint u){
        return os << u.x;
    }
    friend std::istream& operator >> (std::istream& is, modint& u){
        int _x;
        is >> _x;
        u.x = normal((_x % mod));
        return is;
    }
};

using Z = modint<998244353>;
```

这是一份现代竞赛级别的 **高性能自动取模类模板（Modular Integer）**。相比前面的基础版，它支持了**静态/动态模数切换**、**防宏定义内存翻倍优化**以及**完整的运算符重载**。

### 一、 关键设计亮点

- **防止内存翻倍（`#define int long long` 兼容）**：

    - 类内核心存储变量显式指定为 `i32`（即 `int32_t`，占 4 字节）：`i32 x = 0;`。

    - 即使在外部写了 `#define int long long`，开包含数百万个 `Z` 的大数组时，每个元素依然只占 4 字节，有效防止内存超限（MLE）。

- **支持静态模数与动态模数**：

    - **静态模数**：`MInt<998244353>` 或 `MInt<1000000007>`，编译期常量，效率最高。

    - **动态模数**：使用 `MInt<0>`。在程序运行时通过 `MInt<0>::setMod(p)` 动态设置模数，方便处理多组数据且每组模数不同的题目。

### 二、 核心函数与机制拆解

- **快速幂辅助函数 (`mypow`)**

    ```c++
    template <class T> T mypow(T a, int n)
    ```

    通用的二进制快速幂模板，时间复杂度 $O(\log n)$。

- **模数获取与动态设置 (`askMod` / `setMod`)**

    ```c++
    inline static i32 md = P ? P : mod;
    static i32 askMod() { return P ? P : md; }
    static void setMod(i32 v) {
        static_assert(P == 0, "Only MInt<0> supports a dynamic modulus.");
        assert(v > 1);
        md = v;
    }
    ```

    若模板参数 $P \neq 0$，则直接返回静态模数 $P$；若 $P = 0$，则返回静态变量 `md`（由 `setMod` 设定）。

- **规范化映射 (`norm`)**

    ```c++
    static i32 norm(int v) {
        v %= askMod();
        if (v < 0) v += askMod();
        return static_cast<i32>(v);
    }
    ```

    将任意可能越界或为负数的整数规范化映射到 $[0, P - 1]$ 范围内。

- **费马小定理求逆元 (`inv`)**

    ```c++
    MInt inv() const {
        assert(x);
        return pow(askMod() - 2);
    }
    ```

    基于费马小定理 $a^{P-1} \equiv 1 \pmod P \implies a^{-1} \equiv a^{P-2} \pmod P$，在模数为质数时以 $O(\log P)$ 复杂度求出逆元。

- **四则运算与自增减**

    - **`+=` / `-=`**：使用无除法指令的加减判断逻辑，显著提升常数效率。

    - **`*=`**：强转为 `int`（若有宏定义则为 `long long`）做防溢出乘法后再取模。

    - **`/=`**：转化为乘逆元 `*this *= a.inv()`。

    - **`++` / `--`**：实现了前置与后置的自增/自减重载。

- **常量逆元加速预计算 (`IV`)**

    ```c++
    template <i32 V, i32 P> const MInt<P> IV = MInt<P>(V).inv();
    ```

    用于在编译期/全局初始化时快速定义常用数字的逆元（如 `IV<2, mod>` 表示模 `mod` 下 $2$ 的逆元 $\frac{1}{2}$）。

- **流输入输出重载**

    重载了 `cin >>` 和 `cout <<`，可以直接像普通变量一样对 `Z` 进行输入与输出。

### 三、 常见用法示例

```c++
// 1. 默认使用静态模数 998244353
Z a = 5, b = 2;
cin >> a;
cout << a / b << endl; // 自动计算 a * (2 的逆元)
cout << a.pow(10) << endl; // 计算 a^10 % 998244353

// 2. 动态模数用法（模数由输入给定）
int p;
cin >> p;
MInt<0>::setMod(p);
using DynZ = MInt<0>;

DynZ x = 100;
cout << x * x << endl;
```
