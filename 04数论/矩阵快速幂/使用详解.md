struct M {
    array<array<int, sz>, sz> a;
 
    M operator*(const M &T) {

        int r;
        for (int i = 0; i < sz; ++i)
            for (int k = 0; k < sz; ++k) {
                r = a[i][k];
                for (int j = 0; j < sz; ++j)
                    (res.a[i][j] += T.a[k][j] * r) %= mod;
            }
        return res;
    }
 
    M() {
        for (int i = 0; i < sz; i++) {
            for (int j = 0; j < sz; j++)
                a[i][j] = 0;
        }
    }
};
 
M qpow(M a, int n) {
    M res;
    res.a[sz-1][0] = 1;
    while (n) {
        if (n%2) res = a*res;
        a = a*a;
        n /= 2;
    }
    return res;
}
