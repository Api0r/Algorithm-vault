
# 1.扩展欧几里得（EXGCD）+ 二分

[D-东风谷早苗与博丽灵梦_2026牛客寒假算法基础集训营4](https://ac.nowcoder.com/acm/contest/120564/D)

```c++
#include <iostream>
#include <algorithm>

using namespace std;
using LL = long long;
using LLL = __int128;

// 扩展欧几里得算法
LLL extgcd(LLL a, LLL b, LLL &x, LLL &y) {
    if (b == 0) {
        x = 1;
        y = 0;
        return a;
    }
    LLL x1, y1;
    LLL g = extgcd(b, a % b, x1, y1);
    x = y1;
    y = x1 - (a / b) * y1;
    return g;
}

void solve() {
    LL x_in, a_in, s_in;
    cin >> x_in >> a_in >> s_in;

    LLL x = x_in, a = a_in, s = s_in;
    LLL x0, y0;
    LLL g = extgcd(a, s, x0, y0);

    // 无法整除，无整数解
    if (x % g != 0) {
        cout << "No\n";
        return;
    }

    LLL step_x = s / g;
    LLL step_y = a / g;

    // 放大特解，并取 c1 的最小非负整数解
    LLL c1 = x0 * (x / g);
    c1 = (c1 % step_x + step_x) % step_x;

    // 计算对应的 c2
    LLL c2 = (x - a * c1) / s;

    // 如果 c2 < 0，说明没有任何一组非负整数解
    if (c2 < 0) {
        cout << "No\n";
        return;
    }

    cout << "Yes\n";

    // k 的有效非负范围是 [0, max_k]
    LLL max_k = c2 / step_y;

    auto get_pair = [&](LLL k) -> pair<LLL, LLL> {
        return {c1 + k * step_x, c2 - k * step_y};
    };

    // 二分寻找使 c1(k) >= c2(k) 的最小整数 k
    LLL l = 0, r = max_k;
    LLL best_k = max_k;
    while (l <= r) {
        LLL mid = l + (r - l) / 2;
        auto [cur_c1, cur_c2] = get_pair(mid);
        if (cur_c1 >= cur_c2) {
            best_k = mid;
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }

    // 在 best_k 和 best_k - 1 两侧比对，取 max(c1, c2) 更小的一组
    LLL ans_k = best_k;
    if (best_k > 0) {
        auto [c1_prev, c2_prev] = get_pair(best_k - 1);
        auto [c1_curr, c2_curr] = get_pair(best_k);
        if (max(c1_prev, c2_prev) < max(c1_curr, c2_curr)) {
            ans_k = best_k - 1;
        }
    }

    auto [ans_c1, ans_c2] = get_pair(ans_k);
    cout << (LL)ans_c1 << " " << (LL)ans_c2 << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    if (cin >> T) {
        while (T--) {
            solve();
        }
    }
    return 0;
}
```
