```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

const int N = 1e5 + 10;
double a[N];
int len[N << 2];
double mx[N << 2];
double t[N << 2];

void up(int i) {
    mx[i] = max(mx[i << 1], mx[i << 1 | 1]);
}

int up1(double lx, int l, int r, int i) {
    if (mx[i] <= lx) return 0;
    if (a[l] > lx) return len[i];
    if (l == r) return a[l] > lx;

    int s1 = i << 1, s2 = i << 1 | 1;
    int mid = (l + r) >> 1;
    if  (mx[s1] <= lx) return up1(lx, mid + 1, r, s2);
    else return len[i] - len[s1] + up1(lx, l, mid, s1);
}

void change(int jobi, int jobv, int l, int r, int i) {
    if (l == r) {
        len[i] = 1;
        mx[i] = (double)jobv / jobi;
        return;
    }

    int mid = (l + r) >> 1;
    if (jobi <= mid) {
         change(jobi, jobv, l, mid, i << 1);
    } else {
        change(jobi, jobv, mid + 1, r, i << 1 | 1);
    }
    up(i);

    len[i] = len[i << 1] + up1(mx[i << 1], mid + 1, r, i << 1 | 1);
}


void ap() {
	int n, m; cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int x, y; cin >> x >> y;

        a[x] = (double) y / x;
        change(x, y, 1, n, 1);

        cout << len[1] << endl;
    }

    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
