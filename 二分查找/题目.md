```c++
    for (int & i : a) cin >> i;
    for (int & i : b) cin >> i;

    auto check = [&](int x) -> bool {
        vector<array<int, 2>> pre(n), suf(n);
        pre[0][0] = 0, pre[0][1] = a[0] >= x ? 1 : -1e18;
        for (int i = 1; i < n ;i++) {
            pre[i][0] = max(pre[i - 1][0], pre[i - 1][1]);

            pre[i][1] = a[i] >= x ? 1 + pre[i - 1][0] : -1e18;
        }

        suf[n - 1][0] = 0, suf[n - 1][1] = b[n - 1] >= x ? 1 : -1e18;
        for (int i = n - 2; i >= 0; i--) {
            suf[i][0] = max(suf[i + 1][0], suf[i + 1][1]);

            suf[i][1] = b[i] >= x ? 1 + suf[i + 1][0] : -1e18;
        }


        if (max(suf[0][0], suf[0][1]) >= m) return true;
        if (max(pre[n - 1][0], pre[n - 1][1]) >= m) return true;
        
        for (int i = 0; i < n - 1; i++) {
            int ans1 = pre[i][0] + max(suf[i + 1][0], suf[i + 1][1]);
            int ans2 = pre[i][1] + suf[i + 1][0];

            if (max(ans1, ans2) >= m) return true;
        }

        return false;
        
    };
    
    int left = -1e9, right = 1e9;
    while (left + 1 < right) {
        int mid = left + (right - left) / 2;
        if (check(mid)) {
            left = mid;
        } else {
            right = mid;
        }
    }

    cout << left << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, k; cin >> n >> k;
    vector<array<int, 3>> p(n);
    for (int i = 0; i < n; i++) cin >> p[i][0] >> p[i][1] >> p[i][2];

    auto check = [&](int s) -> bool {
        int ans = 0;
        for (auto  [a, b, c] : p) {
            if (a + b + c >= s) continue;
            if (a == b && a == c) return false;

            if (a > b || a > c || b > c) {
                ans += s - (a + b + c);
            } else {
                //a <= b <= c
                int x1 = c - (b - 1);
                int ans1 = x1 + s - (a + b + b - 1);

                int x2 = b - (a - 1);
                int ans2 = x2 + s - (a + a - 1 + c);

                ans += min(ans1, ans2);
            }

            if (ans > k) return false;
            
        }

        return ans <= k;
    };

    int left = -4e9, right = 2e18 + 4e9;
    while (left + 1 < right) {
        int mid = left + (right - left) / 2;
        if (check(mid)) {
            left = mid;
        } else {
            right = mid;
        }

        // cout << left << ' ' << mid << ' ' << right << endl;
    }

    cout << left << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
