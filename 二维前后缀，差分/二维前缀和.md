
# 前缀和 + BFS

[D-走迷宫_牛客周赛 Round 151](https://ac.nowcoder.com/acm/contest/137267/D)

```c++
int dx[4] = {0, 0, 1, -1}, dy[4] = {1, -1, 0, 0};

void ap() {
	int n, m; cin >> n >> m;
    int xx, yy; cin >> xx >> yy;
    vector<string> a(n);
    for (auto & i : a) cin >> i;
    
    array<int, 2> st, ed;
    vector<vector<int>> b(n, vector<int> (m, 0));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] == 'S') {
                st = {i, j};
            } else if (a[i][j] == 'E') {
                ed = {i, j};
            } else if (a[i][j] == '.') {
                b[i][j] = 0;
            } else {
                b[i][j] = 1;
            }
        }
    }
    
    vector<vector<int>> pre(n + 1, vector<int>(m + 1));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            pre[i + 1][j + 1] = pre[i][j + 1] + pre[i + 1][j] - pre[i][j] + b[i][j];
        }
    }
    
    auto ok = [&](int x, int y) -> bool {
        if (x < 0 || y < 0 || x + xx > n || y + yy > m) return false;
        
        return pre[x + xx][y + yy] - pre[x + xx][y] - pre[x][y + yy] + pre[x][y] == 0;
    };
    
    vector<vector<bool>> vis(n, vector<bool> (m, false));
    
    queue<array<int, 2>> q;
    q.push(st);
    vis[st[0]][st[1]] = true;
    while (!q.empty()) {
        auto [x, y] = q.front();
        q.pop();
        
        if (x == ed[0] && y == ed[1]) {
            cout << "YES" << endl;
            return;
        }
        
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i], ny = y + dy[i];
            
            if (!ok(nx, ny)) continue;
            if (vis[nx][ny]) continue;
            
            
            vis[nx][ny] = true;
            q.push({nx, ny});
        }
    }
    cout << "NO" << endl;
    
	return ;
}
```
