
# 差分 + 切比雪夫距离 + 枚举中心

[F-小橙协调网格_牛客小白月赛134](https://ac.nowcoder.com/acm/contest/137072/F)

```c++
void ap() {
	int n, m; cin >> n >> m;
    vector<vector<int>> d(n + 2, vector<int> (m + 2));
    
    auto add = [&](int r1, int c1, int r2, int c2, int val) -> void {
        r1 = max(r1, 1LL), c1 = max(c1, 1LL), r2 = min(r2, n), c2 = min(c2, m);
        if (r1 > r2 || c1 > c2) return ;
        
        d[r1][c1] += val;
        d[r1][c2 + 1] -= val;
        d[r2 + 1][c1] -= val;
        d[r2 + 1][c2 + 1] += val;
        
        return;
    };
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            int k; cin >> k;
            
            add(i - k, j - k, i + k, j + k, 1);
            if (k > 0) add(i - (k - 1), j - (k - 1), i + (k - 1), j + (k - 1), -1);
        }
    }
    
    /*
    当外层循环执行到第 i 行、第 j 列时，
    d[i-1][j]、d[i][j-1] 和 d[i-1][j-1] 其实已经完成了解密（变成了前缀和值），
    而当前的 d[i][j] 还是纯粹的差分值。
    */
    int mx = -1;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            d[i][j] = d[i - 1][j] + d[i][j - 1] - d[i - 1][j - 1] + d[i][j];
            
            mx = max(mx, d[i][j]);
        }
    }
    
    cout << n * m - mx << endl;
    
    
    
	return ;
}
```
