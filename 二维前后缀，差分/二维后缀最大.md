
#  后缀最大 - 矩阵网格覆盖

[E - Fill-Rect Query](https://atcoder.jp/contests/abc464/tasks/abc464_e)

```c++
void ap() {
	int n, m, q; cin >> n >> m >> q;
    vector<vector<int>> a(n, vector<int>(m, -1));
    vector<char> b(q);
    for (int i = 0; i < q; i++) {
        int x, y; cin >> x >> y;
        char c; cin >> c;

        x--, y--;
        a[x][y] = i;
        b[i] = c;
    }

    for (int i = n - 1; i >= 0; i--) {
        for (int j = m - 1; j >= 0; j--) {
            if (i > 0) a[i - 1][j] = max(a[i - 1][j], a[i][j]);

            if (j > 0) a[i][j - 1] = max(a[i][j - 1], a[i][j]);

        }
    }

    for (int i = 0; i < n; i++) {
        string s = "";
        for (int j = 0; j < m; j++) {
            if (a[i][j] == -1) {
                s += "A";
                continue;
            }
            
            s += b[a[i][j]];
        }
        cout << s << endl;
    }

	return ;
}
```
