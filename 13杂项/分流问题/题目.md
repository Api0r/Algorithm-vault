
处理 一次操作的传递问题、连续区间问题
都可以**考虑其独立性，无后效性，看是否可以只考虑相邻的两个，而其操作不会影响后续的操作**

# 1. 矩阵按反对角线分层

[L-Matrix_2026牛客暑期多校训练营5](https://ac.nowcoder.com/acm/contest/133880/L)

- 设最终值为x，操作次数为k
- 则`a1,1 + k = x` 
- `a1,1`只会向下，向右流，因此有`a1,2 + a2,1 + k = 2x`
- 解得`k = a1,2 + a2,1 - 2 * a1,1` , `x = a1,1 + k`

- 即最终值x，操作次数k都为定值，并已知

- 在每一个`i + j = s` 的层s中都满足流到当前层的次数为k
- 可视化的讲，流从左上到右下，一层一层的流，且没有损失

- 发现每一层都是独立的，**无后效性，即当前层处理完后，其流动操作如何不会影响后续的处理**
- 而每一层都是由前一层流过来的，因此只需要处理相邻的每层即可

- 处理相邻的两层时，发现，当前层其实由上一层严格决定好了
- 细节上，由于最上边的数只能向右传递，而最终值确定，则向下传递的值也知道，依次都知道
- 我们可以枚举层数s，`s : [2, n + m]` ，然后从上到下枚举行，则列自然已知
- 从上到下，维护向下的值d即可，d确定，则向右的值也确定

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;
	vector<vector<int>> a(n + 1, vector<int> (m + 1));
	for (int i = 1; i <= n; i++) 
		for (int j = 1; j <= m; j++) 
			cin >> a[i][j];

	if (n == 1 || m == 1) {
		bool ok = true;
		int x = a[1][1];
		for (int i = 1; i <= n; i++) 
			for (int j = 1; j <= m; j++) 
				if (a[i][j] != x) {
					ok = false;
					break;
				}

		if (!ok) {
			cout << "-1\n";
		} else {
			cout << "0\n";
		}
		return;
	}
	
	
	int k = a[1][2] + a[2][1] - 2 * a[1][1];
	int x = a[1][1] + k;

	if (k < 0) {
		cout << "-1\n";
		return;
	}

	for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (x - a[i][j] < 0) {
                cout << "-1\n";
                return;
            }
        }
    }

	for (int s = 3; s <= n + m; s++) {
		int L = max(1LL, s - 1 - m), R = min(n, s - 1);

		int d = 0;
		for (int i = L; i <= R; i++) {
			int pre = s  - 1 - i;
			int cur = s - i;

			int b = (pre < 1 || pre > m) ? 0 : x - a[i][pre];
			int c = (cur < 1 || cur > m) ? 0 : x - a[i][cur];

			int left = c - d;
			if (left < 0) {
				cout << -1 << endl;
				return;
			}

			if (b - left < 0) {
				cout << -1 << endl;
				return;
			}

			d = b - left;
		}

		if (d != 0) {
			cout << -1 << endl;
			return;
		}

	}

	cout << k << endl;
	
	
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 2. 前缀池维护

[又一道纸笔问题 - 题目 - QOJ.ac](https://qoj.ac/contest/4582/problem/22910)

题意：
- 可选连续的列区间，不减的行区间、
- 其构成一系列点，将这些点的值减1
- 求将矩阵变为零矩阵的最少操作次数

思路：
- 假设每个点都单独处理，其一定可行，答案为所有数的累加和
- 在此基础上，思考如何将贡献合并

- 列区间连续，相邻两个列区间相互独立，**则可以只考虑每两个相邻的列，其无后效性**
- 行区间不减，在相邻两列中，对于右边列的任意一点`(r, c)` ，其可以合并左列中`i <= r` 的数
- 越往下，可合并的池子越大，贪心地能合并就合并
- 枚举一遍列，处理完相邻两列即可



```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));

    int sum = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];

            sum += a[i][j];
        }
    }

    for (int j = 0; j < m - 1; j++) {
        int pool = 0;

        for (int i = 0; i < n; i++) {
            pool += a[i][j];

            int take = min(a[i][j + 1], pool);
            pool -= take;

            sum -= take;
        }
    }

    cout << sum << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
