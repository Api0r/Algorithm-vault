
# 分段 - 首尾状态转移
[E-小红的不相邻取数_牛客周赛 Round 152](https://ac.nowcoder.com/acm/contest/137448/E)

```c++
void ap() {
	int n; cin >> n;
    vector<int> a(n), b(n);
    for (int & i : a) cin >> i;
    for (int & i : b) cin >> i;
    
    vector<vector<int>> dp(2, vector<int>(2, -1e18));
    dp[0][0] = 0;
    for (int i = 0; i < n; i++) {
        //当前的这一段，首尾01，的sum
        vector<vector<int>> sum(2, vector<int>(2, -1e18));
        if (a[i] <= 0) {
            sum[0][0] = 0;
        } else {
            if (b[i] == 1) {
                sum[0][0] = 0;
                sum[1][1] = a[i];
            } else {
                sum[0][0] = (b[i] - 1) / 2 * a[i];
                sum[0][1] = (b[i]) / 2 * a[i];
                sum[1][0] = b[i] / 2 * a[i];
                
//                 sum[1][1]
                if (b[i] & 1) {
                    sum[1][1] = (b[i] / 2 + 1) * a[i];
                } else {
                    //偶数不能首尾都选，-1e18
                }
            }
        }
        
        vector<vector<int>> ndp(2, vector<int>(2, -1e18));
        //当前首尾状态
        for (int j = 0; j < 2; j++) {
            for (int k = 0; k < 2; k++) {
                if (sum[j][k] == -1e18) {//当前状态不合法
                    continue;
                }
                
                //前一个状态的
                for (int s = 0; s < 2; s++) {
                    for (int t = 0; t < 2; t++) {
                        if (t == 1 && j == 1) continue;
                        if (dp[s][t] == -1e18) continue;
                        
                        ndp[j][k] = max(ndp[j][k], dp[s][t] + sum[j][k]);
                        
                    }
                }
            }
        }
        dp = move(ndp);
    }
    
    int ans = 0;
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            ans = max(ans, dp[i][j]);
        }
    }
    cout << ans << endl;
    
	return ;
}
```
