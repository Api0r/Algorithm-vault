
# 一、混合背包 - 刷表法（向后推）

[E-小红打游戏_牛客小白月赛131](https://ac.nowcoder.com/acm/contest/131416/E)

### 状态定义与转移

设 $dp[i][j]$ 表示获得**至少** $i$ 份材料 A 且**至少** $j$ 份材料 B 时，所需要花费的**最小体力**。

在实际计算中，由于材料数超过目标 $x$ 或 $y$ 时的效果等同于 $x$ 或 $y$，因此我们在状态转移时，会将超出部分的目标值自动向 $x$ 和 $y$ 取 $\min$。

### 算法步骤

#### 1. 初始化

- $dp[0][0] = 0$，表示不需要任何材料时，花费的最小体力为 0。
    
- 对于其余所有的 $i \in [0, x], j \in [0, y]$（除 $dp[0][0]$ 外），全部初始化为 $\infty$（在代码中通常设为 `1e18`），表示这些状态在初始时均不可达。
    

#### 2. 状态转移

我们需要分阶段处理限定关卡与普通关卡。设当前处理的关卡获得的材料 A 为 $a$，材料 B 为 $b$，消耗体力为 $c$：

- **限定关卡（前 $k$ 个关卡，01背包模型）**：
    
    由于每个限定关卡最多只能游玩一次，为了防止当前关卡的结果被重复累加，我们需要**倒序**遍历当前的所有状态 $(j, t)$：
    
    对于 $j$ 从 $x$ 降序到 $0$，$t$ 从 $y$ 降序到 $0$：
    
    $$dp[\min(x, j + a)][\min(y, t + b)] = \min(dp[\min(x, j + a)][\min(y, t + b)], dp[j][t] + c)$$
    
    其中转移必须满足前置条件：
    
    $$\neg (dp[j][t] == \infty)$$
    
- **普通关卡（后 $n-k$ 个关卡，完全背包模型）**：
    
    由于普通关卡可以游玩任意多次，我们可以利用本轮已经更新过的状态继续向下转移，因此需要**正序**遍历当前的所有状态 $(j, t)$：
    
    对于 $j$ 从 $0$ 升序到 $x$，$t$ 从 $0$ 升序到 $y$：
    
    $$dp[\min(x, j + a)][\min(y, t + b)] = \min(dp[\min(x, j + a)][\min(y, t + b)], dp[j][t] + c)$$
    
    其中转移同样必须满足前置条件：
    
    $$\neg (dp[j][t] == \infty)$$
    

#### 3. 答案

若最终状态 $dp[x][y] == \infty$，则说明永远无法达成目标，输出 `-1`；否则输出：

$$dp[x][y]$$

- **完全背包**：每种物品有**无限多个**。
    
- **多重背包**：每种物品有**限定的多个**（比如物品 A 只有 3 个，物品 B 只有 5 个）。



### 刷表法 - 向后看

$$dp[\text{未来状态}] = \min(dp[\text{未来状态}], dp[\text{当前状态}] + c)$$


```c++
void ap() {
	int n, k; cin >> n >> k;
    int x, y; cin >> x >> y;
    vector<array<int, 3>> a(n);
    for (int i = 0; i < n; i++) cin >> a[i][0] >> a[i][1] >> a[i][2];
    
    vector<vector<int>> dp(x + 1, vector<int> (y + 1, 1e18));
    dp[0][0] = 0;
    
    //01背包 - 逆序遍历 背包容量
    for (int i = 0; i < k; i++) {
        for (int j = x; j >= 0; j--) {
            for (int t = y; t >= 0; t--) {
                if (dp[j][t] == 1e18) continue;//当前状态不可达
                
                int nx = min(x, j + a[i][0]);
                int ny = min(y, t + a[i][1]);
                
                dp[nx][ny] = min(dp[nx][ny], dp[j][t] + a[i][2]);
                
            }
        }
    }
    
    //完全背包 - 正序遍历
    for (int i = k; i < n; i++) {
        for (int j = 0; j <= x; j++) {
            for (int t = 0; t <= y; t++) {
                if (dp[j][t] == 1e18) continue;
                
                int nx = min(x, j + a[i][0]);
                int ny = min(y, t + a[i][1]);
                
                dp[nx][ny] = min(dp[nx][ny], dp[j][t] + a[i][2]);
            }
        }
    }
    
    if (dp[x][y] == 1e18) {
        cout << -1 << endl;
    } else {
        cout << dp[x][y] << endl;
    }
    
    
	return ;
}
```


### 拉表法 - 向前看

```c++
void ap() {
    int n, k; cin >> n >> k;
    int x, y; cin >> x >> y;
    vector<array<int, 3>> a(n);
    for (int i = 0; i < n; i++) cin >> a[i][0] >> a[i][1] >> a[i][2];
    
    // dp[j][t] 表示至少获得 j 份 A 和 t 份 B 的最小体力
    vector<vector<int>> dp(x + 1, vector<int> (y + 1, 1e18));
    dp[0][0] = 0;
    
    // 1. 限定关卡 (01 背包) -> 必须倒序遍历状态
    for (int i = 0; i < k; i++) {
        int cost_a = a[i][0];
        int cost_b = a[i][1];
        int energy = a[i][2];
        
        for (int j = x; j >= 0; j--) {
            for (int t = y; t >= 0; t--) {
                // 向前看：当前状态可以由打这一关之前的状态转移而来
                // 之前的状态至少需要有 max(0LL, j - cost_a) 和 max(0LL, t - cost_b)
                int pre_j = max(0LL, j - cost_a);
                int pre_t = max(0LL, t - cost_b);
                
                if (dp[pre_j][pre_t] == 1e18) continue; // 前置状态不可达
                
                dp[j][t] = min(dp[j][t], dp[pre_j][pre_t] + energy);
            }
        }
    }
    
    // 2. 普通关卡 (完全背包) -> 必须正序遍历状态
    for (int i = k; i < n; i++) {
        int cost_a = a[i][0];
        int cost_b = a[i][1];
        int energy = a[i][2];
        
        for (int j = 0; j <= x; j++) {
            for (int t = 0; t <= y; t++) {
                int pre_j = max(0LL, j - cost_a);
                int pre_t = max(0LL, t - cost_b);
                
                if (dp[pre_j][pre_t] == 1e18) continue; // 前置状态不可达
                
                dp[j][t] = min(dp[j][t], dp[pre_j][pre_t] + energy);
            }
        }
    }
    
    // 输出结果
    if (dp[x][y] == 1e18) {
        cout << -1 << endl;
    } else {
        cout << dp[x][y] << endl;
    }
    return;
}
```
