
# 1.拓扑排序 + 优先队列

[Permutation Inversions - 题目 - QOJ.ac](https://qoj.ac/contest/4071/problem/20018)

inversion 逆序对

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;

	vector<vector<int>> g(n + 1);
	vector<int> in(n + 1);

	for (int i = 0; i < m; i++) {
		int l, r; cin >> l >> r;
		
		int pre; cin >> pre;
		for (int j = l; j < r; j++) {
			int x; cin >> x; 

			g[pre].push_back(x);
			in[x]++;
			pre = x;
		}

	}

	priority_queue<int, vector<int>, greater<int>> pq;
	for (int i = 1; i <= n; i++) {
		if (in[i] == 0) pq.push(i);
	}

	vector<int> ord;
	vector<int> ans(n + 1);
	int k = 1;
	while (!pq.empty()) {
		int u = pq.top();
		pq.pop();

		ans[u] = k++;
		ord.push_back(u);
		for (int v : g[u]) {
			if (--in[v] == 0) pq.push(v);
		}
	}


	if (ord.size() != n) {
		cout << -1 << endl;
		return;
	}

	for (int i = 1; i <= n; i++) cout << ans[i] << ' ';
	cout << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 2.难点在与想到解离的关系

[P17524 [ECUSTPC 2026 Fall] 净化行动 3 - 洛谷](https://www.luogu.com.cn/problem/P17524)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    map<int, vector<array<int,3>> > x, y; // x, (x, y), idx
    bool ok = true;
    for (int i = 0; i < n; i++) {
        int a, b; cin >> a >> b;

        if (!ok) continue;
        
        x[a].push_back({a, b, i});
        y[b].push_back({a, b, i});

        if (x[a].size() > 3 || y[b].size() > 3) {
            ok = false;
        }
    }
    if (!ok) {
        cout << "NO" << endl;
        return;
    }

    vector<vector<int>> g(n);
    vector<int> in(n);
    for (auto & [x, p] : x) {
        if (p.size() < 3) continue;

        sort(p.begin(), p.end());

        int mid = p[1][2], a = p[0][2], b = p[2][2];

        // cout << mid << " -> " << a << ' ' << b << endl;
        
        g[mid].push_back(a);
        g[mid].push_back(b);
        in[a]++;
        in[b]++;
    }

    for (auto & [y, p] : y) {
        if (p.size() < 3) continue;

        sort(p.begin(), p.end());

        int mid = p[1][2], a = p[0][2], b = p[2][2];

        // cout << mid << " -> " << a << ' ' << b << endl;
        
        g[mid].push_back(a);
        g[mid].push_back(b);
        in[a]++;
        in[b]++;
    }

    queue<int> q;
    for (int i = 0; i < n; i++) {
        if (in[i] == 0) {
            q.push(i);
            // cout << i << ' ';
        }
    }
    // cout << endl;

    int cnt = 0;
    while (!q.empty()) {
        int u = q.front();
        q.pop();

        cnt++;
        for (int v : g[u]) {
            if (--in[v] == 0) q.push(v);
        }
    }

    if (cnt != n) {
        cout << "NO" << endl;
    } else {
        cout << "YES" << endl;
    }
    
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
