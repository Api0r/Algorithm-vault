
# 1.拓扑排序 + 优先队列

[Permutation Inversions - 题目 - QOJ.ac](https://qoj.ac/contest/4071/problem/20018)

inversion 逆序对

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;

	vector<vector<int>> g(n + 1);
	vector<int> in(n + 1);

	for (int i = 0; i < m; i++) {
		int l, r; cin >> l >> r;
		
		int pre; cin >> pre;
		for (int j = l; j < r; j++) {
			int x; cin >> x; 

			g[pre].push_back(x);
			in[x]++;
			pre = x;
		}

	}

	priority_queue<int, vector<int>, greater<int>> pq;
	for (int i = 1; i <= n; i++) {
		if (in[i] == 0) pq.push(i);
	}

	vector<int> ord;
	vector<int> ans(n + 1);
	int k = 1;
	while (!pq.empty()) {
		int u = pq.top();
		pq.pop();

		ans[u] = k++;
		ord.push_back(u);
		for (int v : g[u]) {
			if (--in[v] == 0) pq.push(v);
		}
	}


	if (ord.size() != n) {
		cout << -1 << endl;
		return;
	}

	for (int i = 1; i <= n; i++) cout << ans[i] << ' ';
	cout << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
