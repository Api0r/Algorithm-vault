```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int dx[4] = {-1, 1, 0, 0}, dy[4] = {0, 0, -1, 1};

void ap() {
	int n, k; cin >> n >> k;
    string s; cin >> s;
    
    vector<array<int, 2>> a(k);
    for (int i = 0; i < k; i++) {
        cin >> a[i][0] >> a[i][1];
    }
    
    map<char, int> op;
    op['U'] = 0, op['D'] = 1, op['L'] = 2, op['R'] = 3;
    
    int x = 0, y = 0;
    int dis = 0;
    for (char & c : s) {
        if (c == 'N') {
            dis++;
        } else {
            int k = op[c];
            x += dx[k];
            y += dy[k];
        }
        
        for (auto & [x1, y1] : a) {
            if (abs(x1 - x) + abs(y1 - y) <= dis) {
                cout << "Yes" << endl;
                return;
            }
        }
        
        
    }
    
    cout << "No" << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
