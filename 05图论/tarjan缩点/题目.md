# 4. 缩点 + 点权

[P1262 [POI 1996 R3] 间谍网络 - 洛谷](https://www.luogu.com.cn/problem/P1262#ide)

`work()` 中只对能被收买的间谍跑 Tarjan

如果按照正常tarjan的思路，会发现控制不了的人可能出现在不是入度为0的块中，
此时，需要标记能控制的入度为0的块，然后从`cnt -> 1` 拓扑排序的顺序依次控制其后置的块，
最后再从`1 -> n` 遍历，找到第一个不能控制的

```c++
#include <bits/stdc++.h>

#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;


int n, now, cnt;
vector<vector<int>> ver;
vector<int> dfn, low, col, S;
vector<int> mn, mnp;

map<int, int> cost;

//     mn.resize(n + 1, 1e18);
//     dfn.resize(n + 1, -1);
//     col.resize(n + 1, -1);
//     now = cnt = 0;
// }
void add(int x, int y) {
    ver[x].push_back(y);
}
void tarjan(int x) {
    dfn[x] = low[x] = now++;
    S.push_back(x);
    for (auto y : ver[x]) {
        if (dfn[y] == -1) {
            tarjan(y);
            low[x] = min(low[x], low[y]);
        } else if (col[y] == -1) {
            low[x] = min(low[x], dfn[y]);
        }

    if (dfn[x] == low[x]) {
        int pre;
        cnt++;
        do {
            pre = S.back();
            col[pre] = cnt;
            if (cost.count(pre)) mn[cnt] = min(mn[cnt], cost[pre]);
            mnp[cnt] = min(mnp[cnt], pre);
            S.pop_back();
        } while (pre != x);
    }
}
void work() { // [cnt 新图的顶点数量]
    for (int i = 1; i <= n; i++) { // 避免图不连通
        if (cost.count(i) && dfn[i] == -1) {
            tarjan(i);
        }
    }

    // vector<int> siz(cnt + 1); // siz 每个 scc 中点的数量
```

# 8.  数学 + 缩点 + 拓扑排序

[E - lt and le](https://atcoder.jp/contests/abc478/tasks/abc478_e)

小 -> 大

一个强连通分量内，任意两点相互可达
- 如果scc内出现了 `<` 的边，那么这个块内部无法构造，直接输出no
- 否则，总可以缩点成拓扑图  (拓扑图的边存在`<` 的情况，不能直接贪心的全部相等)
- 在拓扑图上将数值填上去即可

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;

struct SCC {
    int n, now, cnt;
    vector<vector<int>> ver;
    vector<int> dfn, low, col, S;

    SCC(int n) : n(n), ver(n + 1), low(n + 1) {
        dfn.resize(n + 1, -1);
        col.resize(n + 1, -1);
        now = cnt = 0;
    }
    void add(int x, int y) {
        ver[x].push_back(y);
    }
    void tarjan(int x) {
        dfn[x] = low[x] = now++;
        S.push_back(x);
        for (auto y : ver[x]) {
            if (dfn[y] == -1) {
                tarjan(y);
                low[x] = min(low[x], low[y]);
            } else if (col[y] == -1) {
                low[x] = min(low[x], dfn[y]);
            }
        }
        if (dfn[x] == low[x]) {
            int pre;
            cnt++;
            do {
                pre = S.back();
                col[pre] = cnt;
                S.pop_back();
            } while (pre != x);
        }
    }
    auto work() { // [cnt 新图的顶点数量]
        for (int i = 1; i <= n; i++) { // 避免图不连通
            if (dfn[i] == -1) {
                tarjan(i);
            }
        }

    //     vector<int> siz(cnt + 1); // siz 每个 scc 中点的数量
        vector<vector<int>> adj(cnt + 1);
        for (int i = 1; i <= n; i++) {
            // siz[col[i]]++;
            for (auto j : ver[i]) {
                int x = col[i], y = col[j];
                if (x != y) {
                    adj[x].push_back(y);
                }
            }
        }
        return adj;
    }
};
```
