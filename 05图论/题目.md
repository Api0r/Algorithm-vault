
# 1.区间01flip - 异或差分 - 差分建图 - 边双连通 - 割边

[异或王国 - 题目 - QOJ.ac](https://qoj.ac/contest/4582/problem/22916)

反转区间 $[l, r]$ 是对区间每个位置异或 $1$。我们希望引入一个**差分序列**，使得原本对一整段区间的修改，在差分序列上变成**恰好两个端点的单点修改**。

只要每次操作只改变差分数组上的**两个位置** $u$ 和 $v$，就可以把一次操作抽象为连接 $(u, v)$ 的一条**无向边**。

此时：

- 对边 $(u, v)$ 进行操作 $\iff$ $d[u]$ 和 $d[v]$ 同时异或 $1$。
    
- 只要全图每个连通块的异或和为 $0$，就能通过若干次边操作将整个差分数组消成全 $0$。

- 容易证得：一条路径上的边都操作一遍，最终改变的只有起点和终点
- 因此，一个连通块中，任意两个为1的都可以变为两个0。即只需要连通块中的1的个数为偶数即可

禁止使用一条边，相当于删这一条边
如果这一条边属于一个边双连通分量内，显然不会影响连通性
缩点后容易看出，只有割边被删除，会影响连通性，此时只需要检查两个连通块各自的异或和是否为0。发现，如果不删边的时候合法，则一整个连通块的异或和为0，割边断开连通后，只需要判断一边的异或和是否为0即可，可以预处理一遍子树的异或和。

# 3. 点是否在路径上lca + 先后移动关系 + 拓扑图

[E-Faith Flower（Easy Version）_牛客练习赛159](https://ac.nowcoder.com/acm/contest/142560/E)

判断一个节点是否在球 i 的移动路径上，可以利用树上距离。如果起点到某节点的距离，加上该节点到终点的距离等于起点到终点的距离，那么该节点就在路径上。可以使用 LCA 快速计算距离。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

//如果球j的起点在球i的路径上，则j需要比i先移动，j -> i
//如果球j的终点在球i的路径上，则i比j先移动, i -> j
//移动顺序需要满足上述拓扑关系
//可以暴力枚举每两个球的关系，建成拓扑图，最后跑一遍即可

//top[x] = up : 节点x所在重链的头节点是up
//seg[cntd] = x : dfn编号为cntd的节点为x

struct HLD {
    int n, cntd;
    vector<vector<int>> ver;
    vector<int> siz, dep;
    vector<int> top, son, parent;
    vector<int> dfn, seg;

    HLD(int n) : n(n) {
        ver.resize(n + 1);
        siz.resize(n + 1);
        dep.resize(n + 1);

        top.resize(n + 1);
        son.resize(n + 1);
        parent.resize(n + 1);

        dfn.resize(n + 1);
        seg.resize(n + 1);
        cntd = 0;
    }
    void add(int x, int y) { // 建立双向边
        ver[x].push_back(y);
        ver[y].push_back(x);
    }
    void dfs1(int x) {
        siz[x] = 1;
        dep[x] = dep[parent[x]] + 1;
        for (auto y : ver[x]) {
            if (y == parent[x]) continue;
            parent[y] = x;
            dfs1(y);
            siz[x] += siz[y];
            if (siz[y] > siz[son[x]]) {
                son[x] = y;
            }
        }
    }
    // 来到节点x，节点x所在重链的头节点是up
    void dfs2(int x, int up) {
        top[x] = up;
        dfn[x] = ++cntd;
        seg[cntd] = x;
        if (son[x]) dfs2(son[x], up);
        for (auto y : ver[x]) {
            if (y == parent[x] || y == son[x]) continue;
            dfs2(y, y);
        }
    }
    int lca(int x, int y) {
        while (top[x] != top[y]) {
            if (dep[top[x]] > dep[top[y]]) {
                x = parent[top[x]];
            } else {
                y = parent[top[y]];
            }
        }
        return dep[x] < dep[y] ? x : y;
    }
    int clac(int x, int y) { // 查询两点间距离
        return dep[x] + dep[y] - 2 * dep[lca(x, y)];
    }
    
    //x是y的祖先节点，求x的直系儿子中 为y的祖先节点的直系儿子
    // 前提：x 必须是 y 的真祖先 (dep[x] < dep[y] 且 lca(x, y) == x)
    int sont(int x, int y) {
        while (top[x] != top[y]) {
            // 如果 y 所在重链的链顶就是 x 的直系儿子
            if (parent[top[y]] == x) {
                return top[y];
            }
            // 否则将 y 沿着链向上跳
            y = parent[top[y]];
        }
        // 此时 x 和 y 位于同一条重链上，x 往 y 方向的儿子必然是 x 的重儿子
        return son[x];
    }
    
    void work(int root = 1) { // 在此初始化
        dfs1(root);
        dfs2(root, root);
    }
};

void ap() {
	int n, m; cin >> n >> m;
    
    HLD hld(n);
    
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        
        hld.add(u, v);
    }
    
    hld.work(1);
    
    vector<array<int, 2>> a(m + 1);
    for (int i = 1; i <= m; i++) {
        cin >> a[i][0] >> a[i][1];
    }
    
    //如果球j的起点在球i的路径上，则j需要比i先移动，j -> i
    //如果球j的终点在球i的路径上，则i比j先移动, i -> j
    
    vector<vector<int>> g(m + 1);
    vector<int> in(m + 1);
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= m; j++) if (i != j) {
            int x = a[i][0], y = a[i][1];
            int s = a[j][0], t = a[j][1];
            
            if (hld.clac(x, s) + hld.clac(s, y) == hld.clac(x, y)) {
                g[j].push_back(i);
                in[i]++;
            }
            
            if (hld.clac(x, t) + hld.clac(t, y) == hld.clac(x, y)) {
                g[i].push_back(j);
                in[j]++;
            }
        }
    }
    
    queue<int> q;
    for (int i = 1; i <= m; i++) if (in[i] == 0) q.push(i);
    
    vector<int> ans;
    while (q.size()) {
        auto x = q.front();
        q.pop();
        
        ans.push_back(x);
        
        for (int & y : g[x]) {
            if (--in[y] == 0) {
                q.push(y);
            }
        }
    }
    
    if (ans.size() != m) {
        cout << -1 << endl;
    } else {
        for (int x : ans) cout << x << ' ';
        cout << endl;
    }
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 4.  数学 + 缩点 + 拓扑图

[E - lt and le](https://atcoder.jp/contests/abc478/tasks/abc478_e)

小 -> 大

一个强连通分量内，任意两点相互可达
- 如果scc内出现了 `<` 的边，那么这个块内部无法构造，直接输出no
- 否则，总可以缩点成拓扑图  (拓扑图的边存在`<` 的情况，不能直接贪心的全部相等)
- 在拓扑图上将数值填上去即可

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

struct SCC {
    int n, now, cnt;
    vector<vector<int>> ver;
    vector<int> dfn, low, col, S;

    SCC(int n) : n(n), ver(n + 1), low(n + 1) {
        dfn.resize(n + 1, -1);
        col.resize(n + 1, -1);
        now = cnt = 0;
    }
    void add(int x, int y) {
        ver[x].push_back(y);
    }
    void tarjan(int x) {
        dfn[x] = low[x] = now++;
        S.push_back(x);
        for (auto y : ver[x]) {
            if (dfn[y] == -1) {
                tarjan(y);
                low[x] = min(low[x], low[y]);
            } else if (col[y] == -1) {
                low[x] = min(low[x], dfn[y]);
            }
        }
        if (dfn[x] == low[x]) {
            int pre;
            cnt++;
            do {
                pre = S.back();
                col[pre] = cnt;
                S.pop_back();
            } while (pre != x);
        }
    }
    auto work() { // [cnt 新图的顶点数量]
        for (int i = 1; i <= n; i++) { // 避免图不连通
            if (dfn[i] == -1) {
                tarjan(i);
            }
        }

    //     vector<int> siz(cnt + 1); // siz 每个 scc 中点的数量
        vector<vector<int>> adj(cnt + 1);
        for (int i = 1; i <= n; i++) {
            // siz[col[i]]++;
            for (auto j : ver[i]) {
                int x = col[i], y = col[j];
                if (x != y) {
                    adj[x].push_back(y);
                }
            }
        }
        return adj;
    }
};

void ap() {

        scc.add(u, v);
    }

    auto g = scc.work();

    int cnt = scc.cnt;

    vector<int> in(cnt + 1, 0);
    for (int i = 0; i < q; i++) {
        int x = scc.col[a[i][0]], y = scc.col[a[i][1]];

        if (x == y && a[i][2] == 1) {
            cout << "No" << endl;
            return;
        } else if (x != y) {
            in[y]++;
        }
    }

    // cout << scc.col[1] << ' ' << scc.col[4] << endl;
    // cout << in[scc.col[1]] << ' ' << in[scc.col[4]] << endl;

    cout << "Yes" << endl;

    vector<int> t(cnt + 1);// col, val
    queue<int> qq;
    for (int i = 1; i <= cnt; i++) if (in[i] == 0) {
        // cout << "in:" << i << ' ';
        qq.push(i);
    }
    // cout << endl;

    while (!qq.empty()) {
        auto u = qq.front();
        qq.pop();

        t[u] = cur++;
        for (int v : g[u]) {
            if (--in[v] == 0) qq.push(v);
        }
    }

    for (int i = 1; i <= n; i++) {
        cout << t[scc.col[i]] << ' ';
    }
```

# 5. 分割点建图 + BFS最短路

[C-Dawn Flower_牛客练习赛159](https://ac.nowcoder.com/acm/contest/142560/C)

考虑分割点作为点建图

tips:后缀dp

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    map<int, vector<int>> mp;
    for (int i = 1; i <= n; i++) {
        int x = i - a[i];
        
        if (x >= 1) mp[x].push_back(i);
    }
    
    vector<int> dis(n + 1, -1), pre(n + 1, -1);
    queue<int> q;
    
    dis[0] = 0;
    q.push(0);
    
    while (q.size()) {
        auto x = q.front();
        q.pop();
        
        if (x == n) break;
        
        int t = x + 1 + a[x + 1];
        
        if (mp.count(t)) {
            for (int y : mp[t]) {
                if (dis[y] == -1) {
                    dis[y] = dis[x] + 1;
                    pre[y] = x;

                    q.push(y);
                }
            }
            mp.erase(t);
        }
    }
    
    if (dis[n] == -1) {
        cout << -1 << endl;
        return;
    } 
    
    vector<int> ans;
    for (int i = n; i; i = pre[i]) ans.push_back(i);
    
    reverse(ans.begin(), ans.end());
    
    cout << ans.size() << endl;
    for (int & x : ans) cout << x << ' ';
    cout << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
