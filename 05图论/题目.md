
# 1.区间01flip - 异或差分 - 差分建图 - 边双连通 - 割边

[异或王国 - 题目 - QOJ.ac](https://qoj.ac/contest/4582/problem/22916)

反转区间 $[l, r]$ 是对区间每个位置异或 $1$。我们希望引入一个**差分序列**，使得原本对一整段区间的修改，在差分序列上变成**恰好两个端点的单点修改**。

只要每次操作只改变差分数组上的**两个位置** $u$ 和 $v$，就可以把一次操作抽象为连接 $(u, v)$ 的一条**无向边**。

此时：

- 对边 $(u, v)$ 进行操作 $\iff$ $d[u]$ 和 $d[v]$ 同时异或 $1$。
    
- 只要全图每个连通块的异或和为 $0$，就能通过若干次边操作将整个差分数组消成全 $0$。

- 容易证得：一条路径上的边都操作一遍，最终改变的只有起点和终点
- 因此，一个连通块中，任意两个为1的都可以变为两个0。即只需要连通块中的1的个数为偶数即可

禁止使用一条边，相当于删这一条边
如果这一条边属于一个边双连通分量内，显然不会影响连通性
缩点后容易看出，只有割边被删除，会影响连通性，此时只需要检查两个连通块各自的异或和是否为0。发现，如果不删边的时候合法，则一整个连通块的异或和为0，割边断开连通后，只需要判断一边的异或和是否为0即可，可以预处理一遍子树的异或和。

struct EDCC {

        m = now = cnt = 0;
    }
    void add(int x, int y) { // 和 scc 相比多了一条连边
        ver[x].push_back({y, m});

        dfn[x] = low[x] = ++now;

            if (!dfn[y]) {

            cnt++;

                pre = S.back();
                col[pre] = cnt;
                S.pop_back();

void ap() {
	int n, m; cin >> n >> m;
    string a, b; cin >> a >> b;

    vector<int> c(n + 2, 0);
    for (int i = 1; i <= n; i++) {
        c[i] = (a[i - 1] - '0') ^ (b[i - 1] - '0');
    }

    vector<int> d(n + 2);//1 - n + 1
    for (int i = 1; i <= n + 1; i++) {
        d[i] = c[i - 1] ^ c[i];     
    }

    EDCC edcc(n + 1);

    vector<array<int, 2>> e(m);
    vector<vector<int>> g(n + 2);
    for (int i = 0; i < m; i++) {
        int l, r; cin >> l >> r;
        r++;

        e[i] = {l, r};
        edcc.add(l, r); 
        g[l].push_back(r);
        g[r].push_back(l);
    }


    edcc.work();

    int sum = 0;
    vector<int> vis(n + 2, 0);
    auto dfs = [&](auto&& dfs, int x, int fa) -> void {
        vis[x] = 1;
        sum ^= d[x];
        for (int & y : g[x]) if (y != fa) {
            if (!vis[y]) dfs(dfs, y, x);
        }
    };

    bool ok = true;
    for (int i = 1; i <= n + 1; i++) {
        if (!vis[i]) {
            sum = 0;
            dfs(dfs, i, 0);
            if (sum != 0) ok = false;
        }
    }

    if (!ok) {
        for (int i = 0; i < m; i++) cout << 0;
        cout << endl;
        return;
    }

    int cnt = edcc.cnt;

    vector<int> w(cnt + 1, 0);
    for (int i = 1; i <= n + 1; i++) {
        w[edcc.col[i]] ^= d[i];
    }

    vector<vector<int>> gg(cnt + 1);
    for (int i = 0; i < m; i++) {
        int x = e[i][0], y = e[i][1];
        int u = edcc.col[x], v = edcc.col[y];

        if (u != v) {
            gg[u].push_back(v);
            gg[v].push_back(u);
        }
    }

    vector<int> son(cnt + 1, 0), dep(cnt + 1, 1);
    vector<int> v(cnt + 1, 0);
    auto dfs1 = [&](auto&& dfs1, int x, int fa) -> void {
        v[x] = 1;
        son[x] = w[x];
        dep[x] = dep[fa] + 1;
        for (int & y : gg[x]) if (y != fa) {
            dfs1(dfs1, y, x);
            son[x] ^= son[y];
        }
    };
    for (int i = 1; i <= cnt; i++) {
        if (!v[i]) {
            dfs1(dfs1, i, 0);
        }
    }

    for (int i = 0; i < m; i++) {
        int x = e[i][0], y = e[i][1];
        int u = edcc.col[x], v = edcc.col[y];

        if (u == v) {
            cout << 1;
            continue;
        }

        if (dep[u] > dep[v]) swap(u, v);
        if (son[v]) {
            cout << 0;
        } else {
            cout << 1;
        }

    }
    cout << endl;


	return ;
}

signed main() {

	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
