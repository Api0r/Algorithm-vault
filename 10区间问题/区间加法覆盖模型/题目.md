```c++
void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;
    
    int ans = a[0];
    
    for (int i = 1; i < n; i++) {
        ans += max(0LL, a[i] - a[i - 1]);
    }
    
    cout << ans << endl;
    
	return ;
}
```

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

bool is2(int n) {
    return n > 0 && (n & (n - 1)) == 0;
}

void ap() {
	int n, m; cin >> n >> m;
    vector<int> a(n), b(n);
    for (int & i : a) cin >> i;
    for (int & i : b) cin >> i;

    bool ok = true;
    for (int i = 0; i < n; i++) {
        if (a[i] != 0 && b[i] == 0) {
            ok = false;
            break;
        }

        if (a[i] == 0 && b[i] != 0) {
            ok = false;
            break;
        }

        if (a[i] == 0 && b[i] == 0) {//!
            continue;
        }

        if (b[i] < a[i]) {
            ok = false;
            break;
        }
        //b[i] < a[i] < m

        if (b[i] > m) {
            ok = false;
            break;
        }
        //b[i] <= m

        if (b[i] % a[i] != 0 && b[i] != m) {
            ok = false;
            break;
        }
        //b[i] % a[i] != 0 -> b[i] = m
        /*b[i] % a[i] == 0 -> 1. b[i] = m -> b[i] / a[i] = ~
                              2. b[i] != m -> b[i] / a[i] = 2^n  
        */


        if (b[i] != m) {
            if (b[i] % a[i] != 0 || !is2(b[i] / a[i])) {
                ok = false;
                break;
            }
        }

    }

    if (!ok) {
        cout << -1 << endl;
        return;
    }

    vector<array<int, 2>> cnt(n);//cnt, T/F
    for (int i = 0; i < n; i++) {
        if (a[i] == b[i]) {
            if (a[i] == m || a[i] == 0) {
                cnt[i] = {0, 1};
            } else {
                cnt[i] = {0, -1};
            }
            continue;
        }

        //b[i] > a[i]
        if (b[i] == m) {
            int cur = a[i], c = 0;
            while (cur < m) {
                cur *= 2;
                c++;
            }
            cnt[i] = {c, 1}; // 只要达到 m，后续可做任意多次
        } else {
            int c = __builtin_ctzll(b[i] / a[i]);
            cnt[i] = {c, -1}; // 严格等于 < m 的某个值，只能精确做 c 次
        }
    }

	//如果前一个点可以无限选，那么可以贪到前一个点的前一个点
    int ans = cnt[0][0];
    int pre = cnt[0][0];
    for (int i = 1; i < n; i++) {
        int cur = 0;
        if (cnt[i][1] == 1) {
            cur = max(pre, cnt[i][0]);
        } else {
            cur = cnt[i][0];
        }

        ans += max(0LL, cur - pre);

        pre = cur;
    }

    cout << ans << endl;


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
