```c++
void ap() {

    map<int ,int> d;
    for (int i = 0; i < n; i++) {
        int l, r; cin >> l >> r;
        d[l - k]++;
        d[r + 1]--;
    }

        s += v;
        ans = max(ans, s);
```

# 二、区间调度
[Problem - A - Codeforces](https://codeforces.com/gym/695065/problem/A)
```c++
int cmp(const pair<int, int> & a, const pair<int, int> & b) {
    if (a.second != b.second) {
        return a.second < b.second;
    }

    return a.first < b.first;
}

void ap() {
	int n; cin >> n;
    vector<int> r(n);
    for (int & i : r) cin >> i;

    vector<pair<int, int>> d(n - 1);
    for (int i = 0; i < n - 1; i++) {
        cin >> d[i].first >> d[i].second;
    }

    vector<pair<int, int>> a(n);
    a[0].first = 0, a[0].second = 0;

    for (int i = 1; i < n; i++) {
        a[i].first = a[i - 1].first + d[i - 1].first;
        a[i].second = a[i - 1].second + d[i - 1].second;
    }

    vector<pair<int, int>> p(n);
    for (int i = 0; i < n; i++) {
        int x = a[i].first + a[i].second;
        p[i] = {x - r[i], x + r[i]};
    }

    sort(p.begin(), p.end(), cmp);

    int cnt = 0;
    int end = -1e18;
    for (auto [start, e] : p) {
        if (start > end) {
            cnt++;
            end = e;
        }
    }

    cout << cnt << endl;

	return ;
}
```

# 三、区间调度 + 二分
[D - Maximize the Gap](https://atcoder.jp/contests/abc463/tasks/abc463_d)
```c++
vector<array<int, 2>> a;
int k;

int cmp(const array<int, 2> & a, const array<int, 2> & b) {
    if (a[1] == b[1]) {
        return a[0] > b[0];
    }
    return a[1] < b[1];
}

bool check(int x) {
    int end = -1e9, cnt = 0;
    for (auto & [s, e] : a) {
        if (s >= end) {
            cnt++;
            end = e + x;
        }
    }

    return cnt >= k;
}

void ap() {
	int n; cin >> n >> k;
    a.resize(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i][0] >> a[i][1];
    }

    sort(a.begin(), a.end(), cmp);

    int l = 0, r = 1e18;
    while (l + 1 < r) {
        int mid = l + (r - l) / 2;
        if (check(mid)) {
            l = mid;
        } else {
            r = mid;
        }
    }

    if (l) {
        cout << l << endl;
    } else {
        cout << -1 << endl;
    }

	return ;
}
```


# 4.区间相交的对数

[4057. 统计相交区间对 II - 力扣（LeetCode）](https://leetcode.cn/problems/number-of-intersecting-interval-pairs-ii/description/)

```c++
class Solution {
    static bool cmp(const vector<int> & a, const vector<int> & b) {
        return a[1] < b[1];
    }

public:
    long long countIntersectingIntervals(vector<vector<int>>& a) {
        sort(a.begin(), a.end(), cmp);

        long long n = a.size();
        long long ans = 1LL * n * (n - 1) / 2;

        vector<int> b;
        for (auto & p : a) b.push_back(p[1]);

        for (auto& p : a) {
            int start = p[0];
            
            int j = lower_bound(b.begin(), b.end(), start) - b.begin() - 1;// < 
            ans -= j + 1;
        }
        return ans;
    }
};
```

# 6. 抽象成区间合并

https://codeforces.com/contest/2290/problem/D

- 记录每个对称位置的所需变化区间，`e -> b`
- 注意到，变化过程中相同的字符会增加，一起减小的次数一定最小
- 因此，从大到小变小的次数一定最小

- 想到区间合并问题
- 记录每个对称的所需变化区间，区间合并后，记录每个区间的贡献即可

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    string s; cin >> s;

    int l = 0, r = n - 1;
    int ans = 0;

    vector<array<int, 2>> a;
    while (l < r) {
        if (s[l] != s[r]) {
            int x = s[l] - 'a', y = s[r] - 'a';

            a.push_back({min(x, y), max(x, y)});
        }

        l++, r--;
    }

    if (a.size() == 0) {
        cout << 0 << endl;
        return;
    }

    sort(a.begin(), a.end());

    int st = a[0][0], ed = a[0][1];
    for (auto & e : a) {
        if (e[0] <= ed) {
            ed = max(ed, e[1]);
        } else {
            ans += ed - st;

            st = e[0], ed = e[1];
        }
    }
    ans += ed - st;

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
