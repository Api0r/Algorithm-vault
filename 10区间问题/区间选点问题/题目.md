```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

bool cmp(const array<int, 2> & a, const array<int, 2> & b) {
    return a[1] < b[1];
}

void ap() {
	int n, q; cin >> n >> q;

    vector<array<int, 2>> a(q);
    for (int i = 0; i < q; i++) cin >> a[i][0] >> a[i][1];

    sort(a.begin(), a.end(), cmp);

    int cnt = 0;
    int end = -1e18;
    for (int i = 0; i < q; i++) {
        if (a[i][0] > end) {
            cnt++;
            end = a[i][1];
        }
    }

    if (cnt == 0) {
        cout << 0 << endl;
        return;
    }

    int ans = __lg(cnt) + 1;
    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
