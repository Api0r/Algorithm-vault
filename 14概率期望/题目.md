# 1、数学 + 期望DP


[F-猴子排序的剪枝_牛客周赛 Round 152](https://ac.nowcoder.com/acm/contest/137448/F)

**知识点：概率期望、排列计数、动态规划**

把每一轮排序看成一次独立试验。设单轮成功概率为 $p$，单轮插入次数为 $X$，那么总期望为

$$\frac{E(X)}{p}$$

设每种数值的出现次数分别为 $c_1, c_2, \dots, c_m$。若把相同数值的元素仍视为不同位置，则总排列数为 $n!$，成功当且仅当整个排列按值非降，因此成功排列数为 $\prod c_i!$，即

$$p = \frac{\prod c_i!}{n!}$$

再用尾和公式计算 $E(X)$。要进行第 $i+1$ 次插入，等价于前 $i$ 个被插入的元素已经非降。设 $s_i$ 表示长度为 $i$ 的非降前缀方案数。若某个值组大小为 $c$，从其中选出并放入前缀 $j$ 个元素的有序方案数为 $A(c, j)$，不同值组之间的相对顺序已经被非降条件唯一确定，因此按值组做背包即可得到所有 $s_i$。

于是答案为

$$\sum_{i=0}^{n-1} \frac{s_i}{A(n,i)} \cdot \frac{1}{p} = \frac{\sum_{i=0}^{n-1} s_i \cdot (n-i)!}{\prod c_i!}$$

时间复杂度 $O(n^2)$。

```c++
void solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (auto &v : a) cin >> v;
    sort(a.begin(), a.end());
    vector<Z> dp(n + 1);
    dp[0] = 1;
    int sum = 0;
    Z down = 1;
    for (int l = 0; l < n;) {
        int r = l;
        while (r < n && a[r] == a[l]) ++r;
        int c = r - l;
        vector<Z> ndp(n + 1);
        for (int i = 0; i <= sum; ++i) {
            Z w = 1;
            for (int j = 0; j <= c; ++j) {
                ndp[i + j] += dp[i] * w;
                w *= c - j;
            }
        }
        dp.swap(ndp);
        sum += c;
        down *= C.fac(c);
        l = r;
    }
    Z ans = 0;
    for (int i = 0; i < n; ++i) ans += dp[i] * C.fac(n - i);

    cout << ans / down << endl;
}
```


# 2. 换根 + 期望

[Shrinking Tree - 题目 - QOJ.ac](https://qoj.ac/contest/4686/problem/24861)

- 枚举每条边，每条边将树分为两个部分A,B
- 考虑将A缩成一个点，则答案加上 (A中的最大值 * p)
- 同理，考虑将B缩成一个点，则答案加上 (B中的最大值 * p)

A，B中的最大值：
- 以任一节点(1)为根，AB中有一个是子树中的最大值，另一个是除去子树中的最大值
- 跑一遍dfs，可以求出以1为根节点，子树的sz和子树中的最大值
- 再跑一遍dfs，考虑处理除去子树中的最大值
- 考虑后序遍历，先处理节点x邻接表中的前后缀最大，只有fa节点分支的最大不知道，设为`sub[x]` ，处理完后，根据前后缀最大，可以处理出x每个邻接节点y的`sub[y]` 
- 然后递归`(y, x)` 

```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
const int mod = 998244353;

const int MAXN = 4e5 + 50;

int ksm(int n, int m) {
    int res = 1;
    n %= mod;
    while (m) {
        if (m & 1) res = res * n % mod;
        n = n * n % mod;
        m >>= 1;
    }
    return res;
}

int fac[MAXN], inv[MAXN];

int C(int n, int m) {
    if (m < 0 || m > n || n < 0) return 0;
    return fac[n] * inv[m] % mod * inv[n - m] % mod;
}

void precompute() {
    fac[0] = inv[0] = 1;
    for (int i = 1; i < MAXN; i++) {
        fac[i] = fac[i - 1] * i % mod;
    }
    inv[MAXN - 1] = ksm(fac[MAXN - 1], mod - 2);
    for (int i = MAXN - 2; i >= 1; i--) {
        inv[i] = inv[i + 1] * (i + 1) % mod;
    }
}

void ap() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<vector<int>> g(n + 1);
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    // 树形 DP 求子树大小、子树内最大权值 max_sub 以及子树外最大权值 max_out
    vector<int> sz(n + 1, 0);
    vector<int> max_sub(n + 1, 0);
    vector<int> max_out(n + 1, 0);

    auto dfs1 = [&](auto& self, int u, int p) -> void {
        sz[u] = 1;
        max_sub[u] = a[u];
        for (int v : g[u]) {
            if (v == p) continue;
            self(self, v, u);
            sz[u] += sz[v];
            max_sub[u] = max(max_sub[u], max_sub[v]);
        }
    };

    auto dfs2 = [&](auto& self, int u, int p) -> void {
        int deg = g[u].size();
        vector<int> pref(deg + 2, 0), suff(deg + 2, 0);

        for (int i = 0; i < deg; i++) {
            int v = g[u][i];
            int val = (v == p) ? max_out[u] : max_sub[v];
            pref[i + 1] = max(pref[i], val);
        }
        for (int i = deg - 1; i >= 0; i--) {
            int v = g[u][i];
            int val = (v == p) ? max_out[u] : max_sub[v];
            suff[i + 1] = max(suff[i + 2], val);
        }

        for (int i = 0; i < deg; i++) {
            int v = g[u][i];
            if (v == p) continue;
            max_out[v] = max({a[u], pref[i], suff[i + 2]});
            self(self, v, u);
        }
    };

    dfs1(dfs1, 1, 0);
    dfs2(dfs2, 1, 0);

    int total_ways = C(n - 1, k);
    int ans = 0;

    // 枚举每一条非根边 (u, parent)
    for (int u = 2; u <= n; u++) {
        // 1. 子树 u 缩为一个叶子结点
        int s = sz[u];
        int need_u = s - 1;
        if (k >= need_u && (n - 1 - s) >= (k - need_u)) {
            int ways = C(n - 1 - s, k - need_u);
            ans = (ans + ways % mod * (max_sub[u] % mod)) % mod;
        }

        // 2. 子树 u 外部整体缩为一个叶子结点
        int s_out = n - s;
        int need_out = s_out - 1;
        if (k >= need_out && (s - 1) >= (k - need_out)) {
            int ways = C(s - 1, k - need_out);
            ans = (ans + ways % mod * (max_out[u] % mod)) % mod;
        }
    }

    ans = ans * ksm(total_ways, mod - 2) % mod;
    cout << ans << endl;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    precompute();
    int T = 1;
    cin >> T;
    while (T--) {
        ap();
    }
    return 0;
}
```


# 3. DAG拓扑反向建图 + 期望DP

[P4316 绿豆蛙的归宿 - 洛谷](https://www.luogu.com.cn/problem/P4316#ide)

全期望公式（Law of Total Expectation）的核心思想是：**“按第一步发生的情况进行分类讨论（条件期望），再按各情况发生的概率做加权求和”**。

在概率论中，它的标准形式为：

$$E[X] = \sum_{i} P(A_i) \cdot E[X \mid A_i]$$

其中 $\{A_i\}$ 是样本空间的一个完备事件组（即互斥且覆盖了所有可能的情况）。



**对应到本题的具体含义**

设随机变量 $X$ 为：**“从节点 $x$ 出发走到终点 $n$ 所经过的路径总长度”**。我们要求的 $E[x]$ 就是它的期望值 $E[X]$。

当绿豆蛙站在节点 $x$ 时，它面临若干种互斥的走向选择：

- **事件划分 $A_y$**：绿豆蛙选择走边 $(x \to y)$ 前往节点 $y$。
    
- **各事件的发生概率 $P(A_y)$**：根据题意，节点 $x$ 有 $\text{deg}[x]$ 条出边，且等概率随机选择，所以：
    
    $$P(A_y) = \frac{1}{\text{deg}[x]}$$
    


条件期望 $E[X \mid A_y]$ 表示：**在已知第一步选择了走向节点 $y$ 的前提下，从 $x$ 走到终点的总长度期望**。

由期望的线性性质，这一段总长度可以拆分成两部分：

1. **第一步的边权**：已经确定走了边 $(x \to y)$，确定贡献了边长 $w_{x,y}$。
    
2. **后续路径的长度**：此时绿豆蛙已经站在了节点 $y$，之后从 $y$ 走到终点 $n$ 的期望长度即为 $E[y]$。
    

因此：

$$E[X \mid A_y] = w_{x,y} + E[y]$$



将 $P(A_y)$ 和 $E[X \mid A_y]$ 代入全期望公式：

$$E[x] = \sum_{y \in \text{successors}(x)} P(A_y) \cdot E[X \mid A_y]$$

$$E[x] = \sum_{y \in \text{successors}(x)} \frac{1}{\text{deg}[x]} \cdot (w_{x,y} + E[y])$$

提取公因式 $\frac{1}{\text{deg}[x]}$，即得：

$$E[x] = \frac{1}{\text{deg}[x]} \sum_{y \in \text{successors}(x)} (w_{x,y} + E[y])$$

这就是为什么可以通过每个后继节点 $y$ 的期望加上对应边权，再除以出度 $\text{deg}[x]$，直接求出当前节点 $x$ 的期望。


```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;

    vector<vector<array<int, 2>>> g(n + 1);

    vector<int> out(n + 1), deg(n + 1);
    for (int i = 0; i < m; i++) {
        int u, v, w; cin >> u >> v >> w;

        g[v].push_back({u, w});
        out[u]++, deg[u]++;
    }

    queue<int> q;
    q.push(n);

    vector<double> f(n + 1);

    while (q.size()) {
        auto x = q.front();
        q.pop();

        for (auto & [y, w] : g[x]) {
            f[y] += (f[x] + w) / deg[y];
            
            if (--out[y] == 0) q.push(y);
        }
    }

    cout << fixed << setprecision(2) << f[1] << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```


# 4. 推式子 - 等概率 + 期望线性性质

[P3802 小魔女帕琪 - 洛谷](https://www.luogu.com.cn/problem/P3802#ide)



```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n = 7;
    vector<double> a(n);
    for (auto & i : a) cin >> i;

    double s = 0;
    for (auto & x : a) s += x;

    if (s < 7) {
        cout << "0.000" << endl;
        return;
    }

    double ans = 1;
    for (int i = 0; i < 7; i++) {
        ans = ans * a[i] / (s - i);
    }

    ans *= 1 * 2 * 3 * 4 * 5 * 6 * 7 * (s - 6);

    cout << fixed << setprecision(3) << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
