# 1. 凸包的定义

[Would You Make a Convex? - 题目 - QOJ.ac](https://qoj.ac/contest/3766/problem/17351/statement/zh_cn)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    sort(a.begin(), a.end());

    int st = 0, cnt = 0;
    int right = 2;
    for (int i = 0; i <= n - 3; i++) {
        int sum = a[i] + a[i + 1];
        while (right < n && sum > a[right]) right++;

        if (right - 1 == i + 1) continue;

        if (right - i > cnt) {
            cnt = right - i;
            st = i;
        }

        if (right == n) break;
    }

    if (cnt < 3) {
        cout << 0 << endl;
    } else {
        cout << cnt << ' ';
        for (int i = st; i < st + cnt; i++) {
            cout << a[i] << ' ';
        }
        cout << endl;
    }

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
