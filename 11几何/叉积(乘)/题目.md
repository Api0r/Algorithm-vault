
# 1. 线段严格的内部相交（规范相交）

**严格的内部相交（规范相交）**，即交点不在任何一条线段的端点上。

```c++
int t1 = cha(dif(x, y), dif(x, a));
int t2 = cha(dif(x, y), dif(x, b));

if (__int128(t1) * t2 >= 0) continue;

int t3 = cha(dif(a, b), dif(a, x));
int t4 = cha(dif(a, b), dif(a, y));

if (__int128(t3) * t4 >= 0) continue;
```

[E-小月的门_牛客周赛 Round 163](https://ac.nowcoder.com/acm/contest/140737/E)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

using p = array<int, 2>;

p a, b;
int cha(const p & a, const p & b) {
    return a[0] * b[1] - a[1] * b[0];
}

p dif(const p & a, const p & b) {
    return {a[0] - b[0], a[1] - b[1]};
}

void ap() {
	int n; cin >> n;
    cin >> a[0] >> a[1] >> b[0] >> b[1];
    
    int ans = 0;
    
    for (int i = 0; i < n; i++) {
        p x, y;
        cin >> x[0] >> x[1] >> y[0] >> y[1];
        
        int t1 = cha(dif(x, y), dif(x, a));
        int t2 = cha(dif(x, y), dif(x, b));
        
        if (__int128(t1) * t2 >= 0) continue;
        
        int t3 = cha(dif(a, b), dif(a, x));
        int t4 = cha(dif(a, b), dif(a, y));
        
        if (__int128(t3) * t4 >= 0) continue;
        
        int t = cha(dif(a, b), dif(x, y));
        if (t == 0) {
            continue;
        } else if (t < 0) {
            ans--;
        } else if (t > 0) {
            ans++;
        }

    }
    
    cout << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
