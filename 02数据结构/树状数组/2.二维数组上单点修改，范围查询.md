// 二维数组上单点增加、范围查询，使用树状数组的模版
// 测试链接 : https://leetcode.cn/problems/range-sum-query-2d-mutable/

struct NumMatrix {
    int n, m;
    vector<vector<int>> tree;
    vector<vector<int>> nums;
	
	// 入参二维数组下标从0开始
	// 树状数组一定下标从1开始
    NumMatrix(vector<vector<int>>& matrix) {
        n = matrix.size();
        m = matrix[0].size();
        tree.assign(n + 1, vector<int>(m + 1, 0));
        nums.assign(n + 1, vector<int>(m + 1, 0));
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                update(i, j, matrix[i][j]);
            }
        }
    }

    int lowbit(int i) {
        return i & -i;
    }

    void add(int x, int y, int v) {
        for (int i = x; i <= n; i += lowbit(i)) {
            for (int j = y; j <= m; j += lowbit(j)) {
                tree[i][j] += v;
            }
        }
    }

	// 从(1,1)到(x,y)这个部分的累加和
    int sum(int x, int y) {
        int ans = 0;
        for (int i = x; i > 0; i -= lowbit(i)) {
            for (int j = y; j > 0; j -= lowbit(j)) {
                ans += tree[i][j];
            }
        }
        return ans;
    }
	
	// 实际二维数组的位置是(x,y)
	// 树状数组上的位置是(x+1, y+1)
	// 题目说的是单点更新，转化成单点增加(老值-新值)即可
	// 不要忘了在nums中把老值改成新值
    void update(int row, int col, int val) {
        add(row + 1, col + 1, val - nums[row + 1][col + 1]);
        nums[row + 1][col + 1] = val;
    }
	
	// 实际二维数组的位置是(x,y)
	// 树状数组上的位置是(x+1, y+1)
    int sumRegion(int row1, int col1, int row2, int col2) {
        return sum(row2 + 1, col2 + 1) 
             - sum(row1, col2 + 1) 
             - sum(row2 + 1, col1) 
             + sum(row1, col1);
    }
};

int main() {
    vector<vector<int>> matrix = {
        {3, 0, 1, 4, 2},
        {5, 6, 3, 2, 1},
        {1, 2, 0, 1, 5}
    };

    // 实例化
    NumMatrix solver(matrix);

    // 查询与修改
    int res = solver.sumRegion(1, 1, 2, 2); // 查询区域和
    solver.update(1, 1, 10);                // 单点更新

class NumMatrix {
private:
    int n, m;
    vector<vector<int>> tree;
    vector<vector<int>> nums;

    int lowbit(int i) {
        return i & -i;
    }

    void add(int x, int y, int v) {
        for (int i = x; i <= n; i += lowbit(i)) {
            for (int j = y; j <= m; j += lowbit(j)) {
                tree[i][j] += v;
            }
        }
    }

    // 从 (1, 1) 到 (x, y) 矩形区域的累加和
    int sum(int x, int y) {
        int ans = 0;
        for (int i = x; i > 0; i -= lowbit(i)) {
            for (int j = y; j > 0; j -= lowbit(j)) {
                ans += tree[i][j];
            }
        }
        return ans;
    }

public:
    NumMatrix(vector<vector<int>>& matrix) {
        n = matrix.size();
        m = matrix[0].size();
        tree.assign(n + 1, vector<int>(m + 1, 0));
        nums.assign(n + 1, vector<int>(m + 1, 0));
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                update(i, j, matrix[i][j]);
            }
        }
    }

    // 单点更新：将位置 (row, col) 的值修改为 val
    void update(int row, int col, int val) {
        add(row + 1, col + 1, val - nums[row + 1][col + 1]);
        nums[row + 1][col + 1] = val;
    }

    // 区域查询：返回以 (row1, col1) 为左上角、(row2, col2) 为右下角的子矩形和
    int sumRegion(int row1, int col1, int row2, int col2) {
        return sum(row2 + 1, col2 + 1) 
             - sum(row1, col2 + 1) 
             - sum(row2 + 1, col1) 
             + sum(row1, col1);
    }
};
