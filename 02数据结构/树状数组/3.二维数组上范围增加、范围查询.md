```c++
/*
用途：二维树状数组，支持矩形区域增加与矩形区域查询。
接口（均为 1-indexed）：
  - add(x, y, v)          : 内部单点差分修改
  - add(a, b, c, d, v)    : 区域 [a..c][b..d] 整体增加 v
  - sum(x, y)             : 查询 (1, 1) 到 (x, y) 的前缀和
  - range(a, b, c, d)     : 查询区域 [a..c][b..d] 的区间和
*/
template <class T = int>
struct RangeBIT2D {
    int n = 0, m = 0;

    // 维护信息: d[i][j], d[i][j] * i, d[i][j] * j, d[i][j] * i * j
    std::vector<std::vector<T>> info1, info2, info3, info4;

    RangeBIT2D() = default;
    RangeBIT2D(int n, int m) {
        init(n, m);
    }

    void init(int _n, int _m) {
        n = _n;
        m = _m;
        info1.assign(n + 1, std::vector<T>(m + 1, T{}));
        info2.assign(n + 1, std::vector<T>(m + 1, T{}));
        info3.assign(n + 1, std::vector<T>(m + 1, T{}));
        info4.assign(n + 1, std::vector<T>(m + 1, T{}));
    }

    // 单点差分贡献更新
    void add(int x, int y, const T &v) {
        T v1 = v;
        T v2 = static_cast<T>(x) * v;
        T v3 = static_cast<T>(y) * v;
        T v4 = static_cast<T>(x) * static_cast<T>(y) * v;
        for (int i = x; i <= n; i += i & -i) {
            for (int j = y; j <= m; j += j & -j) {
                info1[i][j] += v1;
                info2[i][j] += v2;
                info3[i][j] += v3;
                info4[i][j] += v4;
            }
        }
    }

    // 以 (1, 1) 为左上角，以 (x, y) 为右下角的二维前缀和
    T sum(int x, int y) const {
        if (x <= 0 || y <= 0) return T{};
        x = std::min(x, n);
        y = std::min(y, m);
        T ans{};
        for (int i = x; i > 0; i -= i & -i) {
            for (int j = y; j > 0; j -= j & -j) {
                ans += static_cast<T>(x + 1) * static_cast<T>(y + 1) * info1[i][j]
                     - static_cast<T>(y + 1) * info2[i][j]
                     - static_cast<T>(x + 1) * info3[i][j]
                     + info4[i][j];
            }
        }
        return ans;
    }

    // 二维区间加：矩形区域 [a..c][b..d] 整体增加 v
    void add(int a, int b, int c, int d, const T &v) {
        if (a > c || b > d) return;
        add(a, b, v);
        add(a, d + 1, -v);
        add(c + 1, b, -v);
        add(c + 1, d + 1, v);
    }

    // 二维区间查询：矩形区域 [a..c][b..d] 的累加和
    T range(int a, int b, int c, int d) const {
        if (a > c || b > d) return T{};
        return sum(c, d) - sum(a - 1, d) - sum(c, b - 1) + sum(a - 1, b - 1);
    }
};
```


以下是二维树状数组支持**区间修改、区间查询**（二维差分树状数组）的完整数学推导过程。

### 一、二维差分的定义与单点还原

设原二维矩阵为 $A(i, j)$（约定下标从 $1$ 开始，越界位置为 $0$）。

定义其二维差分数组 $D(i, j)$ 为：

$$D(i, j) = A(i, j) - A(i - 1, j) - A(i, j - 1) + A(i - 1, j - 1)$$

根据二维前缀和与差分的互逆性质，原矩阵任意位置的单点值 $A(x, y)$ 可以通过差分数组的前缀和还原：

$$A(x, y) = \sum_{i=1}^x \sum_{j=1}^y D(i, j)$$

### 二、二维前缀和的公式推导

我们需要求的是原矩阵从 $(1, 1)$ 到 $(x, y)$ 的二维前缀和 $S(x, y)$：

$$S(x, y) = \sum_{p=1}^x \sum_{q=1}^y A(p, q)$$

将单点还原公式 $A(p, q) = \sum_{i=1}^p \sum_{j=1}^q D(i, j)$ 代入上式：

$$S(x, y) = \sum_{p=1}^x \sum_{q=1}^y \left( \sum_{i=1}^p \sum_{j=1}^q D(i, j) \right)$$

#### 交换求和次序（统计每个 $D(i, j)$ 的贡献频次）

对于固定的差分项 $D(i, j)$，只要满足 $i \le p \le x$ 且 $j \le q \le y$，它就会被累加一次。

- $p$ 的取值范围为 $[i, x]$，共有 $x - i + 1$ 个合法取值；

- $q$ 的取值范围为 $[j, y]$，共有 $y - j + 1$ 个合法取值。

因此，每一项 $D(i, j)$ 总共被累加了 $(x - i + 1)(y - j + 1)$ 次：

$$S(x, y) = \sum_{i=1}^x \sum_{j=1}^y D(i, j) \cdot (x - i + 1)(y - j + 1)$$

### 三、多项式展开与分离变量

展开多项式 $(x - i + 1)(y - j + 1)$：

$$\begin{aligned} (x - i + 1)(y - j + 1) &= [(x + 1) - i] \cdot [(y + 1) - j] \\ &= (x + 1)(y + 1) - (y + 1) \cdot i - (x + 1) \cdot j + i \cdot j \end{aligned}$$

将展开式代入 $S(x, y)$ 的求和式中：

$$S(x, y) = \sum_{i=1}^x \sum_{j=1}^y D(i, j) \Big[ (x + 1)(y + 1) - (y + 1)i - (x + 1)j + ij \Big]$$

将各项拆开，并将与内层求和指标 $i, j$ 无关的常数因子提到求和符号外：

$$\begin{aligned} S(x, y) = &(x + 1)(y + 1) \sum_{i=1}^x \sum_{j=1}^y D(i, j) \\ &- (y + 1) \sum_{i=1}^x \sum_{j=1}^y \big( D(i, j) \cdot i \big) \\ &- (x + 1) \sum_{i=1}^x \sum_{j=1}^y \big( D(i, j) \cdot j \big) \\ &+ \sum_{i=1}^x \sum_{j=1}^y \big( D(i, j) \cdot i \cdot j \big) \end{aligned}$$

这就是代码中 `sum(x, y)` 能够直接计算前缀和的核心数学恒等式。

### 四、树状数组的维护设计

由上述展开式可知，为了在 $O(\log n \cdot \log m)$ 时间内求出 $S(x, y)$，只需同时用树状数组维护四个二维信息：

1. **$\text{info1}[i][j]$** 维护：$D(i, j)$

2. **$\text{info2}[i][j]$** 维护：$D(i, j) \cdot i$

3. **$\text{info3}[i][j]$** 维护：$D(i, j) \cdot j$

4. **$\text{info4}[i][j]$** 维护：$D(i, j) \cdot i \cdot j$

### 五、区间操作与前缀和映射

#### 1. 区间增加 `add(a, b, c, d, v)`

对子矩阵 $[a, c] \times [b, d]$ 整体加上 $v$，根据标准二维差分原理，只需在 4 个关键坐标上做单点修改：

- $(a, b)$ 加 $v$

- $(a, d + 1)$ 加 $-v$

- $(c + 1, b)$ 加 $-v$

- $(c + 1, d + 1)$ 加 $v$

每次执行单点修改 `add(x, y, v)` 时，同步更新四个树状数组：

- $\text{info1} \gets v$

- $\text{info2} \gets v \cdot x$

- $\text{info3} \gets v \cdot y$

- $\text{info4} \gets v \cdot x \cdot y$

#### 2. 区间查询 `range(a, b, c, d)`

查询子矩阵 $[a, c] \times [b, d]$ 的权值和，直接使用二维容斥原理转换为 4 次前缀和查询：

$$\text{Sum}([a, c] \times [b, d]) = S(c, d) - S(a - 1, d) - S(c, b - 1) + S(a - 1, b - 1)$$
