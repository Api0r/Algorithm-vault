```c++
struct Node {
    Node * son[26]{};
    bool end = false;
};

class Solution {
    Node * root = new Node();

public:
    void insert(string s) {
        Node * cur = root;
        for (char c : s) {
            c -= 'a';

            if (cur->son[c] == NULL) {
                cur->son[c] = new Node();
            }

            cur = cur->son[c];
        }
        cur->end = true;
    }

    bool query(string s) {
        Node * cur = root;
        for (char c : s) {
            c -= 'a';

            if (cur->son[c] == NULL) return false;

            cur = cur->son[c];
            if (!cur->end) return false;
        }

        return true;
    }

};
```

```c++
// 静态数组实现的 Trie，内存一次性分配，无指针和动态扩容开销
// MAXN: 最大可能产生的节点总数 (约等于 总字符数 + 5)
// A: 字符集大小 (默认 26 小写字母)
// F: 字符集起始字符 (默认 'a')
template <int MAXN = 150005, int A = 26, char F = 'a'>
struct Trie {
    int tree[MAXN][A];
    int end_cnt[MAXN];
    int pass_cnt[MAXN];
    int cnt;

    Trie() {
        build();
    }

    // 初始化/重置字典树
    void build() {
        cnt = 1;
        memset(tree[1], 0, sizeof(tree[1]));
        end_cnt[1] = 0;
        pass_cnt[1] = 0;
    }

    // 映射字符到对应下标 [0, A-1]
    int id(char c) const {
        return c - F;
    }

    // 插入字符串
    void insert(const string& word) {
        int cur = 1;
        pass_cnt[cur]++;
        for (char c : word) {
            int path = id(c);
            if (tree[cur][path] == 0) {
                tree[cur][path] = ++cnt;
                // 新建节点时清空其状态（避免多组测试用例时被脏数据污染）
                memset(tree[cnt], 0, sizeof(tree[cnt]));
                end_cnt[cnt] = 0;
                pass_cnt[cnt] = 0;
            }
            cur = tree[cur][path];
            pass_cnt[cur]++;
        }
        end_cnt[cur]++;
    }

    // 查询字符串完全匹配出现的次数
    int search(const string& word) const {
        int cur = 1;
        for (char c : word) {
            int path = id(c);
            if (tree[cur][path] == 0) {
                return 0;
            }
            cur = tree[cur][path];
        }
        return end_cnt[cur];
    }

    // 查询以 pre 为前缀的字符串数量
    int prefixNumber(const string& pre) const {
        int cur = 1;
        for (char c : pre) {
            int path = id(c);
            if (tree[cur][path] == 0) {
                return 0;
            }
            cur = tree[cur][path];
        }
        return pass_cnt[cur];
    }

    // 删除字符串（如果不存在则不执行删除）
    void erase(const string& word) {
        if (search(word) > 0) {
            int cur = 1;
            pass_cnt[cur]--;
            for (char c : word) {
                int path = id(c);
                int next_node = tree[cur][path];
                if (--pass_cnt[next_node] == 0) {
                    tree[cur][path] = 0; // 剪枝，直接断开该分支
                    return;
                }
                cur = next_node;
            }
            end_cnt[cur]--;
        }
    }

    // 多组测试数据时的按需清空（只清空使用过的节点，耗时 O(cnt * A)）
    void clear() {
        for (int i = 1; i <= cnt; i++) {
            memset(tree[i], 0, sizeof(tree[i]));
            end_cnt[i] = 0;
            pass_cnt[i] = 0;
        }
        cnt = 1;
    }
};
```


**不能直接在函数内部定义为普通局部变量**（如直接在 `main()` 里写 `Trie<150005> trie;`），这样会直接导致**栈溢出（Segmentation Fault）**。

### 原因分析

- 该模板内部包含固定大小的原生数组：
    
    - `tree[150005][26]` 占用约 $150005 \times 26 \times 4 \approx 15.6\text{ MB}$
        
    - `end_cnt` + `pass_cnt` 占用约 $1.2\text{ MB}$
        
    - 整个结构体大小约为 **16.8 MB**。
        
- 评测环境（Linux/Windows）的默认栈空间通常只有 **8 MB**（部分 OJ 只有 1~2 MB）。
    
- 如果在 `main()` 等函数内直接写：
    
    C++
    
    ```
    int main() {
        Trie<150005> trie; // ❌ 错误：分配在栈上，直接栈溢出崩溃
    }
    ```
    

### 正确的使用方式

#### 方式 1：声明为全局变量（最推荐，算法竞赛常用）

C++

```
#include <iostream>
// ... 引入 Trie 模板 ...

// 声明在全局作用域（静态存储区，空间由系统堆内存上限决定，不会爆栈）
Trie<150005> trie;

int main() {
    // 直接使用 trie
    trie.insert("apple");
    return 0;
}
```

#### 方式 2：在局部加上 `static` 关键字

C++

```
int main() {
    // 加上 static，变量存放在静态区而非栈上
    static Trie<150005> trie;

    trie.insert("apple");
    return 0;
}
```

#### 方式 3：使用 `new` / `std::unique_ptr` 在堆上分配

C++

```
int main() {
    // 动态在堆上申请内存
    auto trie = std::make_unique<Trie<150005>>();

    trie->insert("apple");
    return 0;
}
```
