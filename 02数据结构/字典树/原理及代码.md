struct Node {
    Node * son[26]{};
    bool end = false;
};

class Solution {
    Node * root = new Node();

public:
    void insert(string s) {
        Node * cur = root;
        for (char c : s) {
            c -= 'a';

            if (cur->son[c] == NULL) {
                cur->son[c] = new Node();
            }

            cur = cur->son[c];
        }
        cur->end = true;
    }

    bool query(string s) {
        Node * cur = root;
        for (char c : s) {
            c -= 'a';

            if (cur->son[c] == NULL) return false;

            cur = cur->son[c];
            if (!cur->end) return false;
        }

        return true;
    }

};
