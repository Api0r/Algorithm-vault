
[Longest Common Prefix - 题目 - QOJ.ac](https://qoj.ac/contest/4071/problem/20027)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

const int N = 500005;
int tree[N][26];
int ps[N];
int dep[N];
int cnt = 1;
int f[N];//当前string集合中取1.2.3..个string，最大的LCP

void ap() {
	int n; cin >> n;
    cnt = 1;
    
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        string s; cin >> s;
        
        //后面有减旧操作
        ans += (f[i] ^ i);
        
        int cur = 1;
        for (char & c : s) {
            int path = c - 'a';
            
            if (tree[cur][path] == 0) {
                tree[cur][path] = ++cnt;
                dep[cnt] = dep[cur] + 1;
            }
            cur = tree[cur][path];
            ps[cur]++;
            
            //每添加一个字母，会改变相应节点的ps
            //意味着恰好新产生了 ps[cur] 个字符串拥有长度为 dep[cur] 的公共前缀
            if (dep[cur] > f[ps[cur]]) {
                ans -= (f[ps[cur]] ^ ps[cur]);
                f[ps[cur]] = dep[cur];
                ans += (f[ps[cur]] ^ ps[cur]);
            }
        }
        
        cout << ans << endl;
        
    }
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
