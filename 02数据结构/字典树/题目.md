
# 1.

[Longest Common Prefix - 题目 - QOJ.ac](https://qoj.ac/contest/4071/problem/20027)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

const int N = 500005;
int tree[N][26];
int ps[N];
int dep[N];
int cnt = 1;
int f[N];//f[j]表示当前string集合中任取j个，最大的LCP

void ap() {
	int n; cin >> n;
    cnt = 1;
    
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        string s; cin >> s;
        
        //后面有减旧操作
        //此前ans中没有(f[i] ^ i)
        ans += (f[i] ^ i);
        
        int cur = 1;
        for (char & c : s) {
            int path = c - 'a';
            
            if (tree[cur][path] == 0) {
                tree[cur][path] = ++cnt;
                dep[cnt] = dep[cur] + 1;
            }
            cur = tree[cur][path];
            ps[cur]++;
            
            //每添加一个字母，会改变相应节点的ps
            //意味着恰好新产生了 ps[cur] 个字符串拥有长度为 dep[cur] 的公共前缀
            if (dep[cur] > f[ps[cur]]) {
                ans -= (f[ps[cur]] ^ ps[cur]);
                f[ps[cur]] = dep[cur];
                ans += (f[ps[cur]] ^ ps[cur]);
            }
        }
        
        cout << ans << endl;
        
    }
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```


# 2.记录insert的字符串的下标

[F-小月的前缀_牛客周赛 Round 163](https://ac.nowcoder.com/acm/contest/140737/F)

`Compiler exceeded OUTPUT limit` 报错，根源在于 **C++ 语言标准中关于“平凡类型（POD / Trivially Constructible）”的定义，以及它对“全局大对象初始化”生成的底层汇编代码量的致命影响**。

```c++
#include <bits/stdc++.h>  
using namespace std;  
// #define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

// 静态数组实现的 Trie，内存一次性分配，无指针和动态扩容开销
// MAXN: 最大可能产生的节点总数 (约等于 总字符数 + 5)
// A: 字符集大小 (默认 26 小写字母)
// F: 字符集起始字符 (默认 'a')
template <int MAXN = 500005, int A = 26, char F = 'a'>
struct Trie {
    int tree[MAXN][A];
    int end_cnt[MAXN];
    int pass_cnt[MAXN];
    int cnt;
    
    int str_id[MAXN];

    int id(char c) const {
        return c - F;
    }

    void insert(const string& word, int x, int idx) {
        int cur = 1;
        pass_cnt[cur] += x;
        for (char c : word) {
            int path = id(c);
            if (tree[cur][path] == 0) {
                tree[cur][path] = ++cnt;
            }
            cur = tree[cur][path];
            pass_cnt[cur] += x;
        }
        end_cnt[cur] += x;
        str_id[cur] = idx;
    }
    
    int query(const string & s) {
        int cur = 1;
        int res = 0;
        for (char c : s) {
            int path = id(c);
            if (tree[cur][path] == 0) {
                break;
            }
        
            cur = tree[cur][path];
            if (end_cnt[cur] > 0) {
                res = cur;
            }
        }
        
        if (res > 0) {
            end_cnt[res]--;
            return str_id[res];
        }
        
        return 0;
    }

};

Trie trie;

void ap() {
    trie.cnt = 1;
    
	int n, q; cin >> n >> q;
    for (int i = 1; i <= n; i++) {
        string s; cin >> s;
        int x; cin >> x;
        
        trie.insert(s, x, i);
    }
    
    while (q--) {
        string s; cin >> s;
        
        cout << trie.query(s) << endl;
    }
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
