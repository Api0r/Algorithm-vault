# 1.离散化 + 范围重置、范围查询 - 维护最大值

[699. 掉落的方块 - 力扣（LeetCode）](https://leetcode.cn/problems/falling-squares/description/)

```c++
/*
用途：线段树（区间重置 / 区间赋值、区间最大值查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间重置 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(l, r, v)     : 将区间 [l, r] 内所有元素重置/赋值为 v。
  - query(l, r)         : 查询区间 [l, r] 内的最大值。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 区间重置操作直接将区间最大值设为 v，无需乘以区间长度。
  - query 查询在跨区间合并时使用 std::max，初始极大负值使用 std::numeric_limits<T>::lowest()。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> max_val;
    vector<T> change;
    vector<bool> update_tag;

    SegTree(int size) : n(size),
                        max_val((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总
    void up(int i) {
        max_val[i] = max(max_val[i << 1], max_val[i << 1 | 1]);
    }

    // 懒标记处理（重置覆盖最大值）
    void lazy(int i, T v) {
        max_val[i] = v;
        change[i] = v;
        update_tag[i] = true;
    }

    // 向下传递懒标记
    void down(int i) {
        if (update_tag[i]) {
            lazy(i << 1, change[i]);
            lazy(i << 1 | 1, change[i]);
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i) {
        if (l == r) {
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv);
            return;
        }
        down(i);
        int mid = (l + r) >> 1;
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的最大值
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return max_val[i];
        }
        down(i);
        int mid = (l + r) >> 1;
        T ans = numeric_limits<T>::lowest();
        if (jobl <= mid) {
            ans = max(ans, query(jobl, jobr, l, mid, i << 1));
        }
        if (jobr > mid) {
            ans = max(ans, query(jobl, jobr, mid + 1, r, i << 1 | 1));
        }
        return ans;
    }

    // 便捷对外接口
    void build() { build(1, n, 1); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

//注意：
//max数组默认是最小，需要调整为初始为0
class Solution {
public:
    vector<int> fallingSquares(vector<vector<int>>& positions) {
        vector<int> sorted;
        for (auto & p : positions) {
            sorted.push_back(p[0]);
            sorted.push_back(p[0] + p[1] - 1);
        }
        sort(sorted.begin(), sorted.end());
        sorted.erase(unique(sorted.begin(), sorted.end()), sorted.end());

        int n = sorted.size();

        SegTree<int> tr(n);
        tr.build();

        vector<int> ans;
        int mx = 0;
        for (auto & p : positions) {
            //得到的rank从1开始，所以后面要加1
            int l = lower_bound(sorted.begin(), sorted.end(), p[0]) - sorted.begin() + 1;
            int r = lower_bound(sorted.begin(), sorted.end(), p[0] + p[1] - 1) - sorted.begin() + 1;

            int h = tr.query(l, r) + p[1];
            mx = max(mx, h);
            ans.push_back(mx);

            tr.update(l, r, h);
        }

        return ans;
    }
};
```

# 2.二分 + 范围重置、范围查询 - 维护累加和

https://acm.hdu.edu.cn/showproblem.php?pid=4614

```c++
#include <bits/stdc++.h>
using namespace std;
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

/*
用途：线段树（区间重置 / 区间赋值、区间和查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间重置 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(l, r, v)     : 将区间 [l, r] 内所有元素重置/赋值为 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 区间重置操作会直接覆盖历史所有数值与未下传标记。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> change;
    vector<bool> update_tag;

    SegTree(int size) : n(size),
                        sum((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 懒标记处理（重置覆盖）
    void lazy(int i, T v, int len) {
        sum[i] = v * len;
        change[i] = v;
        update_tag[i] = true;
    }

    // 向下传递懒标记
    void down(int i, int ln, int rn) {
        if (update_tag[i]) {
            lazy(i << 1, change[i], ln);
            lazy(i << 1 | 1, change[i], rn);
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i) {
        if (l == r) {
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 便捷对外接口
    void build() { build(1, n, 1); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

void ap() {
    int n, m; cin >> n >> m;

    SegTree<int> tr(n);
    tr.build();

    // 返回第k个0的位置
    auto find = [&](int s, int k) -> int {
        int l = s - 1, r = n + 1;
        int ans = 0;

        while (l + 1 < r) {
            int mid = (l + r) >> 1;
            if (mid - s + 1 - tr.query(s, mid) >= k) {//cnt0 >= k
                ans = mid;
                r = mid;
            } else {
                l = mid;
            }
        }
        return ans;
    };

    auto insert = [&](int s, int f) -> array<int, 2> {
        int start, end;

        int cnt0 = n - s + 1 - tr.query(s, n);
        if (cnt0 == 0) {
            start = 0;
            end = 0;
        } else {
            start = find(s, 1);
            end = find(s, min(cnt0, f));

            tr.update(start, end, 1);
        }

        start--;
        end--;

        return {start, end};
    };

    while (m--) {
        int k; cin >> k;
        if (k == 1) {
            int s, f; cin >> s >> f;
            s++;

            array<int, 2> ans = insert(s, f);
            if (ans[0] == -1) {
                cout << "Can not put any one." << endl;
            } else {
                cout << ans[0] << ' ' << ans[1] << endl;
            }

        } else {
            int l, r; cin >> l >> r;
            l++, r++;

            cout << tr.query(l, r) << endl;
            tr.update(l, r, 0);
        }
    }
    return ;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T = 1;
    cin >> T;
    while (T--) {
        ap();
    }
    return 0;
}
```

# 3.势能分析 + 开平方操作 + 剪枝

势能分析的本质是为了计算**暴力**的时间复杂度是否可行

```c++
#include <bits/stdc++.h>

  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 内部空间分配为 4 倍节点大小，单点修改可使用 add(p, p, v)。
```

## 势能分析

该代码采用**势能线段树（吉司机线段树思想 / 均摊复杂度分析）**。由于题目保证数组中所有数均满足 $a_i \ge 1$ 且 $a_i \le 10^{12}$，开方操作具有极强的收敛性。

### 1. 势能函数定义

- 对于序列中每个位置 $k$ 的元素 $v_k$（$v_k \ge 1$），定义其势能 $\Phi_k$ 为：将其通过连续开方运算 $\lfloor\sqrt{v}\rfloor$ 降为 $1$ 所需的操作次数。

- 定义整个线段树系统的总势能为所有单点势能之和：

    $$\Phi = \sum_{k=1}^n \Phi_k$$

### 2. 初始势能的上界

对于任意正整数 $x$：

- $x \le 10^{12}$ 开方 1 次后 $\le 10^6$

- 开方 2 次后 $\le 1000$

- 开方 3 次后 $\le 31$

- 开方 4 次后 $\le 5$

- 开方 5 次后 $\le 2$

- 开方 6 次后 $= 1$

因此，对任意 $a_i \le 10^{12}$，单点最大开方次数 $\Phi_k \le 6$（理论最大值为 $\log_2(\log_2 V)$ 级别）。

初始总势能：

$$\Phi_{\text{init}} = \sum_{k=1}^n \Phi_k \le 6n = O(n \log \log V)$$

### 3. 剪枝有效性证明

代码中的剪枝条件为：

```c++
if (jobl <= mid && sum[i << 1] > mid - l + 1)
```

- **前提成立性**：因为数据满足 $a_i \ge 1$，一个长度为 $L$ 的子区间和最小为 $L \times 1 = L$。

- **等价性**：当且仅当子区间内**所有数都为 1** 时，$\text{sum} = L$（即 $\text{sum} > L$ 不成立）。

- 一旦某个子树内的所有元素都退化为 1，该子树内每个元素 $\lfloor\sqrt{1}\rfloor = 1$，值不再发生改变，对应的势能降为 0。剪枝条件会直接跳过该子树，**不再递归访问**。

### 4. 均摊时间复杂度计算

每次区间开方操作 `add(l, r)` 的开销分为两部分：

1. **未触发元素更新的定位开销（未访问到叶子）**：

    在线段树上定位到不包含任何未退化节点（或超出区间）的子树，每次操作最多访问 $O(\log n)$ 个节点。

    $m$ 次操作的该部分总代价为 $O(m \log n)$。

2. **触发元素更新的真实修改（深入到叶子）**：

    每一次成功递归到叶子节点并执行一次 `sqrt`，都会使该叶子节点的势能 $\Phi_k$ 至少减少 1。

    因为初始总势能为 $\Phi \le 6n$，且**区间开方操作永远不会增加势能**（$\Delta \Phi \le 0$），所以整个运行期间深入到叶子节点的总次数上界为 $6n$ 次。

    每次深入叶子耗时 $O(\log n)$，所有修改操作消耗的总代价为：

    $$O(\Phi_{\text{init}} \cdot \log n) = O(n \log n \log \log V)$$

### 总结

- **单次区间查询复杂度**：$O(\log n)$

- **$m$ 次操作的总时间复杂度**：$O((n \log \log V + m) \log n)$

- 在 $n, m \le 10^5, V \le 10^{12}$ 的数据规模下，有效开方总次数不超过 $6n = 6 \times 10^5$，整体运行时间在几十毫秒内，远优于常规时限。
