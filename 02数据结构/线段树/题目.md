# 1.离散化 + 范围重置、范围查询 - 维护最大值

[699. 掉落的方块 - 力扣（LeetCode）](https://leetcode.cn/problems/falling-squares/description/)

```c++
/*
用途：线段树（区间重置 / 区间赋值、区间最大值查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间重置 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(l, r, v)     : 将区间 [l, r] 内所有元素重置/赋值为 v。
  - query(l, r)         : 查询区间 [l, r] 内的最大值。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 区间重置操作直接将区间最大值设为 v，无需乘以区间长度。
  - query 查询在跨区间合并时使用 std::max，初始极大负值使用 std::numeric_limits<T>::lowest()。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> max_val;
    vector<T> change;
    vector<bool> update_tag;

    SegTree(int size) : n(size),
                        max_val((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总
    void up(int i) {
        max_val[i] = max(max_val[i << 1], max_val[i << 1 | 1]);
    }

    // 懒标记处理（重置覆盖最大值）
    void lazy(int i, T v) {
        max_val[i] = v;
        change[i] = v;
        update_tag[i] = true;
    }

    // 向下传递懒标记
    void down(int i) {
        if (update_tag[i]) {
            lazy(i << 1, change[i]);
            lazy(i << 1 | 1, change[i]);
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i) {
        if (l == r) {
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv);
            return;
        }
        down(i);
        int mid = (l + r) >> 1;
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的最大值
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return max_val[i];
        }
        down(i);
        int mid = (l + r) >> 1;
        T ans = numeric_limits<T>::lowest();
        if (jobl <= mid) {
            ans = max(ans, query(jobl, jobr, l, mid, i << 1));
        }
        if (jobr > mid) {
            ans = max(ans, query(jobl, jobr, mid + 1, r, i << 1 | 1));
        }
        return ans;
    }

    // 便捷对外接口
    void build() { build(1, n, 1); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

//注意：
//max数组默认是最小，需要调整为初始为0
class Solution {
public:
    vector<int> fallingSquares(vector<vector<int>>& positions) {
        vector<int> sorted;
        for (auto & p : positions) {
            sorted.push_back(p[0]);
            sorted.push_back(p[0] + p[1] - 1);
        }
        sort(sorted.begin(), sorted.end());
        sorted.erase(unique(sorted.begin(), sorted.end()), sorted.end());

        int n = sorted.size();

        SegTree<int> tr(n);
        tr.build();

        vector<int> ans;
        int mx = 0;
        for (auto & p : positions) {
            //得到的rank从1开始，所以后面要加1
            int l = lower_bound(sorted.begin(), sorted.end(), p[0]) - sorted.begin() + 1;
            int r = lower_bound(sorted.begin(), sorted.end(), p[0] + p[1] - 1) - sorted.begin() + 1;

            int h = tr.query(l, r) + p[1];
            mx = max(mx, h);
            ans.push_back(mx);

            tr.update(l, r, h);
        }

        return ans;
    }
};
```

```c++
#include <bits/stdc++.h>

#define int long long
#define endl '\n'
const int mod = 1e9 + 7;

/*
用途：线段树（区间重置 / 区间赋值、区间和查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间重置 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(l, r, v)     : 将区间 [l, r] 内所有元素重置/赋值为 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 区间重置操作会直接覆盖历史所有数值与未下传标记。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> change;
    vector<bool> update_tag;

    SegTree(int size) : n(size),
                        sum((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 懒标记处理（重置覆盖）
    void lazy(int i, T v, int len) {
        sum[i] = v * len;
        change[i] = v;
        update_tag[i] = true;
    }

    // 向下传递懒标记
    void down(int i, int ln, int rn) {
        if (update_tag[i]) {
            lazy(i << 1, change[i], ln);
            lazy(i << 1 | 1, change[i], rn);
            update_tag[i] = false;
        }
    }

    // 建树
    void build(int l, int r, int i) {
        if (l == r) {
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 便捷对外接口
    void build() { build(1, n, 1); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

void ap() {
    int n, m; cin >> n >> m;

    SegTree<int> tr(n);
    tr.build();

    // 返回第k个0的位置
    auto find = [&](int s, int k) -> int {
        int l = s - 1, r = n + 1;
        int ans = 0;

        while (l + 1 < r) {
            int mid = (l + r) >> 1;
            if (mid - s + 1 - tr.query(s, mid) >= k) {//cnt0 >= k
                ans = mid;
                r = mid;
            } else {
                l = mid;
            }
        }
        return ans;
    };

    auto insert = [&](int s, int f) -> array<int, 2> {
        int start, end;

        int cnt0 = n - s + 1 - tr.query(s, n);
        if (cnt0 == 0) {
            start = 0;
            end = 0;
        } else {
            start = find(s, 1);
            end = find(s, min(cnt0, f));

            tr.update(start, end, 1);
        }

        start--;
        end--;

        return {start, end};
    };

    while (m--) {
        int k; cin >> k;
        if (k == 1) {
            int s, f; cin >> s >> f;
            s++;

            array<int, 2> ans = insert(s, f);
            if (ans[0] == -1) {
                cout << "Can not put any one." << endl;
            } else {
                cout << ans[0] << ' ' << ans[1] << endl;
            }

        } else {
            int l, r; cin >> l >> r;
            l++, r++;

            cout << tr.query(l, r) << endl;
            tr.update(l, r, 0);
        }
    }
    return ;
}

    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T = 1;
    cin >> T;
    while (T--) {
        ap();
    }
```

## 势能分析

该代码采用**势能线段树（吉司机线段树思想 / 均摊复杂度分析）**。由于题目保证数组中所有数均满足 $a_i \ge 1$ 且 $a_i \le 10^{12}$，开方操作具有极强的收敛性。

### 1. 势能函数定义

- 对于序列中每个位置 $k$ 的元素 $v_k$（$v_k \ge 1$），定义其势能 $\Phi_k$ 为：将其通过连续开方运算 $\lfloor\sqrt{v}\rfloor$ 降为 $1$ 所需的操作次数。

- 定义整个线段树系统的总势能为所有单点势能之和：

    $$\Phi = \sum_{k=1}^n \Phi_k$$

### 2. 初始势能的上界

对于任意正整数 $x$：

- $x \le 10^{12}$ 开方 1 次后 $\le 10^6$

- 开方 2 次后 $\le 1000$

- 开方 3 次后 $\le 31$

- 开方 4 次后 $\le 5$

- 开方 5 次后 $\le 2$

- 开方 6 次后 $= 1$

因此，对任意 $a_i \le 10^{12}$，单点最大开方次数 $\Phi_k \le 6$（理论最大值为 $\log_2(\log_2 V)$ 级别）。

初始总势能：

$$\Phi_{\text{init}} = \sum_{k=1}^n \Phi_k \le 6n = O(n \log \log V)$$

### 3. 剪枝有效性证明

代码中的剪枝条件为：

```c++
if (jobl <= mid && sum[i << 1] > mid - l + 1)
```

- **前提成立性**：因为数据满足 $a_i \ge 1$，一个长度为 $L$ 的子区间和最小为 $L \times 1 = L$。

- **等价性**：当且仅当子区间内**所有数都为 1** 时，$\text{sum} = L$（即 $\text{sum} > L$ 不成立）。

- 一旦某个子树内的所有元素都退化为 1，该子树内每个元素 $\lfloor\sqrt{1}\rfloor = 1$，值不再发生改变，对应的势能降为 0。剪枝条件会直接跳过该子树，**不再递归访问**。

### 4. 均摊时间复杂度计算

每次区间开方操作 `add(l, r)` 的开销分为两部分：

1. **未触发元素更新的定位开销（未访问到叶子）**：

    在线段树上定位到不包含任何未退化节点（或超出区间）的子树，每次操作最多访问 $O(\log n)$ 个节点。

    $m$ 次操作的该部分总代价为 $O(m \log n)$。

2. **触发元素更新的真实修改（深入到叶子）**：

    每一次成功递归到叶子节点并执行一次 `sqrt`，都会使该叶子节点的势能 $\Phi_k$ 至少减少 1。

    因为初始总势能为 $\Phi \le 6n$，且**区间开方操作永远不会增加势能**（$\Delta \Phi \le 0$），所以整个运行期间深入到叶子节点的总次数上界为 $6n$ 次。

    每次深入叶子耗时 $O(\log n)$，所有修改操作消耗的总代价为：

    $$O(\Phi_{\text{init}} \cdot \log n) = O(n \log n \log \log V)$$

### 总结

- **单次区间查询复杂度**：$O(\log n)$

- **$m$ 次操作的总时间复杂度**：$O((n \log \log V + m) \log n)$

- 在 $n, m \le 10^5, V \le 10^{12}$ 的数据规模下，有效开方总次数不超过 $6n = 6 \times 10^5$，整体运行时间在几十毫秒内，远优于常规时限。

# 4.势能分析 + 取模操作 + 剪枝

**忽略单点修改**（即只有建树、区间取模和区间求和），总时间复杂度为：

$$O(n \log V \log n + m \log n)$$

```c++
#include <bits/stdc++.h>

#define int long long
#define endl '\n'

/*
用途：线段树（区间加法、区间和查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间加 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。

*/

    int n;
    vector<T> sum;
    vector<T> max;

    SegTree(int size) : n(size), sum((size + 1) << 2, 0), max((size + 1) << 2, 0) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
        max[i] = std::max(max[i << 1], max[i << 1 | 1]);
    }

    // 懒标记处理
    // void lazy(int i, T v, int len) {
    //     sum[i] += v * len;
    //     add_tag[i] += v;
    // }

    // 向下传递懒标记
    // void down(int i, int ln, int rn) {
    //     if (add_tag[i] != 0) {
    //         lazy(i << 1, add_tag[i], ln);
    //         lazy(i << 1 | 1, add_tag[i], rn);
    //         add_tag[i] = 0;
    //     }
    // }

    }

    //[jobl, jobr] mod jobv
    void add(int jobl, int jobr, T jobv, int l, int r, int i) {
	    //剪枝，如果要模的数v比所有数都大，那么对v取模不影响，跳过
        if (jobv > max[i]) {
            return;
        }

        if (l == r) {
            sum[i] %= jobv;
            max[i] %= jobv;
            return;
        }
        int mid = (l + r) >> 1;
        if (jobl <= mid) {
            add(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

	//单点修改
    void update(int jobi, int jobv, int l, int r, int i) {
        if (l == r) {
            sum[i] = max[i] = jobv;
            return;
        }
        int mid = (l + r) >> 1;
        if (jobi <= mid) {
            update(jobi, jobv, l, mid, i << 1);
        }
        if (jobi > mid) {
            update(jobi, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void add(int l, int r, T v) { add(l, r, v, 1, n, 1); }
    void update(int i, T v) {update(i, v, 1, n, 1);}
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
```

## 势能分析

这道题是经典的 **Segment Tree Beats（吉老师线段树/势能线段树）** 原型（如 _CF438D The Child and Sequence_）。通过势能分析法（Potential Method）可以证明其均摊时间复杂度。

### 一、 核心性质：取模的衰减性

设数值 $x$ 对 $p$ 取模（$x \ge p$）：

- 若 $p \le \frac{x}{2}$，则 $x \bmod p < p \le \frac{x}{2}$。

- 若 $p > \frac{x}{2}$，则 $x \bmod p = x - p < \frac{x}{2}$。

因此，**任意正整数 $x$ 对不大于自身的数 $p$ 取模，其值至少减半**：

$$x \bmod p < \frac{x}{2}$$

一个初始大小为 $V$ 的数，最多经历 $O(\log V)$ 次**有效取模**（即 $x \ge p$）就会变成 $0$。

### 二、 势能函数的定义

定义每个叶子节点 $i$（对应原数组元素 $a_i$）的势能为：

$$\Phi_i = \lfloor \log_2(a_i + 1) \rfloor$$

定义整个线段树系统的总势能为所有叶子节点的势能之和：

$$\Phi = \sum_{i=1}^n \Phi_i = \sum_{i=1}^n \lfloor \log_2(a_i + 1) \rfloor$$

- **势能边界**：由于 $a_i \ge 0$，始终有 $\Phi \ge 0$。

- **初始势能**：建树后 $\Phi_0 = O(n \log V)$，其中 $V = \max(a_i)$。

### 三、 各操作均摊复杂度分析

#### 1. 区间取模操作 (`add`)

在区间取模递归过程中，访问到的线段树节点可分为两类：

- **未剪枝的无效遍历 / 递归边界节点**：

    对应线段树正常区间定位时被分解的节点，以及满足 $\max[u] < p$ 触发剪枝直接 `return` 的节点。

    - 这一部分每次操作最多访问 $O(\log n)$ 个节点。

    - 产生的实际代价为 $O(\log n)$，势能不增加。

- **有效取模（递归至叶子并成功修改）**：

    每当递归到一个叶子节点 $i$ 执行 $a_i \leftarrow a_i \bmod p$（前提 $a_i \ge p$）：

    - 由取模性质可知，修改后 $a_i$ 至少减半，叶子节点的势能 $\Phi_i$ **至少减少 1**。

    - 访问该叶子路径消耗的实际时间为 $O(\log n)$，同时总势能 $\Phi$ 减少至少 $1$。

    - 因此，每次有效修改消耗 $1$ 单位势能，可以支付 $O(\log n)$ 的实际操作代价。

#### 2. 单点修改操作 (`update`)

- 实际耗时：线段树单点定位与更新，代价为 $O(\log n)$。

- 势能变化：将 $a_k$ 赋值为 $x$，该位置的势能最多从 $0$ 增加到 $\lfloor \log_2(x + 1) \rfloor = O(\log V)$。

- 因此单次单点修改注入系统的势能增量为 $\Delta \Phi \le O(\log V)$。

#### 3. 区间求和查询 (`query`)

- 纯查询操作，不改变任何元素数值与势能。

- 单次查询时间复杂度严格为 $O(\log n)$。

### 四、 总时间复杂度总结

设操作总数为 $m$，包含 $m_1$ 次查询、$m_2$ 次区间取模、$m_3$ 次单点修改（$m_1 + m_2 + m_3 = m$）：

- **初始总势能**：$\Phi_0 \le n \log V$

- **总注入势能**：$\sum \Delta \Phi_{\text{update}} \le m_3 \log V$

- **总有效取模次数**：由总势能消耗上限决定，不超过 $\Phi_0 + \sum \Delta \Phi \le (n + m_3) \log V$

- **每次有效取模的代价**：$O(\log n)$

- **所有操作的基础定位代价**：$O(m \log n)$

$$T_{\text{total}} = O\Big((n + m_3) \log V \cdot \log n + m \log n\Big) = O\Big((n + m) \log n \log V\Big)$$

- **单次操作均摊复杂度**：$O(\log n \log V)$

- **空间复杂度**：$O(n)$

# 总结：什么是势能？如何估算时间复杂度？

先对结论中的三个量进行独立推导与界定：

1. **势能（Potential）本身是什么**：

    - 势能是一个**无量纲的状态计数**（即系统剩余可衰减的步数总量）。

    - 数组中共有 $n$ 个数，每个数 $a_i \le V$。因为 $x \bmod p < x/2$，每个数在不被重新单点赋值的情况下，最多只能经历 $\log_2 V$ 次有效取模。

    - 因此，整个数组的**总势能**为：

        $$\Phi = \sum_{i=1}^n \log_2(a_i) = O(n \log V)$$

    - 如果数值范围 $V \approx n$（例如元素大小在序列长度同阶以内），则初始势能为 $O(n \log n)$。

2. **$n \log n \log n$ 是怎么来的**：

    - 每次有效取模递归到叶子节点，在线段树上走一条树高为 $O(\log n)$ 的路径。

    - 因此，把总势能消耗完所对应的**总时间复杂度**为：

        $$\text{总时间} = \text{总势能} \times \text{单次衰减代价} = O(n \log V) \times O(\log n) = O(n \log n \log V)$$

    - 当数据范围满足 $V \le n$ 或以 $n$ 为基准衡量时，总操作时间上限即为 **$O(n \log^2 n)$（即 $n \log n \log n$）**。

3. **结论辨析**：

    - **系统总势能 $\Phi$**：是 $O(n \log V)$（当 $V \approx n$ 时为 $O(n \log n)$）。

    - **消耗全部势能的总时间复杂度**：是 $O(n \log n \log V)$（当 $V \approx n$ 时为 $O(n \log^2 n)$）。

$n \log n \log n$ 是**消耗该势能所付出的总时间复杂度**，而不是势能函数的值本身。

```c++
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间加 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。

template <typename T = int>
struct SegTree {
    int n;
    vector<T> sumAB;
    vector<T> sumB;
    vector<T> add_tag;

    SegTree(int size) : n(size), sumAB((size + 1) << 2, 0), add_tag((size + 1) << 2, 0),
     sumB((size + 1) << 2, 0){}

    // 向上汇总
    void up(int i) {
        sumAB[i] = sumAB[i << 1] + sumAB[i << 1 | 1];
        sumB[i] = sumB[i << 1] + sumB[i << 1 | 1];
    }

    // 懒标记处理
    void lazy(int i, T v, int len) {
        sumAB[i] += sumB[i] * v;//修改这里的懒信息
        add_tag[i] += v;
    }

    // 向下传递懒标记
    void down(int i, int ln, int rn) {
        if (add_tag[i] != 0) {
            lazy(i << 1, add_tag[i], ln);
            lazy(i << 1 | 1, add_tag[i], rn);
            add_tag[i] = 0;
        }
    }

    // 建树

        if (l == r) {
            sumAB[i] = a[l] * b[l];
            sumB[i] = b[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1);
        build(mid + 1, r, i << 1 | 1);
        up(i);
        add_tag[i] = 0;
    }

    // 区间加：[jobl, jobr] 增加 jobv
    void add(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;

        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            add(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sumAB[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {

        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 便捷对外接口
    void build() { build(1, n, 1); }
    void add(int l, int r, T v) { add(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
```

这类题型在算法竞赛中统称为 **“区间加权和维护”** 或 **“带系数的区间信息维护”**，本质是 **线段树/树状数组的高级区间修改与代数展开应用**。

### 题型核心特征

- **问题本质**：维护两个序列的相关运算（如 $a_i \cdot b_i$、$a_i \cdot i$、$a_i \cdot i^2$ 等），其中一个序列频繁进行区间修改，另一个序列固定或满足某种数学规律。

- **解题核心思路（代数展开与线性拆分）**：

    将修改操作对查询目标的影响在代数上拆解，寻找**不变项**与**变化量**的线性关系：

    $$\sum (a_i + v) \cdot b_i = \sum a_i \cdot b_i + v \cdot \sum b_i$$

    从而将“复杂的区间乘积修改”转化为**线段树多维护一个静态信息（$\sum b_i$）**，实现 $O(1)$ 下传懒标记。

### 常见的同类经典变体

- **维护 $\sum (a_i \cdot i)$**：支持区间加 $a_i \leftarrow a_i + v$。

    - 展开式：$\sum (a_i + v) \cdot i = \sum a_i \cdot i + v \cdot \sum i$

    - 做法：$\sum i$ 可以直接用等差数列求和公式 $O(1)$ 算出，无需额外数组。

- **维护区间方差 / 平方和 $\sum a_i^2$**：支持区间加 $a_i \leftarrow a_i + v$。

    - 展开式：$\sum (a_i + v)^2 = \sum a_i^2 + 2v \cdot \sum a_i + v^2 \cdot \text{len}$

    - 做法：线段树同时维护“区间平方和”与“区间一次和”。

- **双数组同时区间加**：支持区间加 $a$，也支持区间加 $b$，查询 $\sum a_i \cdot b_i$。

    - 展开式：$\sum (a_i + v_a)(b_i + v_b) = \sum a_i b_i + v_a \sum b_i + v_b \sum a_i + v_a v_b \cdot \text{len}$

    - 做法：线段树需要同时维护 $\sum a_i b_i$、$\sum a_i$、$\sum b_i$ 以及各自的 lazy tag。

### 常见替代解法

如果**没有区间修改 $b_i$**（即 $b_i$ 静态），此类问题除了线段树外，还可以使用 **两棵树状数组（差分前缀和展开）** 在更小的常数和代码量下解决。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

template <typename T = long long>
struct SegTree {
    int n;
    vector<bool> broken; // true 表示已毁坏
    vector<T> cost;      // cost[i] 表示节点 i 被完全包含时贡献的有效访问数

    SegTree(int size) : n(size), broken((size + 1) << 2, false), cost((size + 1) << 2, 1) {}

    // 向上更新代价
    void up(int i) {
        if (broken[i]) {
            cost[i] = cost[i << 1] + cost[i << 1 | 1];
        } else {
            cost[i] = 1;
        }
    }

    // 操作 1：定位到区间 [jobl, jobr] 对应的单一节点并标记毁坏
    void add(int jobl, int jobr, int l, int r, int i) {
        if (jobl == l && r == jobr) {
            broken[i] = true;
            if (l != r) {
                cost[i] = cost[i << 1] + cost[i << 1 | 1];
            }
            return;
        }
        int mid = (l + r) >> 1;
        //注意这里不是板子，是看[jobl,jobr] 在左/右子树
        if (jobr <= mid) {
            add(jobl, jobr, l, mid, i << 1);
        } else if (jobl > mid) {
            add(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        up(i); // 回溯时更新父节点的 cost
    }

    // 操作 2：查询区间 [jobl, jobr] 的代价
    T query(int jobl, int jobr, int l, int r, int i) {
        // 若当前区间被查询区间完全包含，直接 O(1) 返回维护好的子树贡献
        if (jobl <= l && r <= jobr) {
            return cost[i];
        }

        T ans = 0;
        // 未完全包含时，若当前节点未损坏，产生 1 次有效访问
        if (!broken[i]) {
            ans++;
        }

        int mid = (l + r) >> 1;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    void add(int l, int r) { add(l, r, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};

void ap() {
	int n; cin >> n;
    
    SegTree<int> tr(n);
    
    while (n--) {
        int op, l, r; cin >> op >> l >> r;
        if (op == 1) {
            
            tr.add(l, r);
        } else {
            
            cout << tr.query(l, r) << endl;
        }
    }
	return ;
}

	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
