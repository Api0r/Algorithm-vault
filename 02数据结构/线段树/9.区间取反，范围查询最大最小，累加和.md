/*
用途：线段树（单点赋值修改、区间取相反数翻转、区间求和、区间最大值、区间最小值）。
特性：带懒标记（支持 down 下传与惰性翻转），支持 1-indexed 数据与查询。
复杂度：建树 O(n)，单点修改 O(log n)，区间取反 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - SegTree(n)       : 构造函数，分配大小为 (n + 1) * 4 的数组空间并初始化。
  - build(arr)       : 传入 1-indexed 原始数组以 O(n) 完成建树。
  - update(pos, val) : 将下标 pos 处的值修改为 val（单点覆盖）。
  - negate(l, r)     : 将区间 [l, r] 内的所有元素取相反数。
  - query_sum(l, r)  : 查询区间 [l, r] 的累加和。
  - query_max(l, r)  : 查询区间 [l, r] 的最大值。
  - query_min(l, r)  : 查询区间 [l, r] 的最小值。
  - query(l, r)      : 一次性查询区间 [l, r] 的 {sum, mx, mn}，返回 Node。

注意事项：
  - 传入的原始数组 arr 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 内部已内置边界保护（jobl > jobr 时安全返回空状态），无需外部特判区间合法性。
*/
template <typename T = long long>
struct SegTree {
    struct Node {
        T sum;
        T mx;
        T mn;
    };

    int n;
    vector<T> sum;
    vector<T> mx;
    vector<T> mn;
    vector<int> rev;

    SegTree(int n) : n(n) {
        sum.resize((n + 1) << 2, 0);
        mx.resize((n + 1) << 2, numeric_limits<T>::lowest() / 2);
        mn.resize((n + 1) << 2, numeric_limits<T>::max() / 2);
        rev.resize((n + 1) << 2, 0);
    }

    // 向上合并子节点信息
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
        mx[i] = max(mx[i << 1], mx[i << 1 | 1]);
        mn[i] = min(mn[i << 1], mn[i << 1 | 1]);
    }

    // 单节点打取反标记并维护极值与和
    void apply_rev(int i) {
        sum[i] = -sum[i];
        T cur_mx = mx[i], cur_mn = mn[i];
        mx[i] = -cur_mn;
        mn[i] = -cur_mx;
        rev[i] ^= 1;
    }

    // 下传取反懒标记
    void down(int i) {
        if (rev[i]) {
            apply_rev(i << 1);
            apply_rev(i << 1 | 1);
            rev[i] = 0;
        }
    }

    // 递归建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            mx[i] = arr[l];
            mn[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
    }

    // 单点覆盖赋值：arr[pos] = val
    void update(int pos, T val, int l, int r, int i) {
        if (l == r) {
            sum[i] = val;
            mx[i] = val;
            mn[i] = val;
            return;
        }
        down(i);
        int mid = (l + r) >> 1;
        if (pos <= mid) {
            update(pos, val, l, mid, i << 1);
        } else {
            update(pos, val, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间取相反数：对 [jobl, jobr] 内元素取负
    void negate(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            apply_rev(i);
            return;
        }
        down(i);
        int mid = (l + r) >> 1;
        if (jobl <= mid) negate(jobl, jobr, l, mid, i << 1);
        if (jobr > mid) negate(jobl, jobr, mid + 1, r, i << 1 | 1);
        up(i);
    }

    // 区间综合查询：递归合并 sum, mx, mn
    Node query(int jobl, int jobr, int l, int r, int i) {
        if (jobl > jobr) {
            return {0, numeric_limits<T>::lowest() / 2, numeric_limits<T>::max() / 2};
        }
        if (jobl <= l && r <= jobr) {
            return {sum[i], mx[i], mn[i]};
        }
        down(i);
        int mid = (l + r) >> 1;
        if (jobr <= mid) {
            return query(jobl, jobr, l, mid, i << 1);
        }
        if (jobl > mid) {
            return query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        Node left_res = query(jobl, jobr, l, mid, i << 1);
        Node right_res = query(jobl, jobr, mid + 1, r, i << 1 | 1);
        return {
            left_res.sum + right_res.sum,
            max(left_res.mx, right_res.mx),
            min(left_res.mn, right_res.mn)
        };
    }

    // 便捷对外调用接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void update(int pos, T val) { update(pos, val, 1, n, 1); }
    void negate(int l, int r) { if (l <= r) negate(l, r, 1, n, 1); }

    Node query(int l, int r) {
        if (l > r) return {0, numeric_limits<T>::lowest() / 2, numeric_limits<T>::max() / 2};
        return query(l, r, 1, n, 1);
    }
    T query_sum(int l, int r) { return query(l, r).sum; }
    T query_max(int l, int r) { return query(l, r).mx; }
    T query_min(int l, int r) { return query(l, r).mn; }
};
