/*
用途：线段树（单点修改、区间和查询、区间最大值查询、区间最小值查询）。
特性：无懒标记（无需 down 下传），常数极小，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，单点修改 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(pos, val)    : 将下标 pos 处的值修改为 val（单点覆盖）。
  - add(pos, val)       : 将下标 pos 处的值增加 val（单点累加，按需选用）。
  - query_sum(l, r)     : 查询区间 [l, r] 的累加和。
  - query_max(l, r)     : 查询区间 [l, r] 的最大值。
  - query_min(l, r)     : 查询区间 [l, r] 的最小值。
  - query(l, r)         : 一次性查询区间 [l, r] 的 (sum, max, min)，返回 Node。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    struct Node {
        T sum;
        T mx;
        T mn;
    };

    int n;
    std::vector<T> sum;
    std::vector<T> mx;
    std::vector<T> mn;

    explicit SegTree(int size) : n(size),
                                 sum((size + 1) << 2, 0),
                                 mx((size + 1) << 2, std::numeric_limits<T>::lowest()),
                                 mn((size + 1) << 2, std::numeric_limits<T>::max()) {}

    // 向上汇总信息
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
        mx[i] = std::max(mx[i << 1], mx[i << 1 | 1]);
        mn[i] = std::min(mn[i << 1], mn[i << 1 | 1]);
    }

    // 建树
    void build(int l, int r, int i, const std::vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            mx[i] = arr[l];
            mn[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
    }

    // 单点重置/赋值：arr[pos] = val
    void update(int pos, T val, int l, int r, int i) {
        if (l == r) {
            sum[i] = val;
            mx[i] = val;
            mn[i] = val;
            return;
        }
        int mid = (l + r) >> 1;
        if (pos <= mid) {
            update(pos, val, l, mid, i << 1);
        } else {
            update(pos, val, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 单点增加：arr[pos] += val
    void add(int pos, T val, int l, int r, int i) {
        if (l == r) {
            sum[i] += val;
            mx[i] += val;
            mn[i] += val;
            return;
        }
        int mid = (l + r) >> 1;
        if (pos <= mid) {
            add(pos, val, l, mid, i << 1);
        } else {
            add(pos, val, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 统一区间查询：一次递归合并 sum, max, min
    Node query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return {sum[i], mx[i], mn[i]};
        }
        int mid = (l + r) >> 1;
        if (jobr <= mid) {
            return query(jobl, jobr, l, mid, i << 1);
        }
        if (jobl > mid) {
            return query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        Node left_res = query(jobl, jobr, l, mid, i << 1);
        Node right_res = query(jobl, jobr, mid + 1, r, i << 1 | 1);
        return {
            left_res.sum + right_res.sum,
            std::max(left_res.mx, right_res.mx),
            std::min(left_res.mn, right_res.mn)
        };
    }

    // 便捷对外接口
    void build(const std::vector<T>& arr) { build(1, n, 1, arr); }
    void update(int pos, T val) { update(pos, val, 1, n, 1); }
    void add(int pos, T val) { add(pos, val, 1, n, 1); }
    
    Node query(int l, int r) { return query(l, r, 1, n, 1); }
    T query_sum(int l, int r) { return query(l, r, 1, n, 1).sum; }
    T query_max(int l, int r) { return query(l, r, 1, n, 1).mx; }
    T query_min(int l, int r) { return query(l, r, 1, n, 1).mn; }
};
