# 1.范围增加、范围查询 - 维护累加和

```c++
// 线段树模板：区间加法、区间和查询（1-indexed）
// 线段树支持范围增加、范围查询
// 维护累加和
// 测试链接 : https://www.luogu.com.cn/problem/P3372

/*
用途：线段树（区间加法、区间和查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间加 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 内部空间分配为 4 倍节点大小，单点修改可使用 add(p, p, v)。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> add_tag;

    SegTree(int size) : n(size), sum((size + 1) << 2, 0), add_tag((size + 1) << 2, 0) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 懒标记处理
    void lazy(int i, T v, int len) {
        sum[i] += v * len;
        add_tag[i] += v;
    }

    // 向下传递懒标记
    void down(int i, int ln, int rn) {
        if (add_tag[i] != 0) {
            lazy(i << 1, add_tag[i], ln);
            lazy(i << 1 | 1, add_tag[i], rn);
            add_tag[i] = 0;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
        add_tag[i] = 0;
    }

    // 区间加：[jobl, jobr] 增加 jobv
    void add(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            add(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void add(int l, int r, T v) { add(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};
```

# 3.范围增加、范围查询 - 维护最大值

```c++
/*
用途：线段树（区间加法、区间最大值查询）。
特性：基于懒标记（Lazy Tag）实现，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间加 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 内的最大值。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 维护最大值时，区间加标记只需直接累加到子节点的最大值上（无需乘以区间长度 len）。
  - query 查询在跨区间合并时使用 std::max，初始极大负值使用 std::numeric_limits<T>::lowest()。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> max_val;
    vector<T> add_tag;

    SegTree(int size) : n(size),
                        max_val((size + 1) << 2, numeric_limits<T>::lowest()),
                        add_tag((size + 1) << 2, 0) {}

    // 向上汇总
    void up(int i) {
        max_val[i] = max(max_val[i << 1], max_val[i << 1 | 1]);
    }

    // 懒标记处理（最大值仅加上增量 v）
    void lazy(int i, T v) {
        max_val[i] += v;
        add_tag[i] += v;
    }

    // 向下传递懒标记
    void down(int i) {
        if (add_tag[i] != 0) {
            lazy(i << 1, add_tag[i]);
            lazy(i << 1 | 1, add_tag[i]);
            add_tag[i] = 0;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            max_val[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
        add_tag[i] = 0;
    }

    // 区间加：[jobl, jobr] 增加 jobv
    void add(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            lazy(i, jobv);
            return;
        }
        down(i);
        int mid = (l + r) >> 1;
        if (jobl <= mid) {
            add(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的最大值
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return max_val[i];
        }
        down(i);
        int mid = (l + r) >> 1;
        T ans = numeric_limits<T>::lowest();
        if (jobl <= mid) {
            ans = max(ans, query(jobl, jobr, l, mid, i << 1));
        }
        if (jobr > mid) {
            ans = max(ans, query(jobl, jobr, mid + 1, r, i << 1 | 1));
        }
        return ans;
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void add(int l, int r, T v) { add(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};
```

# 5.同时支持范围重置、范围增加、范围查询 - 维护累加和

```c++
/*
用途：线段树（同时支持区间重置/赋值、区间加法、区间和查询）。
特性：双懒标记（Lazy Tag）协同维护，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，区间重置 O(log n)，区间加 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(l, r, v)     : 将区间 [l, r] 内所有元素重置/赋值为 v。
  - add(l, r, v)        : 将区间 [l, r] 内所有元素加上增量 v。
  - query(l, r)         : 查询区间 [l, r] 的累加和。

注意事项：
  - 标记下放优先级：重置标记（update）高于加法标记（add）。下放时先处理重置，再处理加法。
  - 重置操作会直接清空当前节点的加法懒标记（add_tag = 0）。
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 累加和大范围数据建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    int n;
    vector<T> sum;
    vector<T> add_tag;
    vector<T> change;
    vector<bool> update_tag;

    SegTree(int size) : n(size),
                        sum((size + 1) << 2, 0),
                        add_tag((size + 1) << 2, 0),
                        change((size + 1) << 2, 0),
                        update_tag((size + 1) << 2, false) {}

    // 向上汇总
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
    }

    // 重置懒标记处理（清空加法标记）
    void update_lazy(int i, T v, int len) {
        sum[i] = v * len;
        add_tag[i] = 0;
        change[i] = v;
        update_tag[i] = true;
    }

    // 加法懒标记处理
    void add_lazy(int i, T v, int len) {
        sum[i] += v * len;
        add_tag[i] += v;
    }

    // 向下传递懒标记（先下传重置，再下传加法）
    void down(int i, int ln, int rn) {
        if (update_tag[i]) {
            update_lazy(i << 1, change[i], ln);
            update_lazy(i << 1 | 1, change[i], rn);
            update_tag[i] = false;
        }
        if (add_tag[i] != 0) {
            add_lazy(i << 1, add_tag[i], ln);
            add_lazy(i << 1 | 1, add_tag[i], rn);
            add_tag[i] = 0;
        }
    }

    // 建树
    void build(int l, int r, int i, const vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
        add_tag[i] = 0;
        change[i] = 0;
        update_tag[i] = false;
    }

    // 区间重置：[jobl, jobr] 重置为 jobv
    void update(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            update_lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            update(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间加：[jobl, jobr] 增加 jobv
    void add(int jobl, int jobr, T jobv, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            add_lazy(i, jobv, r - l + 1);
            return;
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            add(jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            add(jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 区间查询：[jobl, jobr] 的区间和
    T query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return sum[i];
        }
        int mid = (l + r) >> 1;
        down(i, mid - l + 1, r - mid);
        T ans = 0;
        if (jobl <= mid) {
            ans += query(jobl, jobr, l, mid, i << 1);
        }
        if (jobr > mid) {
            ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        return ans;
    }

    // 便捷对外接口
    void build(const vector<T>& arr) { build(1, n, 1, arr); }
    void update(int l, int r, T v) { update(l, r, v, 1, n, 1); }
    void add(int l, int r, T v) { add(l, r, v, 1, n, 1); }
    T query(int l, int r) { return query(l, r, 1, n, 1); }
};
```
