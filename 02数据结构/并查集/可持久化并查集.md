```c++
#include<bits/stdc++.h>
using namespace std;
//https://www.luogu.com.cn/problem/P3402#ide

const int N = 200001;
const int T = 8000001;

//rootfa[i] = j, 表示fa数组，i版本的头节点编号为j
//rootsize同上
int rootfa[N], rootsize[N];

//可持久化fa数组和size数组，共用一个ls，rs，val
//因为可持久化时，分配的节点编号不同，所以可以共用
//结构一样，只需要区分了数头节点的编号，就可以区分fa和size数组
int ls[T], rs[T], val[T];
int cnt = 0;

//因为后面的find要1 ~ n 的范围，所以要把n写在外面
int n, m;

//建立可持久化的fa数组，返回的是节点的编号
int buildfa(int l, int r) {
    int rt = ++cnt;
    if (l == r) {
        val[rt] = r;
    } else {
        int mid = (l + r) >> 1;
        ls[rt] = buildfa(l, mid);
        rs[rt] = buildfa(mid + 1, r);
    }
    return rt;
}

int buildsize(int l, int r) {
    int rt = ++cnt;
    if (l == r) {
        val[rt] = 1;
    } else {
        int mid = (l + r) >> 1;
        ls[rt] = buildsize(l, mid);
        rs[rt] = buildsize(mid + 1, r);
    }
    return rt;
}

//线段树范围l ~ r, 信息在i号节点里
//在l ~ r范围上，将x位置的值 更新为 value
//返回生成的新节点的编号
//最终返回的是头节点的编号
int update(int i, int l, int r, int x, int value) {
    //进到了这个函数，就表明l ~ r 范围内有需要修改的位置，那么当前的位置就需要用一个新的节点存储
    //开辟一个全新的节点编号
    //先拷贝之前的所有，再往下去看需要修改的地方
    int rt = ++cnt;
    ls[rt] = ls[i];
    rs[rt] = rs[i];

    if (l == r) {
        val[rt] = value;
    } else {
        int mid = (l + r) >> 1;
        if (x <= mid) {
            ls[rt] = update(ls[rt], l, mid, x, value);
        } else {
            rs[rt] = update(rs[rt], mid + 1, r, x, value);
        }
    }
    return rt;
}

//查询数组中一个位置的值
//线段树范围l ~ r, 信息在i号节点里
//返回l ~ r范围上 x 位置的值
int query(int i, int l, int r, int x) {
    if (l == r) {
        return val[i];
    }

    int mid = (l + r) >> 1;
    if (x <= mid) {
        return query(ls[i], l, mid, x);
    } else {
        return query(rs[i], mid + 1, r, x);
    }
}

//基于 v版本，查询x的集合的头节点，不做扁平化
int find(int x, int v) {
    int fa = query(rootfa[v], 1, n, x);
    while (x != fa) {
        x = fa;
        fa = query(rootfa[v], 1, n, x);
    }
    return x;
}

//v版本先拷贝v - 1版本，再调用merge，合并x所在的集合和y所在的集合，去更新v版本
bool merge(int x, int y, int v) {
    int fx = find(x, v);
    int fy = find(y, v);
    if (fx == fy) return false;
    
    int xsize = query(rootsize[v], 1, n, fx);
    int ysize = query(rootsize[v], 1, n, fy);
    if (xsize >= ysize) {
        rootfa[v] = update(rootfa[v], 1, n, fy, fx);
        rootsize[v] = update(rootsize[v], 1, n, fx, xsize + ysize);
    } else {
        rootfa[v] = update(rootfa[v], 1, n, fx, fy);
        rootsize[v] = update(rootsize[v], 1, n, fy, xsize + ysize);
    }
    
    return true;
}

signed main () {
    ios::sync_with_stdio(0), cin.tie(0);

    cin >> n >> m;
    
    rootfa[0] = buildfa(1, n);
    rootsize[0] = buildsize(1, n);

    for (int v = 1; v <= m; v++) {
        int opt; cin >> opt;
        //不论是什么操作，都先拷贝v - 1 的版本
        rootfa[v] = rootfa[v - 1];
        rootsize[v] = rootsize[v - 1];
        
        if (opt == 1) {
            int x, y; cin >> x >> y;
            merge(x, y, v);
        } else if (opt == 2) {
            int x; cin >> x;
            rootfa[v] = rootfa[x];
            rootsize[v] = rootsize[x];
        } else {
            int x, y; cin >> x >> y;
            if (find(x, v) == find(y, v)) {
                cout << 1 << endl;
            } else {
                cout << 0 << endl;
            }
        }
        
    }


    return 0;
}
```
