[P8779 [蓝桥杯 2022 省 A] 推导部分和 - 洛谷](https://www.luogu.com.cn/problem/P8779)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

vector<int> fa;
vector<int> dis;

int find(int i) {
    if (fa[i] != i) {
        int t = fa[i];
        fa[i] = find(fa[i]);

        dis[i] += dis[t];
    }
    return fa[i];
}

bool merge(int l, int r, int v) {
    int fl = find(l), fr = find(r);
    if (fl == fr) return false;

    fa[fl] = fr;
    dis[fl] = dis[r] - dis[l] + v;
    return true;
}

int query(int l, int r) {
    if (find(l) != find(r)) {
        return 1e18;
    }
    return dis[l] - dis[r];
}

void ap() {
	int n, m, q; cin >> n >> m >> q;

    fa.resize(n + 2), dis.resize(n + 2);
    for (int i = 0; i < n + 2; i++) {
        fa[i] = i;
        dis[i] = 0;
    }
    
    for (int i = 0; i < m; i++) {
        int l, r, s; cin >> l >> r >> s;
        r++;

        merge(l, r, s);
    }

    for (int i = 0; i < q; i++) {
        int l, r; cin >> l >> r;
        r++;

        if (find(l) != find(r)) {
            cout << "UNKNOWN" << endl;
        } else {
            cout << dis[l] - dis[r] << endl;
        }
    }
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;

		ap();
	}

}
```

# 相对距离dis - 详解

`dis[i]` 表示 点i 和 代表节点X 的相对距离
本题tip: 点l 一定在 点r 的左边
### **在find的过程中**
```c++
int find(int i) {
    if (fa[i] != i) {
        int t = fa[i];
        fa[i] = find(fa[i]);

        dis[i] += dis[t];
    }
    return fa[i];
}
```

先不断往上找到代表节点，然后自顶向下不断更新dis
代表节点的dis为0
t为节点i的旧的fa节点

**相对关系如下：**
令代表节点为X
`dis[t]` ：t -> X ，已经更新
`dis[i]`：i -> t, 尚未更新

我们需要更新`dis[i]`，即需要有 i -> t, t -> X
那么更新`dis[i]` = `dis[i] + dis[t]`


### 在merge的过程中
```c++
bool merge(int l, int r, int v) {
    int fl = find(l), fr = find(r);
    if (fl == fr) return false;

    fa[fl] = fr;
    dis[fl] = dis[r] - dis[l] + v;
    return true;
}
```

因为先调用了find，所以有如下关系：
- `dis[l] = l -> fl`
- `dis[r] = r -> fr`
- `v = l -> r`

需求：
- `fl -> fr`

相对关系的解法：
- `fl -> l`
- `l -> r`
- `r -> fr`

转换即 `-dis[l] + dis[r] + v`

即`dis[fl] = dis[r] - dis[l] + v`

### query

因为l， r在一个集合中
所以有
- `dis[l] = l -> X`
- `dis[r] = r -> X`

需要：
- `l -> r`

相对关系的解法：
- `l -> X`
- `X -> r`

转换即 `dis[l] - dis[r]`
