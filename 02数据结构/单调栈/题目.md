#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1), r(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i], r[i] = i + a[i];

    vector<array<int, 2>> st;//id, mianji

    vector<int> ans(n + 1);

    int sum = 0;
    for (int i = n; i >= 1; i--) {
        while (st.size() && r[st.back()[0]] <= r[i]) {
            sum -= st.back()[1];

            st.pop_back();
        }

        int e = (st.empty() ? r[i] : min(st.back()[0], r[i])) - i;

        int t = (a[i] + a[i] - e) * e;

        sum += t;
        st.push_back({i, t});

        ans[i] = sum;
    }

    for (int i = 1; i <= n; i++) cout << ans[i] << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
