
# 1. 线段树判断区间全部相等

[F - Make them monochrome](https://atcoder.jp/contests/abc484/tasks/abc484_f)

- 区间最大值等于最小值，则整个区间的数都相等

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

/*
用途：线段树（单点修改、区间和查询、区间最大值查询、区间最小值查询）。
特性：无懒标记（无需 down 下传），常数极小，支持 1-indexed 数据与查询。
复杂度：建树 O(n)，单点修改 O(log n)，区间查询 O(log n)，空间复杂度 O(n)。

接口（均为 1-indexed）：
  - build(arr)          : 传入 1-indexed 的原始数组以 O(n) 完成建树。
  - update(pos, val)    : 将下标 pos 处的值修改为 val（单点覆盖）。
  - add(pos, val)       : 将下标 pos 处的值增加 val（单点累加，按需选用）。
  - query_sum(l, r)     : 查询区间 [l, r] 的累加和。
  - query_max(l, r)     : 查询区间 [l, r] 的最大值。
  - query_min(l, r)     : 查询区间 [l, r] 的最小值。
  - query(l, r)         : 一次性查询区间 [l, r] 的 (sum, max, min)，返回 Node。

注意事项：
  - 传入的原始数组 `arr` 大小需至少为 n + 1，有效数据从下标 1 开始。
  - 若用于累加和大范围数据，建议模板参数类型使用 long long 防止溢出。
*/
template <typename T = long long>
struct SegTree {
    struct Node {
        T sum;
        T mx;
        T mn;
    };

    int n;
    std::vector<T> sum;
    std::vector<T> mx;
    std::vector<T> mn;

    explicit SegTree(int size) : n(size),
                                 sum((size + 1) << 2, 0),
                                 mx((size + 1) << 2, std::numeric_limits<T>::lowest()),
                                 mn((size + 1) << 2, std::numeric_limits<T>::max()) {}

    // 向上汇总信息
    void up(int i) {
        sum[i] = sum[i << 1] + sum[i << 1 | 1];
        mx[i] = std::max(mx[i << 1], mx[i << 1 | 1]);
        mn[i] = std::min(mn[i << 1], mn[i << 1 | 1]);
    }

    // 建树
    void build(int l, int r, int i, const std::vector<T>& arr) {
        if (l == r) {
            sum[i] = arr[l];
            mx[i] = arr[l];
            mn[i] = arr[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(l, mid, i << 1, arr);
        build(mid + 1, r, i << 1 | 1, arr);
        up(i);
    }

    // 单点重置/赋值：arr[pos] = val
    void update(int pos, T val, int l, int r, int i) {
        if (l == r) {
            sum[i] = val;
            mx[i] = val;
            mn[i] = val;
            return;
        }
        int mid = (l + r) >> 1;
        if (pos <= mid) {
            update(pos, val, l, mid, i << 1);
        } else {
            update(pos, val, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 单点增加：arr[pos] += val
    void add(int pos, T val, int l, int r, int i) {
        if (l == r) {
            sum[i] += val;
            mx[i] += val;
            mn[i] += val;
            return;
        }
        int mid = (l + r) >> 1;
        if (pos <= mid) {
            add(pos, val, l, mid, i << 1);
        } else {
            add(pos, val, mid + 1, r, i << 1 | 1);
        }
        up(i);
    }

    // 统一区间查询：一次递归合并 sum, max, min
    Node query(int jobl, int jobr, int l, int r, int i) {
        if (jobl <= l && r <= jobr) {
            return {sum[i], mx[i], mn[i]};
        }
        int mid = (l + r) >> 1;
        if (jobr <= mid) {
            return query(jobl, jobr, l, mid, i << 1);
        }
        if (jobl > mid) {
            return query(jobl, jobr, mid + 1, r, i << 1 | 1);
        }
        Node left_res = query(jobl, jobr, l, mid, i << 1);
        Node right_res = query(jobl, jobr, mid + 1, r, i << 1 | 1);
        return {
            left_res.sum + right_res.sum,
            std::max(left_res.mx, right_res.mx),
            std::min(left_res.mn, right_res.mn)
        };
    }

    // 便捷对外接口
    void build(const std::vector<T>& arr) { build(1, n, 1, arr); }
    void update(int pos, T val) { update(pos, val, 1, n, 1); }
    void add(int pos, T val) { add(pos, val, 1, n, 1); }
    
    Node query(int l, int r) { return query(l, r, 1, n, 1); }
    T query_sum(int l, int r) { return query(l, r, 1, n, 1).sum; }
    T query_max(int l, int r) { return query(l, r, 1, n, 1).mx; }
    T query_min(int l, int r) { return query(l, r, 1, n, 1).mn; }
};

void ap() {
	int n, q; cin >> n >> q;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    SegTree tree1(n);
    tree1.build(a);

    vector<int> b(n + 1);
    for (int i = 1; i <= n; i++) b[i] = (a[i] == -1 ? 1e9 : a[i]);
    
    SegTree tree2(n);
    tree2.build(b);

    for (int i = 0; i < q; i++) {
        int opt, x, y; cin >> opt >> x >> y;

        if (opt == 1) {
            tree1.update(x, y);
            tree2.update(x, (y == -1 ? 1e9 : y));

        } else {
            int mx = tree1.query_max(x, y);
            int mn = tree2.query_min(x, y);

            if (mn >= mx) {
                cout << "Yes" << endl;
            } else {
                cout << "No" << endl;
            }
        }
    }


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 2. 树状数组 + 前缀和离散化

[E - Subsegment Sum Finder](https://atcoder.jp/contests/abc484/tasks/abc484_e)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

 /*
 用途：一维树状数组，默认维护单点加、前缀和与区间和。

接口（均为 1-indexed）：modify(p,x)、ask(p)、ask(l,r)、askKth(k)。
update(l,r,x) 是差分用法：区间加、单点查询；不要与区间和语义混用。
askKth(k) 返回最小的 p 使前缀和 >= k；要求各点权非负，非法 k 返回 n + 1。
*/
template <class T>
struct BIT {
    int n = 0;
    vector<T> bit;

    BIT() = default;
    explicit BIT(int n) {
        init(n);
    }

    void init(int _n) {
        n = _n;
        bit.assign(n + 1, T{});
    }

    template <class A>
    void build(const A &a) {
        fill(bit.begin(), bit.end(), T{});
        for (int i = 1; i <= n; ++i) {
            bit[i] += a[i];
            int j = i + (i & -i);
            if (j <= n) bit[j] += bit[i];
        }
    }

    //将位置 x 增加 v
    void modify(int x, const T &v) {
        assert(1 <= x && x <= n);
        for (; x <= n; x += x & -x) bit[x] += v;
    }

    void update(int l, int r, const T &v) {
        if (l > r) return;
        assert(1 <= l && r <= n);
        modify(l, v);
        if (r < n) modify(r + 1, -v);
    }

    //查询前缀和 [1, x]
    T ask(int x) const {
        x = min(x, n);
        T res{};
        for (; x > 0; x -= x & -x) res += bit[x];
        return res;
    }

    T ask(int l, int r) const {
        if (l > r) return T{};
        assert(1 <= l && r <= n);
        return ask(r) - ask(l - 1);
    }

    int askKth(const T &k) const {
        if (!(k > T{})) return 1;
        int pos = 0;
        T pre{};
        int stp = 1;
        while ((stp << 1) <= n) stp <<= 1;
        for (; stp; stp >>= 1) {
            int nxt = pos + stp;
            if (nxt <= n && pre + bit[nxt] < k) {
                pre += bit[nxt];
                pos = nxt;
            }
        }
        return pos + 1;
    }
};

void ap() {
	int n, L, R; cin >> n >> L >> R;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    vector<int> pre(n + 1);
    for (int i = 0; i < n; i++) {
        pre[i + 1] = pre[i] + a[i];
    }

    vector<int> b = pre;
    sort(b.begin(), b.end());
    b.erase(unique(b.begin(), b.end()), b.end());

    int m = b.size();
    BIT<int> bit(m);

    int j = lower_bound(b.begin(), b.end(), pre[0]) - b.begin() + 1;
    bit.modify(j, 1);

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        int l = pre[i] - R;
        int r = pre[i] - L;

        int left = lower_bound(b.begin(), b.end(), l) - b.begin() + 1;
        // int right = lower_bound(b.begin(), b.end(), r + 1) - b.begin() - 1;
        int right = upper_bound(b.begin(), b.end(), r) - b.begin();

        ans += bit.ask(left, right);

        int j = lower_bound(b.begin(), b.end(), pre[i]) - b.begin() + 1;
        bit.modify(j, 1);
    }

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
