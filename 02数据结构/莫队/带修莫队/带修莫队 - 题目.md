```c++
#include <bits/stdc++.h>
using namespace std;

// 根据题目数据规模 n, q <= 100000 
// 离散化后最多有 n + q 种不同的数，数组开到 200005 足够
const int MX = 200005;

int a[MX];
int bi[MX];

// cnt[x]: 数值 x 当前在区间内出现的次数
int cnt[MX];
// ccnt[v]: “出现次数为 v”的数值有多少种
// 例如有 3 个数都出现了 2 次，则 ccnt[2] = 3
int ccnt[MX]; 

// 存储修改操作
array<int, 2> update[MX];

int cntu = 0, cntq = 0;

// 莫队排序：左端点所在块 -> 右端点所在块 -> 时间戳
int cmp(const array<int, 4> & a, const array<int, 4> & b) {
    if (bi[a[0]] != bi[b[0]]) return bi[a[0]] < bi[b[0]];
    if (bi[a[1]] != bi[b[1]]) return bi[a[1]] < bi[b[1]];
    return a[2] < b[2];
}

// 核心改动：增加与删除要同时维护 cnt 和 ccnt
void add(int x) {
    ccnt[cnt[x]]--;      // 旧的出现次数种数减 1
    cnt[x]++;            // 出现次数加 1
    ccnt[cnt[x]]++;      // 新的出现次数种数加 1
}

void del(int x) {
    ccnt[cnt[x]]--;      // 旧的出现次数种数减 1
    cnt[x]--;            // 出现次数减 1
    ccnt[cnt[x]]++;      // 新的出现次数种数加 1
}

// 时间轴移动，务必传入莫队当前的真实边界 wl 和 wr
void movetime(int wl, int wr, int tim) {
    int pos = update[tim][0];
    int val = update[tim][1];
    
    if (wl <= pos && pos <= wr) {
        del(a[pos]);
        add(val);
    }

    // 交换历史值
    swap(a[pos], update[tim][1]);
}

signed main () {
    // 优化输入输出，防止数量大时超时
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m; 
    if (!(cin >> n >> m)) return 0;

    // 离散化准备：收集所有可能出现的数值
    vector<int> vals;
    vals.reserve(n + m);

    for (int i = 1; i <= n; i++) {
        cin >> a[i];
        vals.push_back(a[i]);
    }

    vector<array<int, 4>> q(m + 1); // l, r, t, id
    for (int i = 1; i <= m; i++) {
        int type; cin >> type;
        if (type == 1) {
            cntq++;
            cin >> q[cntq][0] >> q[cntq][1];
            q[cntq][2] = cntu;
            q[cntq][3] = cntq;
        } else {
            cntu++;
            cin >> update[cntu][0] >> update[cntu][1];
            vals.push_back(update[cntu][1]); // 收集修改后的新值
        }
    }

    // 离散化三步走：排序、去重
    sort(vals.begin(), vals.end());
    vals.erase(unique(vals.begin(), vals.end()), vals.end());

    // 映射回紧凑的小整数
    for (int i = 1; i <= n; i++) {
        a[i] = lower_bound(vals.begin(), vals.end(), a[i]) - vals.begin() + 1;
    }
    for (int i = 1; i <= cntu; i++) {
        update[i][1] = lower_bound(vals.begin(), vals.end(), update[i][1]) - vals.begin() + 1;
    }

    // 块大小优化：带修改莫队理论最优块大小为 N^(2/3)
    int blen = max(1, (int) pow(n, 2.0 / 3));
    for (int i = 1; i <= n; i++) {
        bi[i] = (i - 1) / blen + 1;
    }
    
    // 对有效查询部分进行排序
    sort(q.begin() + 1, q.begin() + cntq + 1, cmp);

    vector<int> ans(cntq + 1);
    int winl = 1, winr = 0, wint = 0;
    
    // 初始化：没有任何数时，“出现次数为 0”的数有无限个
    // 这里设为一个很大的数，保证求 Mex 的时候不会误判 0
    ccnt[0] = 1e9; 

    for (int i = 1; i <= cntq; i++) {
        int ql = q[i][0];
        int qr = q[i][1];
        int qt = q[i][2];
        int id = q[i][3];

        // 移动区间指针
        while (winl > ql) add(a[--winl]);
        while (winr < qr) add(a[++winr]);
        while (winl < ql) del(a[winl++]);
        while (winr > qr) del(a[winr--]);

        // 移动时间指针
        while (wint < qt) movetime(winl, winr, ++wint);
        while (wint > qt) movetime(winl, winr, wint--);

        // 暴力枚举求 Mex
        // 理论证明：在 N=100000 下，不同的出现次数最多有 sqrt(2N) ≈ 450 种
        // 所以这里的 Mex 答案绝不会超过 500，直接从 1 开始暴力枚举极快
        int mex = 1;
        while (ccnt[mex] > 0) {
            mex++;
        }
        ans[id] = mex;
    }

    for (int i = 1; i <= cntq; i++) {
        cout << ans[i] << "\n";
    }

    return 0;
}
```
