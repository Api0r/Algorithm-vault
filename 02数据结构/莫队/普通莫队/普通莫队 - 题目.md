```c++
#include<bits/stdc++.h>
using namespace std;
#define int long long//后面有* 不开long long 会wa

const int MX = 50005;
vector<int> bi(MX);

int cmp(const array<int, 3> & a, const array<int, 3> & b) {
    if (bi[a[0]] != bi[b[0]]) {
        return bi[a[0]] < bi[b[0]];
    }

    if ((bi[a[0]] & 1) == 1) {
        return a[1] < b[1];
    } else {
        return b[1] < a[1];
    }
}

//C(cnt, 2) / C(k, 2) +...+
//(cnt * (cnt - 1) + ...+) / (k * (k - 1))
vector<int> cnt(MX);
int sum = 0;
void add(int x) {
    sum -= cnt[x] * (cnt[x] - 1);
    cnt[x]++;
    sum += cnt[x] * (cnt[x] - 1);
}

void del(int x) {
    sum -= cnt[x] * (cnt[x] - 1);
    cnt[x]--;
    sum += cnt[x] * (cnt[x] - 1);
}

signed main () {
    int n, m; cin >> n >> m;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<array<int, 3>> q(m + 1);
    for (int i = 1; i <= m; i++) {
        cin >> q[i][0] >> q[i][1];
        q[i][2] = i;
    }

    int blen = (int)sqrt(n);
    for (int i = 1; i <= n; i++) {
        bi[i] = (i - 1) / blen + 1;
    }
    sort(q.begin() + 1, q.end(), cmp);

    vector<string> ans(m + 1);
    int winl = 1, winr = 0;
    for (int i = 1; i <= m; i++) {
        int ql = q[i][0];
        int qr = q[i][1];
        while (winl > ql) add(a[--winl]);
        while (winr < qr) add(a[++winr]);
        while (winl < ql) del(a[winl++]);
        while (winr > qr) del(a[winr--]);

        string s;
        if (sum == 0) {
            s = "0/1";
        } else {
            int x = winr - winl + 1;
            x = x * (x - 1);
            int t = __gcd(sum, x);
            int t1 = sum / t, t2 = x / t;
            s = to_string(t1) + "/" + to_string(t2);
        }
        ans[q[i][2]] = s;
    }

    for (int i = 1; i <= m; i++) {
        cout << ans[i] << endl;
    }

    return 0;
}
```
