```c++
#include<bits/stdc++.h>
using namespace std;
//O(n √q)
//https://www.luogu.com.cn/problem/SP3267

const int MX = 1e6 + 5;
vector<int> cnt(MX);
int kind = 0;

vector<int> bi(MX);

//经典排序
//左边界块号 从小到大排序
//分块相同，按右端点从小到大排序

//奇偶排序
//左边界块号从小到大排序
//分块相同，块号为奇数，根据右边界从小到大排序。块号为偶数，根据右边界从大到小排序
int cmp(const array<int, 3> & a, const array<int, 3> & b) {
    if (bi[a[0]] != bi[b[0]]) {
        return bi[a[0]] < bi[b[0]];
    }

    if ((bi[a[0]] & 1) == 1) {
        return a[1] < b[1];
    } else {
        return b[1] < a[1];
    }
}

void del(int x) {
    if (--cnt[x] == 0) {
        kind--;
    }
}

void add(int x) {
    if (++cnt[x] == 1) {
        kind++;
    }
}

signed main () {
    int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    int q; cin >> q;
    vector<array<int, 3>> query(q + 1);
    for (int i = 1; i <= q; i++) {
        cin >> query[i][0] >> query[i][1];
        query[i][2] = i;
    }

    //分块
    int blen = (int)sqrt(n);
    for (int i = 1; i <= n; i++) {
        bi[i] = (i - 1) / blen + 1;
    }
    sort(query.begin() + 1, query.end(), cmp);

    vector<int> ans(q + 1);
    int winl = 1, winr = 0;
    for (int i = 1; i <= q; i++) {
        int ql = query[i][0];
        int qr = query[i][1];
        while (winl > ql) add(a[--winl]);
        while (winr < qr) add(a[++winr]);
        while (winl < ql) del(a[winl++]);
        while (winr > qr) del(a[winr--]);

        ans[query[i][2]] = kind;
    }

    for (int i = 1; i <= q; i++) {
        cout << ans[i] << endl;
    }


    return 0;
}
```
