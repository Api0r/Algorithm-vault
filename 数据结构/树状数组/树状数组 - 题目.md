- 将一行染黑，**黑色增加的数量** 即 上一次染黑之后， 到这一次之前，中间过程染白的列数
- 将一列染白，**黑色减少的数量** 即 上一次染白之后， 到这一次之前， 中间过程染黑的行数

- 对同一行、列 染色，最新的操作会覆盖所有之前的，因此只需维护最新的，删去之前的即可
- 所以每一行、列 只会留下最多一次的增加。(不会有同一行、列出现两次及以上的影响)

- 由hint1知，我们只需要知道 这一行、列 **上一次操作的时间**，以及**中间过程列、行的操作**。即我们需要知道last， 以及last到i - 1 这个**区间的 操作 之和**
- 由和hint2知，我们需要对**单点进行修改**
- 所以我们需要的是 单点更新，区间查询。这可以用树状数组和线段树解决

- 有意思的是，我们从hint1中思考，可以发现树状数组tree需要维护的是 **时间戳**
- 即tree是 时间1 - q。更新即在这个时间上打点1，中间过程的和 即 时间戳 在区间的和


```c++
void update(vector<int> & tree, int i, int val) {
    for (; i < tree.size(); i += i & -i) {
        tree[i] += val;
    }
}

int pre(const vector<int> & tree, int i) {
    int ans = 0;
    for (; i > 0; i -= i & -i) {
        ans += tree[i];
    }
    return ans;
}

int query(const vector<int> & tree, int l, int r) {
    if (r < l) return 0;

    return pre(tree, r) - pre(tree, l - 1);
}

void ap() {
	int n, q; cin >> n >> q;

    vector<int> last0(n + 1, -1), last1(n + 1, 0);//每一行，列 上一次操作的时间i，
    vector<int> tree0(q + 1), tree1(q + 1);//行、列 的时间戳
    
    int cnt = 0;//黑的个数

    for (int i = 1; i <= q; i++) {
        int opt, x; cin >> opt >> x; 

        if (opt == 1) {//第x行 染黑
            if (last0[x] == -1) {
                cnt += n;
            } else {
                int d = query(tree1, last0[x], i - 1);//操作这一行的上一次到这一次之间，有多少次列的操作，就有多少个被染白的格子
                cnt += d;

                update(tree0, last0[x], -1);//每一行我只记录其最新的操作，之前的删去(去重)
            }

            update(tree0, i, 1);
            last0[x] = i;

        } else {//第x列 染白
            int d = query(tree0, last1[x], i - 1);
            cnt -= d;

            if (last1[x] > 0) {
                update(tree1, last1[x], -1);
            }

            update(tree1, i, 1);
            last1[x] = i;
        }

        cout << cnt << endl;
    }
	return ;
}
```
