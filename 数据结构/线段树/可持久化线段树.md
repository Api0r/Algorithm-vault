```c++
#include<bits/stdc++.h>
using namespace std;
//https://www.luogu.com.cn/problem/P3919#ide
//单点更新，单点查询
const int N = 1e6 + 1;
const int M = N * 25;

//root[version] = i，代表版本v 的头节点的编号是i
//ls[i] = j，代表 编号i 的左儿子节点的 编号为j
int a[N];
int root[N];
int ls[M], rs[M], val[M];//Left Son, Right Son 即 儿子节点的编号。 叶子节点的值value

int cnt = 0;

//建树，返回头节点编号
int build(int l, int r) {//具体的讲，返回范围为l ~ r 的头节点编号
    int rt = ++cnt;
    if (l == r) {
        val[rt] = a[l];
    } else {
        int mid = (l + r) >> 1;
        ls[rt] = build(l, mid);//ls和sr的rt是一样的
        rs[rt] = build(mid + 1, r);//建树是类似先序遍历的顺序去打编号
    }
    return rt;
}

//线段树范围l ~ r, 信息在i号节点里
//在l ~ r范围上，将x位置的值 更新为 value
//返回生成的新节点的编号
//最终返回的是头节点的编号
int update(int i, int l, int r, int x, int value) {
    //进到这里来了，说明这里一定有需要修改的，所以要打编号
    //打编号后，直接拷贝一份旧的，然后往下看需要修改的地方
    int rt = ++cnt;
    ls[rt] = ls[i];
    rs[rt] = rs[i];
    val[rt] = val[i];
    
    if (l == r) {
        val[rt] = value;
    } else {
        int mid = (l + r) >> 1;
        if (x <= mid) {
            ls[rt] = update(ls[rt], l, mid, x, value);
        } else {
            rs[rt] = update(rs[rt], mid + 1, r, x, value);
        }
    }
    return rt;
}

//线段树范围l ~ r, 信息在i号节点里
//返回l ~ r范围上 idx 位置的值
int query(int i, int l, int r, int idx) {
    if (l == r) {
        return val[i];
    }
    int mid = (l + r) >> 1;
    if (idx <= mid) {
        return query(ls[i], l, mid, idx);
    } else {
        return query(rs[i], mid + 1, r, idx);
    }
}

signed main () {
    ios::sync_with_stdio(0), cin.tie(0);
    
    int n, m; cin >> n >> m;
    for (int i = 1; i <= n; i++) {//这里要从1开始
        cin >> a[i];
    }

    //0版本的头节点的编号为root[0], 同时建树
    root[0] = build(1, n);

    for (int i = 1; i <= m; i++) {
        int v, op, x; cin >> v >> op >> x;
        
        if (op == 1) {
            int y; cin >> y;
            root[i] = update(root[v], 1, n, x, y);
        } else {
            root[i] = root[v];
            cout << query(root[i], 1, n, x) << endl;
        }
    }


    return 0;
}
```
