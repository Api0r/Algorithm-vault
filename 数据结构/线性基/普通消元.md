vector<int> basis;

int BIT = 64;


bool insert(int x) {
    for (int i = BIT; i >= 0; i--) {
        if (x >> i == 1) {
            if (basis[i] == 0) {
                basis[i] = x;
                return true;
            }

            x ^= basis[i];
        }
    }
    return false;
}

int max_xor() {
    int ans = 0;
    for (int i = BIT; i >= 0; i--) {//这里一定要从高位开始遍历，不可以直接用加强for循环
        ans = max(ans, ans ^ basis[i]);
    }
    return ans;
}
