#include <iostream>
#include <algorithm>

using namespace std;

// 常量定义
const int MAXN = 100001;
const int BIT = 50;

long long arr[MAXN];
int len = 0;
bool zero = false;
int n;

// 高斯消元构建线性基（简化基/异或空间基底）
void compute() {
    len = 1;
    for (int i = BIT; i >= 0; i--) {
        for (int j = len; j <= n; j++) {
            if ((arr[j] & (1LL << i)) != 0) {
                swap(arr[j], arr[len]);
                break;
            }
        }
        if ((arr[len] & (1LL << i)) != 0) {
            for (int j = 1; j <= n; j++) {
                if (j != len && (arr[j] & (1LL << i)) != 0) {
                    arr[j] ^= arr[len];
                }
            }
            len++;
        }
    }
    len--;
    zero = (len != n);
}

// 查询第 k 小的异或和
long long query(long long k) {
    if (zero && k == 1) {
        return 0;
    }
    if (zero) {
        k--;
    }
    if (k >= (1LL << len)) {
        return -1;
    }
    long long ans = 0;
    for (int i = len, j = 0; i >= 1; i--, j++) {
        if ((k & (1LL << j)) != 0) {
            ans ^= arr[i];
        }
    }
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> arr[i];
    }

    compute();

    int q;
    cin >> q;
    while (q--) {
        long long k;
        cin >> k;
        cout << query(k) << "\n";
    }

    return 0;
}
