```c++
#include <bits/stdc++.h>
using namespace std;
//只需要撤回操作，不需要查询每个版本的信息

const int MAXN = 200005; // 根据题目要求修改最大节点数

// 并查集核心数组
int fa[MAXN];
int sz[MAXN];

// 历史记录栈（用三个同步的 vector 分别记录被修改的节点和状态）
vector<int> history_u;        // 记录合并中的父亲节点
vector<int> history_v;        // 记录合并中的儿子节点
vector<int> history_old_sz_u; // 记录父亲节点在合并前的旧 size

// 1. 初始化并查集
void init_dsu(int n) {
    for (int i = 1; i <= n; i++) {
        fa[i] = i;
        sz[i] = 1;
    }
    // 清空历史栈
    history_u.clear();
    history_v.clear();
    history_old_sz_u.clear();
}

// 2. 查找代表元（注意：绝对不能路径压缩，纯通过 while 循环往上找）
int find(int x) {
    while (x != fa[x]) {
        x = fa[x];
    }
    return x;
}

// 3. 合并操作
bool merge_elements(int x, int y) {
    int fx = find(x);
    int fy = find(y);
    if (fx == fy) return false; // 已经在同一个集合，不需要合并

    // 按秩合并：把 size 小的树 挂到 size 大的树 下面，保证树的高度平衡
    if (sz[fx] < sz[fy]) {
        swap(fx, fy);
    }

    // 【核心】在修改前，把当前的合并关系和受影响的旧数据压入历史栈中
    history_u.push_back(fx);
    history_v.push_back(fy);
    history_old_sz_u.push_back(sz[fx]);

    // 正式修改并查集状态
    fa[fy] = fx;       // 小树挂到大树上
    sz[fx] += sz[fy];  // 大树的 size 加上小树的 size
    
    return true;
}

// 4. 获取当前版本号（其实就是当前历史栈里存了多少次合并）
int get_version() {
    return history_u.size();
}

// 5. 回滚到指定的历史版本
void rollback(int target_version) {
    // 只要当前的合并次数大于目标版本，就执行“吐出”和“还原”
    while (history_u.size() > target_version) {
        int u = history_u.back();
        int v = history_v.back();
        int old_sz_u = history_old_sz_u.back();
        
        // 弹出栈顶
        history_u.pop_back();
        history_v.pop_back();
        history_old_sz_u.pop_back();

        // 逆向完全复原状态
        fa[v] = v;          // 把儿子节点断开，重新变回它自己的根
        sz[u] = old_sz_u;   // 还原父亲节点合并前的旧 size
    }
}

signed main () {
    ios::sync_with_stdio(0), cin.tie(0);

    return 0;
}
```
