
# 一、普通并查集的巧妙做法

[F-小橙的异或和_牛客小白月赛129](https://ac.nowcoder.com/acm/contest/128675/F)
![](../../image/Pasted%20image%2020260625112335.png)
每个数最多被修改32次就会变成0，变成0之后无论怎操作都是0
也就是说，最多只需要修改n * 32 次，就无需再修改

- 对于需要修改的区间中的元素，如果是0，那么跳过
- 我们可以用set集合存储非零的下标
- 通过set内lower_bound找到第一个 >= l + 1的下标，然后迭代遍历
- set的增删时间复杂度为logn

- 可以用并查集维护，作为跳板，如果当前数变为了0，那么将其fa指向下一个数的代表节点
- 那么每次我们只需要find(i + 1)，找到下一个非零的点。
- 因为中间所有的0都被合并了，都指向下一个非零的点

```c++
vector<int> fa;

int find (int i) {
    if (fa[i] != i) {
        fa[i] = find(fa[i]);
    }
    return fa[i];
}

bool merge(int i, int j) {
    int x = find(i), y = find(j);
    if (x == y) return false;
    
    fa[x] = y;
    return true;
}

void ap() {
	int n, q; cin >> n >> q;
    vector<int> a(n);
    int x = 0;
    for (int & i : a) cin >> i, x ^= i;
    
    fa.resize(n + 5);//注意这里需要开大一位，因为后面会用到i = find(i + 1);
    for (int i = 0; i < n + 5; i++) {
        fa[i] = i;
    }
    
    while (q--) {
        int opt; cin >> opt;
        if (opt == 1) {
            int l, r; cin >> l >> r;
            l--, r--;
            
            int i = find(l + 1);
            while (i <= r) {
                int t = a[i] / (i - l + 1);
                x = x ^ a[i] ^ t;
                
                a[i] = t;
                
                if (a[i] == 0) {
                    fa[i] = find(i + 1);
                }
                
                i = find(i + 1);
            }
        } else {
            cout << x << endl;
        }
    }
    
    
    
	return ;
}
```

```c++
void ap() {
    int n, q; 
    cin >> n >> q;
    vector<int> a(n);
    int x = 0;
    
    set<int> active_idx; // 存储所有当前值不为 0 的下标

    for (int i = 0; i < n; i++) {
        cin >> a[i];
        x ^= a[i]; // 维护初始全局异或和
        if (a[i] > 0) {
            active_idx.insert(i); // 只有非 0 元素的下标才放进去
        }
    }
    
    while (q--) {
        int opt; cin >> opt;
        if (opt == 1) {
            int l, r; cin >> l >> r;
            l--, r--; // 转换为 0-indexed
            
            // 题意：当 i = l 时除数是 1，值不变。因此我们直接找 >= l + 1 的第一个非零位置
            auto it = active_idx.lower_bound(l + 1);
            
            // 遍历所有在 [l+1, r] 闭区间内的有效非零下标
            while (it != active_idx.end() && *it <= r) {
                int i = *it;
                int t = a[i] / (i - l + 1);
                
                x = x ^ a[i] ^ t; // 更新异或和
                a[i] = t;
                
                if (a[i] == 0) {
                    // 如果变成 0 了，需要从 set 中删掉
                    // erase(it++) 是标准写法：删除当前迭代器，并让 it 顺势指向下一个
                    active_idx.erase(it++); 
                } else {
                    // 如果没变成 0，继续往后看
                    it++; 
                }
            }
        } else {
            cout << x << endl;
        }
    }
    return;
}
```

细节
```c++
int find(int i) {

```
这个代码中，`fa[ i ]` 会不断更新，直到结束，最终fa[i] 指向的是最后的根节点

但是我们需要的是根据上一个节点来更新req数组

所以需要这样写
```c++
int find(int i) {
    if (fa[i] == i) return fa[i];
    
    int root = find(fa[i]);//这里递归完了，返回的是最终的节点
    req[i] = max(req[i], req[fa[i]]);//但是这里的fa[i]仍然是i的上一个节点
    fa[i] = root;//最后在修改fa[i] 为 最终的根节点
    
    return fa[i];
}
```
每次递归，先不直接将fa[i] 连起来，而是先将最终的根节点存起来，再利用上一个节点的信息去更新req，最后在更新fa[i]

## 代码实现

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

vector<int> fa;
vector<int> have;
int cc = 0;
int cnt = 0;

int find(int i) {
    if (fa[i] != i) {
        fa[i] = find(fa[i]);
    }
    return fa[i];
}

bool merge(int i, int j) {
    int x = find(i), y = find(j);
    if (x == y) return false;

    if (have[x] && have[y]) cnt--;
    
    fa[x] = y;
    have[y] = have[y] || have[x];
    cc--;
    return true;
}

int dx[2] = {0, -1}, dy[2] = {-1, 0};

void ap() {
	int n, m; cin >> n >> m;
    vector<string> a(n);
    for (auto & i : a) cin >> i;

	//初始化
    fa.resize(n * m);
    have.assign(n * m, 0);
    for (int i = 0; i < n * m; i++) {
        fa[i] = i;
    }
    cc = 0;
    cnt = 0;

    auto id = [&](int i, int j) -> int {
        return i * m + j;  
    };

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (a[i][j] == '0') continue;
			
			//遍历的过程初始化cc，cnt
            cc++;
            if (a[i][j] > '1') have[id(i, j)] = 1, cnt++;
            
            for (int k = 0; k < 2; k++) {
                int nx = i + dx[k], ny = j + dy[k];

                if (nx < 0 || nx >= n || ny < 0 || ny >= m || a[nx][ny] == '0') continue;

                merge(id(i, j), id(nx, ny));
            }
        }
    }

    cout << cc << ' ' << cnt;

    cout << endl;
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
```

## 信息维护

- **岛屿的数量**（`cc`）：经典并查集连通块计数问题。
  
  需要注意的是，不同于一开始就将 `cc` 初始化为 `n`，这里是在遍历网格的过程中进行 `cc++`，因为初始时并非所有格子都是陆地（`'0'` 表示海洋）。

- **宝藏岛屿的数量**（`cnt`）：通过 `have` 数组维护。

---

### 核心总结：`have` 数组的状态转移与维护

#### 1. 核心观念转变：`have` 维护的是”整个岛”而非”单点”

在并查集中，`have` 数组的数据**只对根节点有意义**：

- **若 `i` 不是根节点**：`have[i]` 的值失效，不用关注也不必更新。
- **若 `i` 是根节点**（即 `find(i) == i`）：`have[i]` 严格代表**以 `i` 为根的整个岛屿**是否包含宝藏。

#### 2. 为什么只在 `merge` 时更新就够了？

因为所有涉及岛屿的操作都是通过 `find` 找到最高祖先（根节点）来完成的：

1. **合并前寻根**：`merge(i, j)` 第一步就是通过 `int x = find(i)` 和 `int y = find(j)` 找到两个岛屿的最高根节点 `x` 和 `y`。
2. **读取全岛状态**：此时 `have[x]` 和 `have[y]` 代表的是合并前这两个大岛的整体宝藏情况。
3. **状态汇入新根**：当把 `x` 挂到 `y` 下面后（`fa[x] = y`），`y` 成为了合并后新大岛的唯一根节点。我们执行 `have[y] = have[x] || have[y]`，成功把 `x` 岛的宝藏信息传递给了新根 `y`。
4. **子节点自动生效**：由于后续对于原 `x` 岛内任意节点的查询，`find` 都会顺着指针找到新根 `y`，因此**不需要去更新那些子节点的 `have` 值**，整个岛的状态就已经更新完成了。



# 四、普通并查集 维护环的巧妙用法

[C-小橙买水果_牛客小白月赛134](https://ac.nowcoder.com/acm/contest/137072/C)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

vector<int> fa;

int find(int i) {
    if (fa[i] != i) {
        fa[i] = find(fa[i]);
    }
    return fa[i];
}

bool merge(int i, int j) {
    int x = find(i), y = find(j);
    if (x == y) return false;
    
    fa[x] = y;
    return true;
}

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;
    
    fa.resize(n + 1);
    for (int i = 0; i < n + 1; i++) {
        fa[i] = i;
    }
    
    for (int i = 0; i < 2 * n; i++) {
        if (a[i % n] >= a[(i + 1) % n]) {
            fa[(i + 1) % n] = find(i % n);
        }
    }
    
    int ans = 0;
    set<int> st;
    
    for (int i = 0; i < n; i++) {
        int x = find(i);
        if (st.count(x)) continue;
        
		ans += a[x];
		st.insert(x);
	    
    }
    
    cout << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
