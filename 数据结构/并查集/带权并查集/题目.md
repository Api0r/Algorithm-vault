# 一维数轴相对距离模型

## 1）连续线段的和 - 转化为 - 点的距离

[P8779 [蓝桥杯 2022 省 A] 推导部分和 - 洛谷](https://www.luogu.com.cn/problem/P8779)

**连续线段`[l, r]`的和，可以转换为 点` [l, r + 1]`的距离**

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

vector<int> fa;
vector<int> dis;

int find(int i) {
    if (fa[i] != i) {
        int t = fa[i];
        fa[i] = find(fa[i]);

        dis[i] += dis[t];
    }
    return fa[i];
}

bool merge(int l, int r, int v) {
    int fl = find(l), fr = find(r);
    if (fl == fr) return false;

    fa[fl] = fr;
    dis[fl] = dis[r] - dis[l] + v;
    return true;
}

int query(int l, int r) {
    if (find(l) != find(r)) {
        return 1e18;
    }
    return dis[l] - dis[r];
}

void ap() {
	int n, m, q; cin >> n >> m >> q;

    fa.resize(n + 2), dis.resize(n + 2);
    for (int i = 0; i < n + 2; i++) {
        fa[i] = i;
        dis[i] = 0;
    }
    
    for (int i = 0; i < m; i++) {
        int l, r, s; cin >> l >> r >> s;
        r++;

        merge(l, r, s);
    }

    for (int i = 0; i < q; i++) {
        int l, r; cin >> l >> r;
        r++;

        if (find(l) != find(r)) {
            cout << "UNKNOWN" << endl;
        } else {
            cout << dis[l] - dis[r] << endl;
        }
    }
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

## 2）

[P2294 [HNOI2005] 狡猾的商人 - 洛谷](https://www.luogu.com.cn/problem/P2294)

**两个在同一集合中的点，其相对距离已经确定**

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

vector<int> fa, dis;

int find(int i) {
    if (fa[i] != i) {
        int t = fa[i];

        fa[i] = find(fa[i]);

        dis[i] += dis[t];
    }
    return fa[i];
}

bool merge(int l, int r, int v) {
    int fl = find(l), fr = find(r);
    if (fl == fr) return false;

    fa[fl] = fr;
    dis[fl] = dis[r] - dis[l] + v;
    return true;
}

bool ok(int l, int r, int v) {
    if (find(l) == find(r)) {
        if (dis[l] - dis[r] != v) {
            return false;
        }
    }
    return true;
}

//如果两个点在同一个集合中，那么这两个点的相对距离就可以查询出来
//因此我们只需要在merge之前 check一下就可以
void ap() {
    int n, m; cin >> n >> m;

    fa.resize(n + 2), dis.resize(n + 2);
    for (int i = 0; i < n + 2; i++) {
        fa[i] = i;
        dis[i] = 0;
    }

    bool ans = true;
    for (int i = 0; i < m; i++) {
        int l, r, v; cin >> l >> r >> v;
        r++;

        if (!ok(l, r, v)) {
            ans = false;//不可以return，因为还没读完数据
        } else {
            merge(l, r, v);
        }
    }

    if (ans) {
        cout << "true" << endl;
    } else {
        cout << "false" << endl;
    }

        
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```


# 二、维护距离关系

[P1196 [NOI2002] 银河英雄传说 - 洛谷](https://www.luogu.com.cn/problem/P1196)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

const int MAXN = 30005;
int fa[MAXN], dis[MAXN], sz[MAXN];

int find(int i) {
    if (fa[i] != i) {
        int t = fa[i];

        fa[i] = find(fa[i]);

        dis[i] += dis[t];
    }
    return fa[i];
}

bool merge(int l, int r) {
    int fl = find(l), fr =  find(r);
    if (fl == fr) return false;

    fa[fl] = fr;
    
    dis[fl] += sz[fr];
    sz[fr] += sz[fl];
    return true;
}

int query(int i, int j) {
    if (find(i) != find(j)) {
        return -1;
    }

    return abs(dis[i] - dis[j]) - 1;
}

void ap() {
	int n; cin >> n;
    for (int i = 0; i < MAXN; i++) {
        fa[i] = i;
        dis[i] = 0;
        sz[i] = 1;
    }
    
    for (int i = 0; i < n; i++) {
        char c; cin >> c;
        int x, y; cin >> x >> y;
        if (c == 'M') {
            merge(x, y);
        } else {
            cout << query(x, y) << endl;
        }
    }
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
