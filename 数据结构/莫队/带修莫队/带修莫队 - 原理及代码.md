```c++
#include<bits/stdc++.h>
using namespace std;

const int MX = 133333 + 5;
const int M = 1e6 + 5;
vector<int> a(MX);
vector<int> bi(MX);
vector<int> cnt(M);

vector<array<int, 2>> update(MX);

int cntu = 0, cntq = 0;
int kind = 0;

int cmp(const array<int, 4> & a, const array<int, 4> & b) {
    if (bi[a[0]] != bi[b[0]]) {
        return bi[a[0]] < bi[b[0]];
    }
    if (bi[a[1]] != bi[b[1]]) {
        return bi[a[1]] < bi[b[1]];
    }
    return a[2] < b[2];
}

void del(int x) {
    if (--cnt[x] == 0) {
        kind--;
    }
}

void add(int x) {
    if (++cnt[x] == 1) {
        kind++;
    }
}

void movetime(int ql, int qr, int tim) {
    int pos = update[tim][0];
    int val = update[tim][1];
    if (ql <= pos && pos <= qr) {
        del(a[pos]);
        add(val);
    }

    int t = a[pos];
    a[pos] = val;
    update[tim][1] = t;
}

signed main () {
    int n, m; cin >> n >> m;
    a.resize(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<array<int, 4>> q(m + 1);//l, r, t, id
    for (int i = 1; i <= m; i++) {
        char c; cin >> c;
        if (c == 'Q') {
            cntq++;

            cin >> q[cntq][0] >> q[cntq][1];
            q[cntq][2] = cntu;
            q[cntq][3] = cntq;
        } else {
            cntu++;

            int pos, val; cin >> pos >> val;
            update[cntu][0] = pos;
            update[cntu][1] = val;
            
        }
    }

    int blen = max(1, (int) pow(n, 2.0 / 3));
    for (int i = 1; i <= n; i++) {
        bi[i] = (i - 1) / blen + 1;
    }
    sort(q.begin() + 1, q.begin() + cntq + 1, cmp);

    vector<int> ans(cntq + 1);
    int winl = 1, winr = 0, wint = 0;
    for (int i = 1; i <= cntq; i++) {
        int ql = q[i][0];
        int qr = q[i][1];
        int qt = q[i][2];
        int id = q[i][3];

        while (winl > ql) add(a[--winl]);
        while (winr < qr) add(a[++winr]);
        while (winl < ql) del(a[winl++]);
        while (winr > qr) del(a[winr--]);

        while (wint < qt) movetime(ql, qr, ++wint);
        while (wint > qt) movetime(ql, qr, wint--);

        ans[id] = kind;
    }

    for (int i = 1; i <= cntq; i++) {
        cout << ans[i] << endl;
    }
    

    return 0;
}
```
