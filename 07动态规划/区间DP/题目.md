[Palindromic and Balanced - 题目 - QOJ.ac](https://qoj.ac/contest/3766/problem/17344/statement/zh_cn)

```c++
#include <iostream>
#include <string>
#include <vector>
#include <algorithm>

using namespace std;

void solve() {
    int n;
    if (!(cin >> n)) return;
    string s;
    cin >> s;

    // 1. 寻找最外层的 '(' 和 ')'
    int first_open = -1;
    int last_close = -1;
    for (int i = 0; i < n; ++i) {
        if (s[i] == '(') {
            first_open = i;
            break;
        }
    }
    for (int i = n - 1; i >= 0; --i) {
        if (s[i] == ')') {
            last_close = i;
            break;
        }
    }

    // 如果不存在配对的外层基底，输出 0
    if (first_open == -1 || last_close == -1 || first_open >= last_close) {
        cout << 0 << "\n";
        return;
    }

    // 内部区间范围
    int L = first_open + 1;
    int R = last_close - 1;
    if (L > R) {
        cout << 2 << "\n";
        return;
    }

    // 2. 预处理指针数组 (动态开辟，规避固定大小溢出)
    vector<vector<int>> nxt(n + 1, vector<int>(2, -1));
    for (int i = n - 1; i >= 0; --i) {
        nxt[i][0] = (s[i] == '(') ? i : nxt[i + 1][0];
        nxt[i][1] = (s[i] == ')') ? i : nxt[i + 1][1];
    }

    vector<vector<int>> prv(n + 1, vector<int>(2, -1));
    for (int i = 0; i < n; ++i) {
        prv[i][0] = (s[i] == '(') ? i : (i > 0 ? prv[i - 1][0] : -1);
        prv[i][1] = (s[i] == ')') ? i : (i > 0 ? prv[i - 1][1] : -1);
    }

    // 3. 区间 DP：dp[i][j] 表示在区间 [i, j] 中能取得的最大扩充长度
    // 使用递推避免递归爆栈
    vector<vector<int>> dp(n, vector<int>(n, 0));

    // 按区间长度从 4 递推至 (R - L + 1)
    for (int len = 4; len <= (R - L + 1); ++len) {
        for (int i = L; i + len - 1 <= R; ++i) {
            int j = i + len - 1;

            // 基础收缩转移
            int cur = max(dp[i + 1][j], dp[i][j - 1]);

            // 转移 1: 左边取 "()"，右边取 ")("
            int p1 = nxt[i][0]; // '('
            if (p1 != -1 && p1 + 1 <= j) {
                int p2 = nxt[p1 + 1][1]; // ')'
                int q2 = prv[j][0];      // '('
                if (q2 != -1 && q2 - 1 >= i) {
                    int q1 = prv[q2 - 1][1]; // ')'
                    if (p2 != -1 && q1 != -1 && p2 < q1) {
                        int inner = (p2 + 1 <= q1 - 1) ? dp[p2 + 1][q1 - 1] : 0;
                        cur = max(cur, 4 + inner);
                    }
                }
            }

            // 转移 2: 左边取 ")("，右边取 "()"
            p1 = nxt[i][1]; // ')'
            if (p1 != -1 && p1 + 1 <= j) {
                int p2 = nxt[p1 + 1][0]; // '('
                int q2 = prv[j][1];      // ')'
                if (q2 != -1 && q2 - 1 >= i) {
                    int q1 = prv[q2 - 1][0]; // '('
                    if (p2 != -1 && q1 != -1 && p2 < q1) {
                        int inner = (p2 + 1 <= q1 - 1) ? dp[p2 + 1][q1 - 1] : 0;
                        cur = max(cur, 4 + inner);
                    }
                }
            }

            dp[i][j] = cur;
        }
    }

    cout << 2 + dp[L][R] << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    if (cin >> t) {
        while (t--) {
            solve();
        }
    }
    return 0;
}
```
