```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int ksm(int n, int m){
    int res = 1;
    n %= mod;
    while(m){
        if(m & 1){
            res = res * n % mod;
        }
        n = n * n % mod;
        m = m >> 1;
    }
    return res;
}

const int N = 5e5 + 5;
int fac[N], inv[N];
int init = [](){
    fac[0] = inv[0] = 1;
    for(int i = 1; i < N; i++){
        fac[i] = fac[i - 1] * i % mod;
        inv[i] = ksm(fac[i], mod - 2);
    }
    return 0;
}();

int C(int n, int m){
    return fac[n] * inv[n - m] % mod * inv[m] % mod;
}
int A(int n, int m){
    return fac[n] * inv[m] % mod;
}

void ap() {
	int n; cin >> n;
    string s; cin >> s;
    
    int ans = fac[n];
    vector<int> a;
    for (int i = 0; i < 2 * n; i++) {
        if (s[i] == '(') {
            a.push_back(i);
            continue;
        }
        
        ans = ans * ksm((i - a.back() + 1) / 2, mod - 2) % mod;
        a.pop_back();
    }
    
    cout << ans << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
