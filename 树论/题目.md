# 一、

[E-Flower_Rainbow_and_Game_牛客小白月赛127](https://ac.nowcoder.com/acm/contest/126634/E)

```c++
int n;
int ans1 = 0, ans2 = 0;

vector<vector<int>> g;
vector<int> sub, dep;

void dfs(int i, int fa, int d) {
    sub[i] = 1;
    dep[i] = d;
    
    for (int x : g[i]) {
        if (x == fa) continue;
        
        dfs(x, i, d + 1);
        sub[i] += sub[x];
        
        int cnt = sub[x];
        int other = n - cnt - dep[x];
        ans1 += cnt * other;
    }
    
    ans2 += sub[i] - 1;
    return;
}

void ap() {
	cin >> n;
    
    g.assign(n, vector<int>());
    sub.assign(n, 0), dep.assign(n, 0);
    ans1 = 0, ans2 = 0;
    
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v;
        u--, v--;
        
        g[u].push_back(v);
        g[v].push_back(u);
    }
    
    dfs(0, -1, 0);
    
    cout << (ans1 + ans2) % MOD << endl;
    
    
	return ;
}
```
