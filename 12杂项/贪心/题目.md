
# 1. 贪心模拟

[P14258 好感（favor） - 洛谷](https://www.luogu.com.cn/problem/P14258#ide)

我们显然有个贪心的思路，就是我们在最后的位置翻转，前面的位置修改代价都会减少 1，但是如果最前面的是 1，我们就应该先处理，然后就做完了，用双端队列实现。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int f(deque<char> & dq) {
    int ans = 0;
    while (dq.size()) {
        if (dq.front() == '1') {
            dq.pop_front();
            dq.push_front('0');
            ans++;
        } else {
            while (dq.size() && dq.back() == '0') dq.pop_back();
            if (dq.empty()) break;
             
            ans += dq.size();
            dq.pop_back();
            if (dq.size()) dq.pop_front();
        }
    }
    return ans;
}

void ap() {
	int n; cin >> n;
    string s; cin >> s;

    deque<char> s1, s2;
    for (char c : s) {
        s1.push_back(c);

        if (c == '1') s2.push_back('0');
        else s2.push_back('1');
    }

    cout << min(f(s1), f(s2)) << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 2.前缀最大值序列


https://codeforces.com/gym/106567/problem/I

![](image/5449bb1f46c4a1244ca724dcffa7c7db.png)

数字大的`L` 能放的可能性越大
从右往左放，大的贪心的越长越好，遇到最左边0就break

如果遇到数量为1的数字，此时就需要将这个数字放到L的右上去
- 如果不能放，则后面的数字都小于这个数，一定不可以放到比其小的数字的右上
为此，记录可以放的位置数组为pos

最后
剩下的数字一定`<=` `L` 中的数


```c++
#include <bits/stdc++.h>
#include <cassert>

void solve() {
    int n;
    std::cin >> n;

    std::vector<int> c(2 * n);
    for (int i = 0; i < 2 * n; i++) {
        std::cin >> c[i];
        c[i]--;
    }

    std::vector<int> cnt(n);
    for (int i = 0; i < 2 * n; i++) {
        cnt[c[i]]++;
    }

    std::vector<int> ans(n, -1);
    int p = n - 1;

    std::vector<int> pos;
    for (int x = n - 1; x >= 0; x--) {
        if (cnt[x] > 1) {
            int k = std::max(0, p - cnt[x] + 2);
            ans[k] = x;
            cnt[x] -= p - k + 2;

            for (int i = k + 1; i <= p; i++) {
                pos.push_back(i);
            }

            p = k - 1;
            if (p == -1) {
                break;
            }
        } else if (cnt[x] == 1) {
            if (pos.empty()) {
                std::cout << "No\n";
                return;
            }
            ans[pos.back()] = x;
            pos.pop_back();
            cnt[x]--;
        }

    }

    if (p != -1) {
        std::cout << "No\n";
        return;
    }

    std::cout << "Yes\n";
    int x = 0;
    for (int i = 0; i < n; i++) {
        if (ans[i] == -1) {
            while (cnt[x] == 0) {
                x++;
            }
            ans[i] = x;
            cnt[x]--;
        }
    }

    for (int x = 0; x < n; x++) {
        assert(cnt[x] == 0);
    }

    for (int i = 0; i < n; i++) {
        std::cout << ans[i] + 1 << " \n"[i == n - 1];
    }
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int t;
    std::cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
```
