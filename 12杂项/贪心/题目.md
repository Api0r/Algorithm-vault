
# 1. 贪心模拟

[P14258 好感（favor） - 洛谷](https://www.luogu.com.cn/problem/P14258#ide)

我们显然有个贪心的思路，就是我们在最后的位置翻转，前面的位置修改代价都会减少 1，但是如果最前面的是 1，我们就应该先处理，然后就做完了，用双端队列实现。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int f(deque<char> & dq) {
    int ans = 0;
    while (dq.size()) {
        if (dq.front() == '1') {
            dq.pop_front();
            dq.push_front('0');
            ans++;
        } else {
            while (dq.size() && dq.back() == '0') dq.pop_back();
            if (dq.empty()) break;
             
            ans += dq.size();
            dq.pop_back();
            if (dq.size()) dq.pop_front();
        }
    }
    return ans;
}

void ap() {
	int n; cin >> n;
    string s; cin >> s;

    deque<char> s1, s2;
    for (char c : s) {
        s1.push_back(c);

        if (c == '1') s2.push_back('0');
        else s2.push_back('1');
    }

    cout << min(f(s1), f(s2)) << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
