
# 1. 贪心 + 模拟

[P14258 好感（favor） - 洛谷](https://www.luogu.com.cn/problem/P14258#ide)

我们显然有个贪心的思路，就是我们在最后的位置翻转，前面的位置修改代价都会减少 1，但是如果最前面的是 1，我们就应该先处理，然后就做完了，用双端队列实现。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

int f(deque<char> & dq) {
    int ans = 0;
    while (dq.size()) {
        if (dq.front() == '1') {
            dq.pop_front();
            dq.push_front('0');
            ans++;
        } else {
            while (dq.size() && dq.back() == '0') dq.pop_back();
            if (dq.empty()) break;
             
            ans += dq.size();
            dq.pop_back();
            if (dq.size()) dq.pop_front();
        }
    }
    return ans;
}

void ap() {
	int n; cin >> n;
    string s; cin >> s;

    deque<char> s1, s2;
    for (char c : s) {
        s1.push_back(c);

        if (c == '1') s2.push_back('0');
        else s2.push_back('1');
    }

    cout << min(f(s1), f(s2)) << endl;
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
```

# 2.前缀最大值序列


https://codeforces.com/gym/106567/problem/I

![](image/5449bb1f46c4a1244ca724dcffa7c7db.png)

数字大的`L` 能放的可能性越大
从右往左放，大的贪心的越长越好，遇到最左边0就break

如果遇到数量为1的数字，此时就需要将这个数字放到L的右上去
- 如果不能放，则后面的数字都小于这个数，一定不可以放到比其小的数字的右上
为此，记录可以放的位置数组为pos

最后
剩下的数字一定`<=` `L` 中的数


```c++
#include <bits/stdc++.h>
#include <cassert>

void solve() {
    int n;
    std::cin >> n;

    std::vector<int> c(2 * n);
    for (int i = 0; i < 2 * n; i++) {
        std::cin >> c[i];
        c[i]--;
    }

    std::vector<int> cnt(n);
    for (int i = 0; i < 2 * n; i++) {
        cnt[c[i]]++;
    }

    std::vector<int> ans(n, -1);
    int p = n - 1;

    std::vector<int> pos;
    for (int x = n - 1; x >= 0; x--) {
        if (cnt[x] > 1) {
            int k = std::max(0, p - cnt[x] + 2);
            ans[k] = x;
            cnt[x] -= p - k + 2;

            for (int i = k + 1; i <= p; i++) {
                pos.push_back(i);
            }

            p = k - 1;
            if (p == -1) {
                break;
            }
        } else if (cnt[x] == 1) {
            if (pos.empty()) {
                std::cout << "No\n";
                return;
            }
            ans[pos.back()] = x;
            pos.pop_back();
            cnt[x]--;
        }

    }

    if (p != -1) {
        std::cout << "No\n";
        return;
    }

    std::cout << "Yes\n";
    int x = 0;
    for (int i = 0; i < n; i++) {
        if (ans[i] == -1) {
            while (cnt[x] == 0) {
                x++;
            }
            ans[i] = x;
            cnt[x]--;
        }
    }

    for (int x = 0; x < n; x++) {
        assert(cnt[x] == 0);
    }

    for (int i = 0; i < n; i++) {
        std::cout << ans[i] + 1 << " \n"[i == n - 1];
    }
}

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    int t;
    std::cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
```


# 3. 反悔贪心

[C-小芳的比赛_牛客小白月赛139](https://ac.nowcoder.com/acm/contest/141882/C)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'

void ap() {
	int n, x; cin >> n >> x;
    vector<int> a(n);
    for (int & i : a) cin >> i;
    
    if (x >= n) {
        cout << 0 << endl;
        return;
    }
    
    int ans = 0;
    int cnt1 = 0, cnt2 = 0;
    priority_queue<int, vector<int>, greater<int>> pq;
    for (int & t : a) {
        cnt1++;
        pq.push(t);
        
        if (cnt1 - cnt2 > x) {
            ans += pq.top();
            pq.pop();
            
            cnt1--, cnt2++;
        }
    }
    
    if (cnt1 <= cnt2) {
        cout << -1 << endl;
    } else {
        cout << ans << endl;
    }
    
	return ;
}
```

# 4.观察 + 简单贪心

https://codeforces.com/contest/2268/problem/A

考虑固定点和其对称点的位置
- 固定点在其对称点右边，发现就是在当前的每两个对称的中取一个。取较大的即可
- 固定点在左边，发现固定点在左边时，固定点左边的数无法删掉，对称点右边的数也无法删掉，操作过程中，固定点不断靠近对称点，然后对称点移动到固定点的左边，变成上面的情况
- 发现第二种情况中，前缀和后缀总是无法删除，但是总是需要删除一个数字，因此，中间的数字总是会选上的。从反面来看更加明显，当然也可以正向证明

总结：
- 位于固定点右边，且位于对称点左边的数字无论怎么操作总是会被全删
- 当固定点位于对称点右边时，每次操作总是在对称的两个数中选一个，并且不会影响后续操作

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, k; cin >> n >> k;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<int> b;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (i >= k && i <= n - k + 1) {
            ans += a[i];
        } else {
            b.push_back(a[i]);
        }
    }

    int t = max(0LL, (int)b.size() - (k - 1));
    int left = 0, right = b.size() - 1;
    while (t--) {
        ans += max(b[left], b[right]);

        left++, right--;
    }

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 5. 不得不加时，贪心选前缀最大的

https://codeforces.com/gym/106799/problem/M

**题目大意**

给定 $n$ 个区间 $[l_i, r_i]$，从每个区间选出一个整数 $a_i$，构成序列 $\{a\}$。 若 $a_i$ 严格大于此前的所有元素，则称位置 $i$ 刷新了纪录（记录位置）。位置 1 总是刷新纪录。 求可以达到的**最少纪录次数**。 数据范围：$1 \le n \le 10^6$，$\sum n \le 10^6$，$-10^9 \le l_i \le r_i \le 10^9$。

**核心贪心策略**

该算法采用的是一种**区间贪心 + 动态维护前缀最大值**的策略，其核心思想可以分为以下几点：

**1.极值最大化原则**

- 每当选择一个新的极大值（纪录）位置时，都可以将该位置的值增大到区间右端点 $r_i$。
    
- **原因**：这样做不会改变它已经是极大值位置的事实，并且只会让后面的选择更容易（门槛拉高，后面遇到 $l_j > M$ 的概率更小，从而有机会不增加纪录）。因此，我们只需要考虑把极大值位置取为对应的 $r_i$。
    

**2. 扫描与状态维护**

- 设当前已经确定的最大值为 $M$（初始化为第一个区间的右端点，即 $M = a_0[1]$）。
    
- 从上一次确定答案之后开始向后扫描，并维护这一段内出现过的**最大右端点** $R$。
    

**3. 分类讨论**

对于当前遍历到的区间 $[l_i, r_i]$：

- **若** $l_i \le M$： 说明位置 $i$ 可以选择一个不超过 $M$ 的整数（例如直接选 $l_i$），因此**暂时不需要增加极大值位置**。此时只需要用当前的 $r_i$ 去更新并维护这一段的最大右端点 $R$（即 `r = max(r, a[i][1])`）。
    
- **若** $l_i > M$： 说明位置 $i$ **不可能**选择一个不超过 $M$ 的数。因此，从上一段开始到位置 $i$ 之间，任何方案都**必须至少增加一个极大值位置**。
    
    - **贪心选择**：我们贪心地选择其中右端点最大的区间作为这个极大值位置（即利用前面累积的最大右端点 $R$），令新的最大值 $M = R$，并将纪录数 `ans++`。

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<array<int, 2>> a(n);
    for (int i = 0; i < n; i++) {
        int l, r; cin >> l >> r;

        a[i] = {l, r};
    }

    int ans = 1, mx = a[0][1];
    int r = -1e18;
    for (int i = 1; i < n; i++) {
        if (a[i][0] > mx) {
            ans++;

            r = max(r, a[i][1]);
            mx = max(mx, r);

            // r = 0;

        } else {
            r = max(r, a[i][1]);
        }
    }

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 6. 反悔贪心 + 大根堆

https://qoj.ac/contest/4707/problem/25506

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, t; cin >> n >> t;
    vector<int> a(n),c(n + 1);
    for (int & i : a) cin >> i;
    for (int i = 1; i <= n; i++) cin >> c[i];

    vector<int> pre(n + 1);
    for (int i = 0; i < n; i++) {
        pre[i + 1] = pre[i] + a[i];
    }

    priority_queue<int> pq;

    int s = 0;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        if (pre[i] > t) break;

        pq.push(c[i]);
        s += c[i];

        while (pre[i] + s > t) {
            int x = pq.top();
            pq.pop();

            s -= x;
        }

        ans = max(ans, (int)pq.size());
    }

    cout << ans << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
