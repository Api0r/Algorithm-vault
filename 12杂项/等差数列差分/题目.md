# 2. 模板题

[P4231 三步必杀 - 洛谷](https://www.luogu.com.cn/problem/P4231#ide)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;

    vector<int> a(n + 3);
    
    for (int i = 0; i < m; i++) {
        int l, r, s, e; cin >> l >> r >> s >> e;

        int d = (e - s) / (r - l);

        a[l] += s;
        a[l + 1] += d - s;
        a[r + 1] -= d + e;
        a[r + 2] += e;
    }

    for (int i = 1; i <= n; i++) {
        a[i] += a[i - 1];
    }

    int mx = -1e18, sum = 0;
    for (int i = 1; i <= n; i++) {
        a[i] += a[i - 1];

        sum ^= a[i];
        mx = max(mx, a[i]);
    }

    cout << sum << ' ' << mx << endl;

    
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

```c++
void ap() {
	int n, m; cin >> n >> m;

    int base = 33000;
    vector<int> a(base + m + base);

    auto set = [&](int l, int r, int s, int e, int d) -> void {        
        a[l + base] += s;
        a[l + 1 + base] += d - s;
        a[r + 1 + base] -= d + e;
        a[r + 2 + base] += e;
    };
    
    for (int i = 0; i < n; i++) {
        int v, x; cin >> v >> x;

        set(x - 3 * v + 1, x - 2 * v, 1, v, 1);
		set(x - 2 * v + 1, x, v - 1, -v, -1);
		set(x + 1, x + 2 * v, -v + 1, v, 1);
		set(x + 2 * v + 1, x + 3 * v - 1, v - 1, 1, -1);
    }

    for (int i = 1; i <= m + base; i++) a[i] += a[i - 1];
    for (int i = 1; i <= m + base; i++) a[i] += a[i - 1];

    int start = base + 1;
    for (int i = 1; i <= m; i++) {
        cout << a[start++] << ' ';
    }
    
    cout << endl;
    
	return ;
}
```
