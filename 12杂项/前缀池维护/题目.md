```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));

    int sum = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];

            sum += a[i][j];
        }
    }

    for (int j = 0; j < m - 1; j++) {
        int pool = 0;

        for (int i = 0; i < n; i++) {
            pool += a[i][j];

            int take = min(a[i][j + 1], pool);
            pool -= take;

            sum -= take;
        }
    }

    cout << sum << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
