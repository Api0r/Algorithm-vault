```c++
	int q, w; cin >> q >> w;

    set<array<int, 4>> st;
    
    //time, id, t
    set<array<int, 3>> zip;

    map<int, int> has; // id, token

    int cnt = 0, sum = 0;

    auto f = [&]() -> void {
        while (sum > w && st.size()) {
            auto it = st.begin();
            auto [r, id, t, q] = *it;

            sum -= t;
            sum += (t + 1) / 2;

            zip.insert({-q, id, (t + 1) / 2});
            has[id] = (t + 1) / 2;

            st.erase(it);
        }

        while (sum > w) {
            auto it = zip.begin();
            auto [tim, id, t] = *it;

            sum -= t;
            has.erase(id);
            cnt--;

            zip.erase(it);
        }
    };

    while (q--) {
        string s; cin >> s;

        if (s == "ADD") {
            int id, t, r; cin >> id >> t >> r;

            st.insert({r, id, t, q});
            has[id] = t;

            cnt++;
            sum += t;

            if (sum > w) {
                f();
            }

        } else if (s == "ASK") {
            int id; cin >> id;

            if (has.count(id)) {
                cout << has[id] << endl;
            } else {
                cout << -1 << endl;
            }

        } else if (s == "STAT") {
            cout << cnt << ' ' << sum << endl;
```

# 3. set维护值

[C - Second Choice Election](https://atcoder.jp/contests/abc483/tasks/abc483_c)

```c++
#include <bits/stdc++.h>  
using namespace std;  
// #define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

bool cmp(const array<int, 2> & a, const array<int, 2> & b) {
    if (a[1] != b[1]) {
        return a[1] > b[1];
    }

    return a[0] < b[0];
}

void ap() {
	int n, m; cin >> n >> m;

    vector<int> cnt(n + 1);
    map<int, vector<int>> pos;

    for (int i = 0; i < m; i++) {
        int x, y; cin >> x >> y;

        cnt[x]++;
        pos[x].push_back(y);
    }
    
    vector<array<int, 2>> a(n);
    for (int i = 1; i <= n; i++) {
        a[i - 1] = {i, cnt[i]};
    }

    sort(a.begin(), a.end(), cmp);

    int ans1 = a[0][0], ans2 = a[1][0];
    for (int k = 1; k <= n; k++) {
        int ans = (k == ans1) ? ans2 : ans1;

        for (int x : pos[k]) {
            cnt[x]++;
            if (cnt[x] > cnt[ans]) {
                ans = x;
            } else if (cnt[x] == cnt[ans] && x < ans) {
                ans = x;
            }  
        }

        cout << ans << endl;

        for (int x : pos[k]) {
            cnt[x]--;
        }

    }

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
