#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<array<int, 4>> a(n);
    for (int i = 0; i < n; i++) {
        int x, y, z; cin >> x >> y >> z;
        
        a[i] = {x, y, z, i + 1};
    }
    
    sort(a.begin(), a.end());
    
    vector<int> finish(n + 1);
    for (int i = 1; i <= n; i++) {
        finish[i] = max(finish[i - 1], a[i - 1][0]) + a[i - 1][2];
    }
    
    vector<int> ans(n + 1);
    
    vector<int> last(10);
    for (int i = 1; i <= n; i++) {
        int j = i % 10;
        
        int get = max({a[i - 1][1], finish[i], last[j]});
        
        ans[a[i - 1][3]] = max(get - a[i - 1][1], 0LL);
        
        last[j] = get;
    }
    
    for (int i = 1; i <= n; i++) {
        cout << ans[i] << ' ';
    }
    cout << endl;
    
    
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
