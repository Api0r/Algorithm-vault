
# 1. 矩阵按反对角线分层

[L-Matrix_2026牛客暑期多校训练营5](https://ac.nowcoder.com/acm/contest/133880/L)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;
	vector<vector<int>> a(n + 1, vector<int> (m + 1));
	for (int i = 1; i <= n; i++) 
		for (int j = 1; j <= m; j++) 
			cin >> a[i][j];

	if (n == 1 || m == 1) {
		bool ok = true;
		int x = a[1][1];
		for (int i = 1; i <= n; i++) 
			for (int j = 1; j <= m; j++) 
				if (a[i][j] != x) {
					ok = false;
					break;
				}

		if (!ok) {
			cout << "-1\n";
		} else {
			cout << "0\n";
		}
		return;
	}
	
	
	int k = a[1][2] + a[2][1] - 2 * a[1][1];
	int x = a[1][1] + k;

	if (k < 0) {
		cout << "-1\n";
		return;
	}

	for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (x - a[i][j] < 0) {
                cout << "-1\n";
                return;
            }
        }
    }

	for (int s = 3; s <= n + m; s++) {
		int L = max(1LL, s - 1 - m), R = min(n, s - 1);

		int d = 0;
		for (int i = L; i <= R; i++) {
			int pre = s  - 1 - i;
			int cur = s - i;

			int b = (pre < 1 || pre > m) ? 0 : x - a[i][pre];
			int c = (cur < 1 || cur > m) ? 0 : x - a[i][cur];

			int left = c - d;
			if (left < 0) {
				cout << -1 << endl;
				return;
			}

			if (b - left < 0) {
				cout << -1 << endl;
				return;
			}

			d = b - left;
		}

		if (d != 0) {
			cout << -1 << endl;
			return;
		}

	}

	cout << k << endl;
	
	
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 2. 前缀池维护

[又一道纸笔问题 - 题目 - QOJ.ac](https://qoj.ac/contest/4582/problem/22910)

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, m; cin >> n >> m;
    vector<vector<int>> a(n, vector<int>(m));

    int sum = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> a[i][j];

            sum += a[i][j];
        }
    }

    for (int j = 0; j < m - 1; j++) {
        int pool = 0;

        for (int i = 0; i < n; i++) {
            pool += a[i][j];

            int take = min(a[i][j + 1], pool);
            pool -= take;

            sum -= take;
        }
    }

    cout << sum << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
