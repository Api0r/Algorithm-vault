```c++
void ap() {
	int n, k; cin >> n >> k;
    vector<int> b(n + 1);
    for (int i = 1; i <= n; i++) cin >> b[i];

    vector<int> cnt(k);
    for (int i = 1; i <= n; i++) {
        if (b[i] > k) continue;

        // cout << ((b[i] - i) % k + k) % k << ' ';
        cnt[((b[i] - i) % k + k) % k]++;
    }

    int mx = 0;
    for (int x : cnt) mx = max(mx, x);
    cout << n - mx << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    vector<int> cnt(4);
    for (int & x : a) cnt[x]++;

    int ans = 1e18;
    for (int i = 0; i < 4; i++) {//chaoxinag
        int t = 0;
        for (int j = 0; j < 4; j++) {
            t += cnt[j] * ((j - i + 4) % 4);
        }

        ans = min(ans, (4 - (t + i) % 4) % 4 + t);
    }

    cout << ans << endl;


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	}

}
```
