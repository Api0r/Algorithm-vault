
# 1.

[D - Range Set Insertion Query](https://atcoder.jp/contests/abc478/tasks/abc478_d)

将 `N` 集 `S1​,S2​,…,SN​ `视为时间轴，问题可解释如下：

> 在集合 S 上执行 Q 个操作。其中 i 个操作是 "从时间 Li​ 到时间 Ri​+21​ 将 X 插入 X "。求时间为 t 时 S 中元素的个数。 `(1≤t≤N) .`

这可以用事件排序的方式来解决。具体来说，管理要求插入 `Xi​ `的元素个数。当计数从 0 变为 1 ，或从 1 变为 0 时，更新答案。



```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;


void ap() {
    int n, q; cin >> n >> q;

	vector<array<int, 3>> a;
    for (int i = 0; i < q; i++) {
        int l, r, x; cin >> l >> r >> x;
        l--; x--;

        a.push_back({l, x, 1});
        a.push_back({r, x, -1});
    }
    a.push_back({n, 0, 0});

    sort(a.begin(), a.end());

    int now = 0;
    int ans = 0;
    vector<int> cnt(q + 1, 0);
    for (auto [t, x, v] : a) {
        while (now < t) {
            now++;
            cout << ans << ' ';
        }

        if (cnt[x] == 0) ans++;
        cnt[x] += v;
        if (cnt[x] == 0) ans--;

    }

    cout << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n, q; cin >> n >> q;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    vector<int> ver(n + 1, 0);

    int cnt = 0;

    using p = array<int, 3>;//last_time, idx, version
    priority_queue<p, vector<p>, greater<p>> pq;
    for (int i = 1; i <= n; i++) {
        if (a[i] > 0) {
            cnt++;
            pq.push({a[i], i, 0});
        }
    }

    for (int t = 1; t <= q; t++) {
        int idx, v; cin >> idx >> v;

        //last_time <= t. pop
        while (!pq.empty() && pq.top()[0] <= t) {
            auto p = pq.top();
            pq.pop();

            if (p[2] == ver[p[1]]) cnt--;
        }
        
        if (idx != 0 && v != 0) {
            //a[idx] = max(a[idx], v);
            //a[idx],bubian
            //v, t + v
            if (t + v > a[idx]) {
                if (a[idx] <= t) {
                    cnt++;
                }

                a[idx] = t + v;
                ver[idx]++;
                pq.push({a[idx], idx, ver[idx]});
            }
        }

        cout << cnt << endl;
    }

    for (int i = 1; i <= n; i++) {
        cout << max(0LL, a[i] - q) << ' ';
    }
    cout << endl;



	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
//	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
