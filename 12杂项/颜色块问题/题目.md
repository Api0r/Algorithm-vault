
# 1.区间反转贡献 + DP + 逆序对

https://codeforces.com/gym/106567/problem/J

- 翻转一个区间 $[l, r]$ 时，区间外部的相对位置不变，仅区间内部的点对被完全倒置。
    
- 对于区间内的任意数对 $i < j$：
    
    - 若原先 $p_i > p_j$（构成逆序对），翻转后相对顺序颠倒，逆序对数量减少 $1$；
        
    - 若原先 $p_i < p_j$（构成顺序对），翻转后变为逆序对，逆序对数量增加 $1$。
        
- 设翻转 $[l, r]$ 相比原来增加的逆序对数为 $\Delta(l, r)$，则最终答案为：
    
    $$\text{原逆序对数} + \max(0, \max_{l \le r} \Delta(l, r))$$




- 考察从区间 $[l+1, r]$ 扩展到 $[l, r]$ 的变化：
    
    新加入的数对恰好是以 $l$ 为左端点、$j \in [l+1, r]$ 为右端点的所有点对。
    
- 因此有递推式：
    
    $$\Delta(l, r) = \Delta(l+1, r) + \sum_{j=l+1}^r \Big([p_l < p_j] - [p_l > p_j]\Big)$$
    
- 令 $dp[r] = \Delta(l, r)$。让 $l$ 从 $n$ 递减到 $1$，同时用一个变量 `cur` 在 $r$ 从 $l+1$ 到 $n$ 遍历时累加对 $p_l$ 的增量贡献，即可在 $O(n)$ 空间内原地更新：
    
    $$dp[r] = dp[r] + cur$$

`dp[r]` 表示以`r`为右端点的变化量
从右到左枚举`L`，加入新的`L`时`dp[r]` 存的信息是`[L + 1, r]` 区间内的变化量，其中`L + 1 <= r <= n`
要更新`[L, r]` 区间的变化量，只需要更新`L`所新加的变化，即从枚举`r` 从`[L + 1, n]`，每次累加，对于`r`就更新对应的`dp` 信息 


```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

struct Fenwick {
    int n;
    vector<int> tree;
    Fenwick(int n) : n(n), tree(n + 1, 0) {}

    void add(int i, int delta) {
        for (; i <= n; i += i & -i) tree[i] += delta;
    }

    int query(int i) {
        int sum = 0;
        for (; i > 0; i -= i & -i) sum += tree[i];
        return sum;
    }
};

void ap() {
	int n; cin >> n;
    vector<int> a(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];

    Fenwick tree(n);

    int s = 0;
    for (int i = 1; i <= n; i++) {
        s += (i - 1 - tree.query(a[i]));
        tree.add(a[i], 1);
    }

    vector<int> dp(n + 1, 0);

    int mx = 0;
    for (int l = n - 1; l >= 1; l--) {
        dp[l] = 0;
        int cur = 0;
        for (int r = l + 1; r <= n; r++) {
            if (a[r] > a[l]) {
                cur++;
            } else if (a[r] < a[l]) {
                cur--;
            }
            dp[r] += cur;
            mx = max(mx, dp[r]);
        }

    }

    cout << mx + s << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
