# 2. i , i + 2 - swap

https://codeforces.com/contest/2267/problem/D

奇数下标的每个数可以任意排序，同理偶数下标的数也可以任意排序

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    if (n == 1 || n == 2) {
        cout << "YES" << endl;
        return;
    }

    vector<array<int, 2>> b(n);
    for (int i = 0; i < n; i++) {
        b[i] = {a[i], (i & 1)};
    }

    sort(b.rbegin(), b.rend());

    int cnt0 = 0, cnt1 = 0;
    for (int i = 0; i < n; i++) {
        if (b[i][1] & 1) {
            cnt1++;
        } else {
            cnt0++;
        }

        if (abs(cnt1 - cnt0) > 1) {
            cout << "NO" << endl;
            return;
        }
    }

    cout << "YES" << endl;


	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 4.

https://codeforces.com/contest/2254/problem/C1

`Notice that the operation can be treated as swapping ai with ai+2, this works since the operation does not change the middle element, and only changes the first one and the third one.`

`Because an element can only move in steps of two, it can never change the parity of its index. Any character at an odd position will always remain at an odd position, and the same applies for even indices.`

`Since we have the power to swap the characters at odd indices endlessly (as many times as we want), we can rearrange them into any arrangement we want, independently of the characters at even indices. Therefore, the elements at odd positions in a can match the elements at odd positions in b if they have the same number of ones (or zeros). The same can be applied for even positions.`

因为我们可以swap奇数indices的字符任意次，所以我们可以将其重排为任意的想要的arrangement

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
	string a, b; cin >> a >> b;

		cout << "YES" << endl;

	}

		cout << "NO" << endl;
		return;
	}

	array<int, 2> cnt1{0, 0}, cnt2{0, 0};
	for (int i = 0; i < n; i++) {
		if (a[i] == '1') cnt1[i % 2]++;
		if (b[i] == '1') cnt2[i % 2]++;

		cout << "NO" << endl;
		return;
	}
```

# 5.转换 + 普通任意swap - 任排

https://codeforces.com/contest/2278/problem/E

我们定义一个新数组 $b$，满足 $b_i = (-1)^{i-1}a_i$。

现在，让我们看一下题目给定的操作会如何影响这个新数组 $b$。如果我们在下标 $i$ 处执行该操作，数组 $a$ 中的新值会变为 $a'_i = -a_{i+1}$ 以及 $a'_{i+1} = -a_i$。观察 $b$ 中对应的元素：

$$b'_{i+1} = (-1)^i a'_{i+1} = (-1)^i (-a_i) = (-1)^{i-1} a_i = b_i$$

$$b'_i = (-1)^{i-1} a'_i = (-1)^{i-1} (-a_{i+1}) = (-1)^i a_{i+1} = b_{i+1}$$

对数组 $a$ 的操作恰好等价于**交换数组 $b$ 中相邻的两个元素**！


由于我们可以任意次交换相邻元素，因此我们可以将 $b$ 重排成任意我们想要的排列。这就是本题的核心思想。剩下的就只是简单的数学推导了。现在，让我们用 $b$ 来表示数组 $a$ 的总和：

  

$$\sum_{i=1}^n a_i = b_1 - b_2 + b_3 - b_4 + \dots$$

  

为了使这个总和最大化，我们采用贪心策略：将数组 $b$ 中最小的 $\lfloor \frac{n}{2} \rfloor$ 个元素放在偶数下标位置，将最大的 $\lceil \frac{n}{2} \rceil$ 个元素放在奇数下标位置。最终得到的数组 $b$ 的交错和即为所求答案。

  

时间复杂度：$\mathcal{O}(n \log n)$。`

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n);
    for (int & i : a) cin >> i;

    int c = 1;
    for (int i = 0; i < n; i++) {
        a[i] = a[i] * c;
        c *= -1;
    }

    sort(a.begin(), a.end());

    int s = 0;

    int t = n / 2;
    int i = 0;
    for (; i < t; i++) {
        s -= a[i];
    }

    for (; i < n; i++) {
        s += a[i];
    }

    cout << s << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```

# 6. 连通块 + 相对位置

https://codeforces.com/gym/106783/problem/I

- 余数为0的相邻的数可以任意swap，即任排
- 余数为1，2的相邻的数构成一个连通块
- 发现余数为1，2的连通块里边的1的相对位置无法改变，同理2的相对位置也无法改变

总结：
- 余数为0的相邻的数构成一个连通块，余数为1，2的相邻的数构成一个连通块
- 0可以任排，可以排序比较
- 1.2相对位置不变，即分类放到各自类别的数组中，最后比较即可

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n), b(n);
    for (int & i : a) cin >> i;
    for (int & i : b) cin >> i;

    for (int i = 0; i < n; ) {
        int j = i;

        if (a[j] % 3 == 0) {
            vector<int> s, t;
            while (j < n && a[j] % 3 == 0) {
                s.push_back(a[j]);
                t.push_back(b[j]);
                j++;
            }

            sort(s.begin(), s.end());
            sort(t.begin(), t.end());

            if (s != t) {
                cout << "No" << endl;
                return;
            }
        } else {
            vector<int> s1, s2, t1, t2;
            while (j < n && a[j] % 3 != 0) {
                if (b[j] % 3 == 0) {
                    cout << "No" << endl;
                    return;
                }

                if (a[j] % 3 == 1) s1.push_back(a[j]);
                else s2.push_back(a[j]);

                if (b[j] % 3 == 1) t1.push_back(b[j]);
                else t2.push_back(b[j]);

                j++;
            }

            if (s1 != t1 || s2 != t2) {
                cout << "No" << endl;
                return;
            }
        }

        i = j;
    }

    cout << "Yes" << endl;

	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
