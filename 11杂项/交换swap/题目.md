
# 1. X - swap

```c++
#include <bits/stdc++.h>  
using namespace std;  
#define int long long
#define endl '\n'
const int mod = 1e9 + 7;
const int MOD = 998244353;

void ap() {
	int n; cin >> n;
    vector<int> a(n), b(n), c(n), d(n);
    for (int & i : a) cin >> i;
    for (int & i : b) cin >> i;
    for (int & i : c) cin >> i;
    for (int & i : d) cin >> i;
    
    set<array<int, 2>> st, t;
    for (int i = 0; i < n; i++) {
        st.insert({c[i], d[i]}), st.insert({d[i], c[i]});
        t.insert({c[i], d[i]});
    }
    
    int cnt1 = 0, cnt2 = 0, cnt3 = 0;
    for (int i = 0; i < n ; i++) {
        if (!st.count({a[i], b[i]}) && !st.count({b[i], a[i]})) {
            cout << "No" << endl;
            return;
        }
        
        if (a[i] == d[i] && b[i] == c[i]) {
            cnt3++;
        } else if (t.count({a[i], b[i]})) {
            cnt2++;
        } else if (t.count({b[i], a[i]})) {
            cnt1++;
        }
        
    }
    
//     cout << cnt1 << ' ' << 2 * cnt2 << ' ' << 3 * cnt3 << endl;
    
    if ((cnt1 + 2 * cnt2 + 3 * cnt3) % 2 == 0) {
        cout << "Yes" << endl;
    } else {
        cout << "No" << endl;
    }
    
	return ;
}

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int T = 1;
	cin >> T;
	while (T--) {
		ap();
	} 
    return 0;
}
```
